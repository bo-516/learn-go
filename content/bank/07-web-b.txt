=== web-12-1
title: trace ID 中间件
--- desc
实现全链路追踪的入口：

- `Trace(next)`：从请求头 `X-Request-ID` 读取 ID；没有时用 `crypto/rand` 生成 8 个随机字节，编码成 16 位十六进制字符串。把 ID 放进请求的 context，并写回响应头 `X-Request-ID`；
- `TraceID(ctx)`：从 context 中取出 ID，没有时返回空字符串；
- `Inject(ctx, req)`：调用下游时，把 ctx 里的 ID 设置到下游请求的 `X-Request-ID` 头上。
--- solution
package main

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"net/http"
)

type traceKey struct{}

func newID() string {
	b := make([]byte, 8)
	rand.Read(b)
	return hex.EncodeToString(b)
}

func Trace(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		// @solution
		id := r.Header.Get("X-Request-ID")
		if id == "" {
			id = newID()
		}
		w.Header().Set("X-Request-ID", id)
		next.ServeHTTP(w, r.WithContext(context.WithValue(r.Context(), traceKey{}, id)))
		// @starter
		// next.ServeHTTP(w, r) // TODO
		// @end
	})
}

func TraceID(ctx context.Context) string {
	// @solution
	id, _ := ctx.Value(traceKey{}).(string)
	return id
	// @starter
	// return "" // TODO
	// @end
}

func Inject(ctx context.Context, req *http.Request) {
	// @solution
	if id := TraceID(ctx); id != "" {
		req.Header.Set("X-Request-ID", id)
	}

title: 让 slog 自动带上 trace_id
--- desc
`log/slog`（Go 1.21+）的 Handler 可以包装。实现 `TraceHandler`：嵌入一个 `slog.Handler`，只重写 `Handle` 方法——如果 ctx 里有 trace ID，就给日志记录追加属性 `trace_id`，再交给内层 Handler。

这样业务代码只要用 `logger.InfoContext(ctx, …)`，每条日志都会自动带上 trace_id。

注意：嵌入接口时还要重写 `WithAttrs` 和 `WithGroup`，否则 `logger.With(...)` 返回的 Handler 会丢掉这层包装。
--- solution
package main

import (
	"context"
	"log/slog"
)

type traceKey struct{}

func WithTraceID(ctx context.Context, id string) context.Context {
	return context.WithValue(ctx, traceKey{}, id)
}

type TraceHandler struct {
	slog.Handler
}

// @solution
func (h TraceHandler) Handle(ctx context.Context, r slog.Record) error {
	if id, ok := ctx.Value(traceKey{}).(string); ok {
		r.AddAttrs(slog.String("trace_id", id))
	}
	return h.Handler.Handle(ctx, r)
}

func (h TraceHandler) WithAttrs(attrs []slog.Attr) slog.Handler {
	return TraceHandler{h.Handler.WithAttrs(attrs)}
}

func (h TraceHandler) WithGroup(name string) slog.Handler {
	return TraceHandler{h.Handler.WithGroup(name)}
}

--- desc
实现 `Save(r, w, max)`：把上传内容从 `r` **流式**写入 `w`，同时计算 SHA-256，返回写入的字节数和十六进制哈希。超过 `max` 字节时返回 `ErrTooLarge`。

不能先 `io.ReadAll` 把整个文件读进内存：测试的 reader 会统计被读取了多少字节，超过限制后不应该继续读取整个文件。

提示：`io.LimitReader(r, max+1)` 多读一个字节就能判断是否超限；`io.MultiWriter(w, hasher)` 一份数据同时写到两个地方。
--- solution
package main

import (
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"io"
)

var ErrTooLarge = errors.New("file too large")

func Save(r io.Reader, w io.Writer, max int64) (int64, string, error) {
	// @solution
	h := sha256.New()
	n, err := io.Copy(io.MultiWriter(w, h), io.LimitReader(r, max+1))
	if err != nil {
		return n, "", err
	}
	if n > max {
		return n, "", ErrTooLarge
	}
	return n, hex.EncodeToString(h.Sum(nil)), nil
	// @starter
	// data, err := io.ReadAll(r) // TODO: 整个文件读进内存
	// if err != nil {
	// 	return 0, "", err
	// }
	// if int64(len(data)) > max {
	// 	return 0, "", ErrTooLarge
	// }
	// n, err := w.Write(data)
	// sum := sha256.Sum256(data)
	// return int64(n), hex.EncodeToString(sum[:]), err
	// @end
}

package main

import (
	"bytes"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"io"
	"strings"
	"testing"
)

type countingReader struct {
	r    io.Reader
	read int64
}

func (c *countingReader) Read(p []byte) (int, error) {
	n, err := c.r.Read(p)
	c.read += int64(n)
	return n, err
}

func TestSave(t *testing.T) {
	content := strings.Repeat("hello gopher ", 1000)
	var out bytes.Buffer
	n, sum, err := Save(strings.NewReader(content), &out, 1<<20)
	want := sha256.Sum256([]byte(content))
	if err != nil || n != int64(len(content)) || sum != hex.EncodeToString(want[:]) || out.String() != content {
		t.Fatalf("n=%d sum=%s err=%v", n, sum, err)
	}
}

func TestTooLarge(t *testing.T) {
	cr := &countingReader{r: bytes.NewReader(make([]byte, 8<<20))}
	_, _, err := Save(cr, io.Discard, 1<<20)
	if !errors.Is(err, ErrTooLarge) {
		t.Fatalf("超过限制应返回 ErrTooLarge，实际 %v", err)
	}
	if cr.read > 1<<20+64<<10 {
		t.Fatalf("超限后仍然读取了 %d 字节，应该流式处理并及时停止", cr.read)
	}
}

func TestExactLimit(t *testing.T) {
	if _, _, err := Save(strings.NewReader("12345"), io.Discard, 5); err != nil {
		t.Fatalf("正好等于上限应成功，实际 %v", err)
	}
}

=== web-13-2
title: 分片上传与断点续传
--- desc
大文件切成固定数量的分片上传，中断后只补传缺失的分片。实现 `Upload`：

- `NewUpload(total, hash)`：共 `total` 个分片，`hash` 是整个文件的 SHA-256（十六进制）；
- `Put(i, data)`：保存第 i 个分片（从 0 开始），下标越界返回错误，重复上传覆盖；
- `Missing()`：返回还没上传的分片下标（升序）——客户端续传前先查询它；
- `Assemble()`：有缺失返回错误；按顺序拼接后校验哈希，不一致返回错误。
--- solution
package main

import (
	"bytes"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"fmt"
)

type Upload struct {
	total  int
	hash   string
	chunks map[int][]byte
}

func NewUpload(total int, hash string) *Upload {
	return &Upload{total: total, hash: hash, chunks: map[int][]byte{}}
}

func (u *Upload) Put(i int, data []byte) error {
	// @solution
	if i < 0 || i >= u.total {
		return fmt.Errorf("chunk %d out of range", i)
	}
	u.chunks[i] = bytes.Clone(data)
	return nil
	// @starter
	// return errors.New("TODO")
	// @end
}

func (u *Upload) Missing() []int {
	// @solution
	var out []int
	for i := 0; i < u.total; i++ {
		if _, ok := u.chunks[i]; !ok {
			out = append(out, i)
		}
	}
	return out
	// @starter
	// return nil // TODO
	// @end
}

func (u *Upload) Assemble() ([]byte, error) {
	// @solution
	if m := u.Missing(); len(m) > 0 {
		return nil, fmt.Errorf("missing chunks %v", m)
	}
	var buf bytes.Buffer
	for i := 0; i < u.total; i++ {
		buf.Write(u.chunks[i])
	}
	sum := sha256.Sum256(buf.Bytes())
	if hex.EncodeToString(sum[:]) != u.hash {
		return nil, errors.New("hash mismatch")
	}
	return buf.Bytes(), nil
	// @starter
	// return nil, errors.New("TODO")
	// @end
}

=== web-14-1
title: 支持继承和通配符的 RBAC
--- desc
实现基于角色的权限控制：

- `Grant(role, perms...)`：给角色授予权限，权限格式为 `资源:操作`，如 `order:read`；`order:*` 表示 order 的所有操作，`*` 表示所有权限；
- `Inherit(child, parent)`：子角色拥有父角色的全部权限，可以多层继承，配置错误时**可能出现环**，不能死循环；
- `Assign(user, role)`：给用户分配角色（可以多个）；
- `Can(user, perm)`：用户是否拥有某个权限。
--- solution
package main

import "strings"

type RBAC struct {
	perms   map[string][]string
	parents map[string][]string
	users   map[string][]string
}

func NewRBAC() *RBAC {
	return &RBAC{perms: map[string][]string{}, parents: map[string][]string{}, users: map[string][]string{}}
}

func (r *RBAC) Grant(role string, perms ...string) { r.perms[role] = append(r.perms[role], perms...) }
func (r *RBAC) Inherit(child, parent string)      { r.parents[child] = append(r.parents[child], parent) }
func (r *RBAC) Assign(user, role string)          { r.users[user] = append(r.users[user], role) }

func match(granted, want string) bool {
	// @solution
	if granted == "*" || granted == want {
		return true
	}
	res, act, ok := strings.Cut(granted, ":")
	wres, _, _ := strings.Cut(want, ":")
	return ok && act == "*" && res == wres
	// @starter
	// return granted == want // TODO: 通配符
	// @end
}

func (r *RBAC) Can(user, perm string) bool {
	// @solution
	seen := map[string]bool{}
	stack := append([]string(nil), r.users[user]...)
	for len(stack) > 0 {
		role := stack[len(stack)-1]
		stack = stack[:len(stack)-1]
		if seen[role] {
			continue
		}
		seen[role] = true
		for _, p := range r.perms[role] {
			if match(p, perm) {
				return true
			}
		}
		stack = append(stack, r.parents[role]...)
	}
	return false
	// @starter
	// for _, role := range r.users[user] { // TODO: 没有处理继承
	// 	for _, p := range r.perms[role] {
	// 		if match(p, perm) {
	// 			return true
	// 		}
	// 	}
	// }
	// return false
	// @end
}

=== web-15-1
title: 配置热更新：校验后整体替换
race: true
--- desc
实现配置管理器 `Manager`：

- `NewManager(source)`：从 `source()` 读取 JSON 初始化，初始加载失败返回错误；
- `Current()`：无锁读取当前配置（`atomic.Pointer`）；
- `Reload()`：重新读取：内容和上次相同返回 `(false, nil)`；解析失败或 `Validate()` 不通过，**保留旧配置**并返回错误；否则整体替换，返回 `(true, nil)`。

读取方拿到的永远是一个完整的配置对象，不会看到更新了一半的状态。
--- solution
package main

import (
	"bytes"
	"encoding/json"
	"errors"
	"sync"
	"sync/atomic"
)

type Config struct {
	RateLimit int      `json:"rate_limit"`
	Features  []string `json:"features"`
}

func (c *Config) Validate() error {
	if c.RateLimit < 1 || c.RateLimit > 10000 {
		return errors.New("rate_limit must be in [1, 10000]")
	}
	return nil
}

type Manager struct {
	cur     atomic.Pointer[Config]
	source  func() ([]byte, error)
	mu      sync.Mutex // 串行化 Reload
	lastRaw []byte
}

func parse(raw []byte) (*Config, error) {
	var c Config
	if err := json.Unmarshal(raw, &c); err != nil {
		return nil, err
	}
	if err := c.Validate(); err != nil {
		return nil, err
	}
	return &c, nil
}

func NewManager(source func() ([]byte, error)) (*Manager, error) {
	m := &Manager{source: source}
	if _, err := m.Reload(); err != nil {
		return nil, err
	}
	return m, nil
}

func (m *Manager) Current() *Config { return m.cur.Load() }

func (m *Manager) Reload() (bool, error) {
	// @solution
	m.mu.Lock()
	defer m.mu.Unlock()
	raw, err := m.source()
	if err != nil {
		return false, err
	}
	if m.lastRaw != nil && bytes.Equal(raw, m.lastRaw) {
		return false, nil
	}
	cfg, err := parse(raw)
	if err != nil {
		return false, err
	}
	m.cur.Store(cfg)
	m.lastRaw = raw
	return true, nil
	// @starter
	// raw, err := m.source()
	// if err != nil {
	// 	return false, err
	// }
	// var c Config
	// json.Unmarshal(raw, &c) // TODO: 没有处理错误和校验，也没有判断内容是否变化
	// m.cur.Store(&c)
	// return true, nil
	// @end
}

type Pool struct {

	idle           map[string][]int
	nextID         int
	Dialed         int
	Closed         int
}

func NewPool(maxIdlePerHost int) *Pool {
	return &Pool{maxIdlePerHost: maxIdlePerHost, idle: map[string][]int{}}

func (p *Pool) Get(host string) int {

	if conns := p.idle[host]; len(conns) > 0 {
		c := conns[len(conns)-1]
		p.idle[host] = conns[:len(conns)-1]
		return c
	}
	p.nextID++
	p.Dialed++
	return p.nextID

title: 零分配地生成 JSON
--- desc
在 QPS 很高的接口上，`encoding/json` 的反射和分配可能成为热点。为固定结构手写序列化：

`AppendJSON(dst, u)` 把 `User` 按 `{"id":1,"name":"bob","admin":false}` 的格式追加到 `dst` 后面。字符串转义规则和 `encoding/json` 一致（本题不涉及 `<>&` 等 HTML 字符）：

- `"` → `\"`，`\` → `\\`；
- `\n`、`\r`、`\t` → `\n`、`\r`、`\t`；
- 其他小于 0x20 的控制字符 → `\u00XX`（小写十六进制）；
- 其余字符（包括中文）原样输出。

调用方提供足够容量的 `dst` 时必须**零分配**。
--- solution
package main

import "strconv"

type User struct {
	ID    int64
	Name  string
	Admin bool
}

const hexDigits = "0123456789abcdef"

func appendString(dst []byte, s string) []byte {
	dst = append(dst, '"')
	// @solution
	for i := 0; i < len(s); i++ {
		c := s[i]
		switch {
		case c == '"' || c == '\\':
			dst = append(dst, '\\', c)
		case c == '\n':
			dst = append(dst, '\\', 'n')
		case c == '\r':
			dst = append(dst, '\\', 'r')
		case c == '\t':
			dst = append(dst, '\\', 't')
		case c < 0x20:
			dst = append(dst, '\\', 'u', '0', '0', hexDigits[c>>4], hexDigits[c&0xf])
		default:
			dst = append(dst, c)
		}
	}
	// @starter
	// dst = append(dst, s...) // TODO: 转义
	// @end
	return append(dst, '"')
}

func AppendJSON(dst []byte, u User) []byte {
	dst = append(dst, `{"id":`...)
	dst = strconv.AppendInt(dst, u.ID, 10)
	dst = append(dst, `,"name":`...)
	dst = appendString(dst, u.Name)
	dst = append(dst, `,"admin":`...)
	dst = strconv.AppendBool(dst, u.Admin)
	return append(dst, '}')
}

=== web-20-1
title: 校验 Webhook 签名并防重放
--- desc
回调接口必须验证请求确实来自对方。约定请求头格式为 `t=<unix秒>,v1=<十六进制签名>`，签名内容是 `<t>.<body>`，算法 HMAC-SHA256。密钥轮换期间可能出现多个 `v1=`，任意一个匹配即可。

实现 `Verify(body, header, secret, now, tolerance)`：

- 头部格式错误（缺少 `t` 或 `v1`、时间戳不是整数）→ `ErrBadHeader`；
- 时间戳和 `now` 相差超过 `tolerance`（过去或未来都算）→ `ErrExpired`，防止旧请求被重放；
- 没有任何签名匹配 → `ErrSignature`；比较时用 `hmac.Equal`（常量时间）。

`Sign` 已经实现，供测试生成合法的头部。
--- solution
package main

import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"strconv"
	"strings"
	"time"
)

var (
	ErrBadHeader = errors.New("bad signature header")
	ErrExpired   = errors.New("timestamp outside tolerance")
	ErrSignature = errors.New("signature mismatch")
)

func mac(secret []byte, ts string, body []byte) []byte {
	m := hmac.New(sha256.New, secret)
	m.Write([]byte(ts + "."))
	m.Write(body)
	return m.Sum(nil)
}

func Sign(body, secret []byte, t time.Time) string {
	ts := strconv.FormatInt(t.Unix(), 10)
	return "t=" + ts + ",v1=" + hex.EncodeToString(mac(secret, ts, body))
}

func Verify(body []byte, header string, secret []byte, now time.Time, tolerance time.Duration) error {
	// @solution
	var ts string
	var sigs [][]byte
	for _, part := range strings.Split(header, ",") {
		k, v, ok := strings.Cut(strings.TrimSpace(part), "=")
		if !ok {
			continue
		}
		switch k {
		case "t":
			ts = v
		case "v1":
			if b, err := hex.DecodeString(v); err == nil {
				sigs = append(sigs, b)
			}
		}
	}
	sec, err := strconv.ParseInt(ts, 10, 64)
	if err != nil || len(sigs) == 0 {
		return ErrBadHeader
	}
	if d := now.Sub(time.Unix(sec, 0)); d > tolerance || d < -tolerance {
		return ErrExpired
	}
	expected := mac(secret, ts, body)
	for _, s := range sigs {
		if hmac.Equal(s, expected) {
			return nil
		}
	}
	return ErrSignature
	// @starter
	// return nil // TODO
	// @end
}

		{"/users/1", "application/vnd.shop.v2+json", 2},
		{"/users/1", "application/json", 1},
		{"/users/1", "", 1},
	}
	for _, c := range cases {
		r := httptest.NewRequest("GET", c.path, nil)
		if c.accept != "" {
			r.Header.Set("Accept", c.accept)
		}
		if got := Version(r); got != c.want {
			t.Errorf("Version(%s, Accept=%q) = %d，期望 %d", c.path, c.accept, got, c.want)
		}
	}
}

func TestRenderUser(t *testing.T) {
	u := User{"Ada", "Lovelace"}
	v1, _ := RenderUser(1, u)
	v2, _ := RenderUser(2, u)
	if string(v1) != `{"name":"Ada Lovelace"}` {
		t.Errorf("v1 = %s", v1)
	}
	if string(v2) != `{"first_name":"Ada","last_name":"Lovelace"}` {
		t.Errorf("v2 = %s", v2)
	}
}
