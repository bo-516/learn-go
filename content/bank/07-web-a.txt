=== web-01-1
title: 实现一个 http.Handler
--- desc
实现 `HelloHandler` 的 `ServeHTTP` 方法：

- 只接受 `GET`，其他方法返回 `405`，并设置响应头 `Allow: GET`；
- 从查询参数 `name` 读取名字，没有时默认 `world`；
- 返回 `200`，`Content-Type: text/plain; charset=utf-8`，响应体 `hello, <name>`。

任何实现了 `ServeHTTP(http.ResponseWriter, *http.Request)` 的类型都是一个 `http.Handler`。
--- solution
package main

import (
	"fmt"
	"net/http"
)

type HelloHandler struct{}

func (HelloHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
	// @solution
	if r.Method != http.MethodGet {
		w.Header().Set("Allow", http.MethodGet)
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
		return
	}
	name := r.URL.Query().Get("name")
	if name == "" {
		name = "world"
	}
	w.Header().Set("Content-Type", "text/plain; charset=utf-8")
	fmt.Fprintf(w, "hello, %s", name)
	// @starter
	// // TODO
	// @end
}

=== web-02-1
title: 组合标准库中间件
--- desc
中间件就是「接收一个 Handler、返回一个新 Handler」的函数。实现：

`Chain(h, mws...)`：依次包装，保证 `mws[0]` 在**最外层**（最先执行）。测试里提供了一个会记录执行顺序的 `tag` 中间件。

调用 `Chain(h, A, B)` 后，执行顺序应该是 `A 前 → B 前 → handler → B 后 → A 后`。
--- solution
package main

import "net/http"

type Middleware func(http.Handler) http.Handler

func Chain(h http.Handler, mws ...Middleware) http.Handler {
	// @solution
	for i := len(mws) - 1; i >= 0; i-- {
		h = mws[i](h)
	}
	return h
	// @starter
	// return h // TODO
	// @end
}
--- test
package main

import (
	"net/http"
	"net/http/httptest"
	"slices"
	"testing"
)

func tag(name string, log *[]string) Middleware {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			*log = append(*log, name+" before")
			next.ServeHTTP(w, r)
			*log = append(*log, name+" after")
		})
	}
}

func TestChainOrder(t *testing.T) {
	var log []string
	h := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) { log = append(log, "handler") })
	Chain(h, tag("A", &log), tag("B", &log)).ServeHTTP(httptest.NewRecorder(), httptest.NewRequest("GET", "/", nil))
	want := []string{"A before", "B before", "handler", "B after", "A after"}
	if !slices.Equal(log, want) {
		t.Fatalf("执行顺序 = %v\n期望 %v", log, want)
	}
}

func TestChainEmpty(t *testing.T) {
	called := false
	h := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) { called = true })
	Chain(h).ServeHTTP(httptest.NewRecorder(), httptest.NewRequest("GET", "/", nil))
	if !called {
		t.Fatal("没有中间件时应直接调用 handler")
	}
}

=== web-02-2
title: 实现 Gin 风格的 c.Next() 和 c.Abort()
--- desc
Gin 把中间件和最终处理函数放在一个切片里，`Context` 记录当前执行到的下标。实现：

- `Run(handlers...)`：创建 Context，从第一个处理函数开始执行；
- `c.Next()`：执行后续的处理函数，全部返回后才回到调用 `Next` 的地方，形成洋葱模型；
- `c.Abort()`：后续处理函数不再执行（但当前函数会继续执行完）；
- 中间件**不调用** `Next` 时，框架也要在它返回后自动执行下一个处理函数。

提示：Gin 的 `Next` 是一个循环：`c.index++; for c.index < len(c.handlers) { c.handlers[c.index](c); c.index++ }`，`Abort` 把 index 设成一个很大的值。
--- solution
package main

import "math"

type HandlerFunc func(*Context)

type Context struct {
	handlers []HandlerFunc
	index    int
	Log      []string
}

const abortIndex = math.MaxInt / 2

func Run(handlers ...HandlerFunc) *Context {
	c := &Context{handlers: handlers, index: -1}
	c.Next()
	return c
}

func (c *Context) Next() {
	// @solution
	c.index++
	for c.index < len(c.handlers) {
		c.handlers[c.index](c)
		c.index++
	}
	// @starter
	// // TODO
	// @end
}

func (c *Context) Abort() {
	// @solution
	c.index = abortIndex
	// @starter
	// // TODO
	// @end
}

func (c *Context) IsAborted() bool { return c.index >= abortIndex }

		t.Fatalf("Log = %v\n期望 %v", c.Log, want)
	}
}

--- desc
用标准库的 `http.NewServeMux()` 注册路由（Go 1.22 起支持请求方法和路径参数）：

| 路由 | 响应 |
| --- | --- |
| `GET /users/{id}` | `200`，响应体 `user <id>` |

package main

	"net/http"

		w.WriteHeader(http.StatusCreated)
		fmt.Fprint(w, "created")
	})
	mux.HandleFunc("GET /files/{path...}", func(w http.ResponseWriter, r *http.Request) {
		fmt.Fprint(w, r.PathValue("path"))
	})
	// @starter

}

=== web-04-1
title: 优雅关闭 HTTP 服务
--- desc
实现 `Serve(ctx, srv, ln, grace)`：

1. 在后台用 `srv.Serve(ln)` 启动服务；
2. `ctx` 被取消（收到退出信号）时，调用 `srv.Shutdown`：停止接收新连接，**等待处理中的请求完成**，最多等 `grace`；
3. 正常关闭时返回 nil（`Serve` 返回的 `http.ErrServerClosed` 不算错误）；服务自己出错时返回该错误。

测试会启动一个真实的本地服务：请求处理到一半时取消 ctx，这个请求仍然必须成功完成。
--- solution
package main

import (
	"context"
	"errors"
	"net"
	"net/http"
	"time"
)

func Serve(ctx context.Context, srv *http.Server, ln net.Listener, grace time.Duration) error {
	// @solution
	errCh := make(chan error, 1)
	go func() { errCh <- srv.Serve(ln) }()
	select {
	case err := <-errCh:
		return err
	case <-ctx.Done():
	}
	shutdownCtx, cancel := context.WithTimeout(context.Background(), grace)
	defer cancel()
	if err := srv.Shutdown(shutdownCtx); err != nil {
		return err
	}
	if err := <-errCh; !errors.Is(err, http.ErrServerClosed) {
		return err
	}
	return nil
	// @starter
	// return srv.Serve(ln) // TODO: 没有处理退出信号
	// @end
}
--- test
package main

import (
	"context"
	"io"
	"net"
	"net/http"
	"testing"
	"time"
)

func TestGracefulShutdown(t *testing.T) {
	ln, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	srv := &http.Server{Handler: http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		time.Sleep(200 * time.Millisecond)
		io.WriteString(w, "done")
	})}
	ctx, cancel := context.WithCancel(context.Background())
	served := make(chan error, 1)
	go func() { served <- Serve(ctx, srv, ln, 2*time.Second) }()

	url := "http://" + ln.Addr().String()
	type result struct {
		body string
		err  error
	}
	resCh := make(chan result, 1)
	go func() {
		resp, err := http.Get(url)
		if err != nil {
			resCh <- result{err: err}
			return
		}
		defer resp.Body.Close()
		b, err := io.ReadAll(resp.Body)
		resCh <- result{string(b), err}
	}()

	time.Sleep(50 * time.Millisecond) // 请求正在处理中
	cancel()

	select {
	case r := <-resCh:
		if r.err != nil || r.body != "done" {
			t.Fatalf("处理中的请求应正常完成：body=%q err=%v", r.body, r.err)
		}
	case <-time.After(3 * time.Second):
		t.Fatal("请求超时")
	}
	select {
	case err := <-served:
		if err != nil {
			t.Fatalf("优雅关闭应返回 nil，实际 %v", err)
		}
	case <-time.After(3 * time.Second):
		t.Fatal("ctx 取消后 Serve 应该返回")
	}
	if _, err := http.Get(url); err == nil {
		t.Fatal("关闭后不应再接受新连接")
	}
}

title: 就绪探针：关闭前先摘流量
--- desc
在 Kubernetes 里，Pod 收到 SIGTERM 后，Service 摘除它的 Endpoint 有延迟。常见做法：先让就绪探针失败，等几秒让流量切走，再真正关闭。

实现 `Health`：

- `GET /livez` 永远返回 `200 ok`（存活探针，失败会被重启）；
- `GET /readyz` 正常时返回 `200 ok`，调用 `MarkShuttingDown()` 之后返回 `503 shutting down`；
- `MarkShuttingDown` 可能和请求并发执行，用 `atomic.Bool`。

	// @solution
	h.shuttingDown.Store(true)
	// @starter
	// // TODO
	// @end

	mux.HandleFunc("GET /readyz", func(w http.ResponseWriter, r *http.Request) {
		// @solution

=== web-05-1
title: 正确使用 http.Client
--- desc
实现 `FetchJSON(ctx, client, url, v)`，把 JSON 响应解码到 `v`：

1. 用 `http.NewRequestWithContext` 创建请求，让 ctx 的超时和取消生效；
2. **一定要关闭** `resp.Body`，否则连接无法复用、goroutine 泄漏；
3. 状态码不是 200 时返回错误（net/http 不会因为 4xx/5xx 返回 error）；
4. 用 `io.LimitReader` 限制最多读取 1MB，防止对方返回超大响应。

测试会统计响应体的 `Close` 调用次数。
--- solution
package main

import (
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
)

func FetchJSON(ctx context.Context, client *http.Client, url string, v any) error {
	// @solution
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
	if err != nil {
		return err
	}
	resp, err := client.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		io.Copy(io.Discard, io.LimitReader(resp.Body, 4<<10))
		return fmt.Errorf("unexpected status %d", resp.StatusCode)
	}
	return json.NewDecoder(io.LimitReader(resp.Body, 1<<20)).Decode(v)
	// @starter
	// resp, err := client.Get(url) // TODO: 没有用 ctx，没有关闭 Body，没有检查状态码
	// if err != nil {
	// 	return err
	// }
	// return json.NewDecoder(resp.Body).Decode(v)
	// @end
}

package main

import (
	"context"
	"io"
	"net/http"
	"net/http/httptest"
	"sync/atomic"
	"testing"
	"time"
)

type countingBody struct {
	io.ReadCloser
	closes *atomic.Int32
}

func (b countingBody) Close() error {
	b.closes.Add(1)
	return b.ReadCloser.Close()
}

type countingTransport struct{ closes atomic.Int32 }

func (t *countingTransport) RoundTrip(r *http.Request) (*http.Response, error) {
	resp, err := http.DefaultTransport.RoundTrip(r)
	if err == nil {
		resp.Body = countingBody{resp.Body, &t.closes}
	}
	return resp, err
}

func server() *httptest.Server {
	mux := http.NewServeMux()
	mux.HandleFunc("/ok", func(w http.ResponseWriter, r *http.Request) {
		io.WriteString(w, `{"name":"gopher","age":13}`)
	})
	mux.HandleFunc("/fail", func(w http.ResponseWriter, r *http.Request) {
		http.Error(w, `{"error":"boom"}`, 500)
	})
	mux.HandleFunc("/slow", func(w http.ResponseWriter, r *http.Request) {
		select {
		case <-time.After(2 * time.Second):
		case <-r.Context().Done():
		}
	})
	return httptest.NewServer(mux)
}

type user struct {
	Name string `json:"name"`
	Age  int    `json:"age"`
}

func TestFetchJSON(t *testing.T) {
	srv := server()
	defer srv.Close()
	tr := &countingTransport{}
	client := &http.Client{Transport: tr}

	var u user
	if err := FetchJSON(context.Background(), client, srv.URL+"/ok", &u); err != nil || u.Name != "gopher" {
		t.Fatalf("u=%+v err=%v", u, err)
	}
	if err := FetchJSON(context.Background(), client, srv.URL+"/fail", &u); err == nil {
		t.Fatal("500 应返回错误")

=== web-06-1
title: 手写 JWT（HS256）签发与校验
--- desc
JWT 由三段组成：`base64url(header).base64url(payload).base64url(签名)`，签名是对前两段用 HMAC-SHA256 计算的。实现：

- `Sign(claims, secret)`：header 固定为 `{"alg":"HS256","typ":"JWT"}`，三段都用 `base64.RawURLEncoding`；
- `Verify(token, secret, now)`：
  - 不是三段、base64 或 JSON 解析失败 → `ErrMalformed`；
  - header 的 `alg` 不是 `HS256`（比如 `none`）→ `ErrMalformed`；
  - 签名不匹配 → `ErrSignature`（用 `hmac.Equal` 做常量时间比较）；
  - `exp`（Unix 秒）不晚于 `now` → `ErrExpired`。

注意 payload 只是编码，**不是加密**，任何人都能解码看到内容。
--- solution
package main

import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/base64"
	"encoding/json"
	"errors"
	"strings"
	"time"
)

var (
	ErrMalformed = errors.New("malformed token")
	ErrSignature = errors.New("invalid signature")
	ErrExpired   = errors.New("token expired")
)

type Claims struct {
	Sub string `json:"sub"`
	Exp int64  `json:"exp"`
}

type header struct {
	Alg string `json:"alg"`
	Typ string `json:"typ"`
}

var enc = base64.RawURLEncoding

func sign(data string, secret []byte) []byte {
	mac := hmac.New(sha256.New, secret)
	mac.Write([]byte(data))
	return mac.Sum(nil)
}

func Sign(c Claims, secret []byte) (string, error) {
	// @solution
	h, _ := json.Marshal(header{"HS256", "JWT"})
	p, err := json.Marshal(c)
	if err != nil {
		return "", err
	}
	unsigned := enc.EncodeToString(h) + "." + enc.EncodeToString(p)
	return unsigned + "." + enc.EncodeToString(sign(unsigned, secret)), nil
	// @starter
	// return "", errors.New("TODO")
	// @end
}

func Verify(token string, secret []byte, now time.Time) (Claims, error) {
	var c Claims
	// @solution
	parts := strings.Split(token, ".")
	if len(parts) != 3 {
		return c, ErrMalformed
	}

title: CORS 中间件
--- desc
实现 `CORS(allowed, next)`：

- 请求没有 `Origin` 头：直接交给 `next`；
- `Origin` 在白名单中：
  - 设置 `Access-Control-Allow-Origin: <origin>`（带 Cookie 时不能用 `*`）、`Access-Control-Allow-Credentials: true`、`Vary: Origin`；
  - 如果是**预检请求**（`OPTIONS` 且带 `Access-Control-Request-Method`）：再设置 `Access-Control-Allow-Methods: GET, POST, PUT, DELETE`、`Access-Control-Allow-Headers`（回显请求里的 `Access-Control-Request-Headers`）、`Access-Control-Max-Age: 600`，返回 `204`，**不调用** `next`；
  - 否则调用 `next`；
- `Origin` 不在白名单：不加任何 CORS 头；预检请求返回 `403`，普通请求照常交给 `next`（浏览器会拦截响应）。
--- solution
package main

import (
	"net/http"
	"slices"
)

func CORS(allowed []string, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		// @solution
		origin := r.Header.Get("Origin")
		if origin == "" {
			next.ServeHTTP(w, r)
			return
		}
		preflight := r.Method == http.MethodOptions && r.Header.Get("Access-Control-Request-Method") != ""
		if !slices.Contains(allowed, origin) {
			if preflight {
				w.WriteHeader(http.StatusForbidden)
				return
			}
			next.ServeHTTP(w, r)
			return
		}
		h := w.Header()
		h.Set("Access-Control-Allow-Origin", origin)
		h.Set("Access-Control-Allow-Credentials", "true")
		h.Add("Vary", "Origin")
		if preflight {
			h.Set("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE")
			if reqHeaders := r.Header.Get("Access-Control-Request-Headers"); reqHeaders != "" {
				h.Set("Access-Control-Allow-Headers", reqHeaders)
			}
			h.Set("Access-Control-Max-Age", "600")
			w.WriteHeader(http.StatusNoContent)
			return
		}
		next.ServeHTTP(w, r)
		// @starter
		// w.Header().Set("Access-Control-Allow-Origin", "*") // TODO: 带 Cookie 时不允许 *，也没有处理预检
		// next.ServeHTTP(w, r)
		// @end
	})
}

title: 幂等键：重复请求返回第一次的结果
race: true
--- desc
客户端为每次操作生成一个唯一的幂等键，服务端保证同一个键只执行一次。实现 `Store.Do(key, fn)`：

- key 第一次出现：执行 `fn`，保存结果，返回 `(result, false, err)`；
- **并发**的重复请求：等待第一次执行完，返回同样的结果和 `replayed = true`；
- 之后的重复请求：直接返回保存的结果，`replayed = true`；
- `fn` 返回错误：**不保存**（删除记录），允许客户端用同一个键重试。
--- solution
package main

import "sync"

type Result struct {
	Status int
	Body   string
}

type entry struct {
	done chan struct{}
	res  Result
	err  error
}

type Store struct {
	mu      sync.Mutex
	entries map[string]*entry
}

func NewStore() *Store { return &Store{entries: map[string]*entry{}} }

func (s *Store) Do(key string, fn func() (Result, error)) (Result, bool, error) {
	// @solution
	s.mu.Lock()
	if e, ok := s.entries[key]; ok {
		s.mu.Unlock()
		<-e.done
		return e.res, true, e.err
	}
	e := &entry{done: make(chan struct{})}
	s.entries[key] = e
	s.mu.Unlock()

	e.res, e.err = fn()
	if e.err != nil {
		s.mu.Lock()
		delete(s.entries, key)
		s.mu.Unlock()
	}
	close(e.done)
	return e.res, false, e.err
	// @starter
	// res, err := fn() // TODO: 每次都执行
	// return res, false, err
	// @end
}

--- desc
漏桶以恒定速率「漏水」：每个请求往桶里加一份水，桶满了就拒绝。它能把突发流量整形为平稳的速率，但不允许合理的突发（这是和令牌桶的主要区别）。

1. 先按距离上次计算经过的时间，漏掉 `rate × 秒数` 的水（水位不低于 0）；

package main

import "time"

type LeakyBucket struct {
	capacity float64
	rate     float64 // 每秒漏掉的水量
	water    float64
	last     time.Time
	now      func() time.Time
}

func NewLeakyBucket(capacity, rate float64, now func() time.Time) *LeakyBucket {
	return &LeakyBucket{capacity: capacity, rate: rate, last: now(), now: now}
}

func (b *LeakyBucket) Allow() bool {
	// @solution
	now := b.now()
	elapsed := now.Sub(b.last).Seconds()
	b.last = now
	b.water = max(0, b.water-elapsed*b.rate)
	if b.water+1 > b.capacity {
		return false
	}
	b.water++
	return true

	// @end

package main

import (
	"sync"
	"time"
)

// window 是一个已经实现好的固定窗口计数器：每个窗口最多 limit 个请求。
type window struct {
	start time.Time
	count int
}

func (w *window) allow(now time.Time, limit int, size time.Duration) bool {
	if now.Sub(w.start) >= size {
		w.start, w.count = now, 0
	}
	if w.count >= limit {
		return false
	}
	w.count++
	return true
}

type keyState struct {
	w        window
	lastSeen time.Time
}

type KeyedLimiter struct {
	mu    sync.Mutex
	now   func() time.Time
	limit int
	size  time.Duration
	idle  time.Duration
	keys  map[string]*keyState
}

func NewKeyedLimiter(now func() time.Time, limit int, size, idle time.Duration) *KeyedLimiter {
	return &KeyedLimiter{now: now, limit: limit, size: size, idle: idle, keys: map[string]*keyState{}}
}

	// @solution
	l.mu.Lock()
	defer l.mu.Unlock()
	now := l.now()
	st, ok := l.keys[key]
	if !ok {
		st = &keyState{}
		l.keys[key] = st
	}
	st.lastSeen = now
	return st.w.allow(now, l.limit, l.size)
	// @starter
	// return true // TODO
	// @end

	l.mu.Lock()
	defer l.mu.Unlock()
	now := l.now()

			n++

测试里的 handler 已经写好：它调用一个会响应 ctx 的慢查询，遇到 `context.DeadlineExceeded` 返回 `504`。只要超时传进去了，慢请求就会及时返回 504，而不是一直挂着。
--- solution
package main

import (
	"context"
	"net/http"
	"time"
)

func Timeout(d time.Duration) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			// @solution
			ctx, cancel := context.WithTimeout(r.Context(), d)
			defer cancel()
			next.ServeHTTP(w, r.WithContext(ctx))
			// @starter
			// next.ServeHTTP(w, r) // TODO
			// @end
		})
	}
}

package main

import (
	"context"
	"errors"
	"net/http"
	"net/http/httptest"
	"testing"
	"time"
)

func slowQuery(ctx context.Context, cost time.Duration) error {
	select {
	case <-time.After(cost):
		return nil
	case <-ctx.Done():
		return ctx.Err()
	}
}

func handler(cost time.Duration) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if err := slowQuery(r.Context(), cost); errors.Is(err, context.DeadlineExceeded) {
			http.Error(w, "timeout", http.StatusGatewayTimeout)
			return
		}
		w.Write([]byte("ok"))
	})
}

func TestTimeout(t *testing.T) {
	mw := Timeout(30 * time.Millisecond)

	rec := httptest.NewRecorder()
	mw(handler(time.Millisecond)).ServeHTTP(rec, httptest.NewRequest("GET", "/", nil))
	if rec.Code != 200 {
		t.Fatalf("快请求应返回 200，实际 %d", rec.Code)
	}

	start := time.Now()
	rec = httptest.NewRecorder()
	mw(handler(2*time.Second)).ServeHTTP(rec, httptest.NewRequest("GET", "/", nil))
	if rec.Code != http.StatusGatewayTimeout || time.Since(start) > 500*time.Millisecond {
		t.Fatalf("慢请求应在约 30ms 后返回 504，实际 %d（耗时 %v）", rec.Code, time.Since(start))
	}
}

title: 统一错误响应
--- desc
实现统一的错误出口 `WriteError(w, err)`，响应体都是 `{"code":…,"message":"…"}`：

| 错误 | HTTP 状态码 | code | message |
| --- | --- | --- | --- |
| 错误链上有 `*AppError` | `e.Status` | `e.Code` | `e.Message` |
| 错误链上有 `context.DeadlineExceeded` | 504 | 50400 | `upstream timeout` |
| 其他 | 500 | 50000 | `internal error` |

最后一种情况**不能**把内部错误信息（比如 SQL 报错）暴露给用户。`AppError` 通过 `Unwrap` 保留内部原因，方便记录日志。
--- solution
package main

import (
	"context"
	"encoding/json"
	"errors"
	"net/http"
)

type AppError struct {
	Status  int
	Code    int
	Message string
	cause   error
}

func (e *AppError) Error() string { return e.Message }
func (e *AppError) Unwrap() error { return e.cause }

func NewAppError(status, code int, msg string, cause error) *AppError {
	return &AppError{Status: status, Code: code, Message: msg, cause: cause}
}

type body struct {
	Code    int    `json:"code"`
	Message string `json:"message"`
}

func WriteError(w http.ResponseWriter, err error) {
	status, b := http.StatusInternalServerError, body{50000, "internal error"}
	// @solution
	var ae *AppError
	switch {
	case errors.As(err, &ae):
		status, b = ae.Status, body{ae.Code, ae.Message}
	case errors.Is(err, context.DeadlineExceeded):
		status, b = http.StatusGatewayTimeout, body{50400, "upstream timeout"}
	}
	// @starter
	// b.Message = err.Error() // TODO: 泄漏了内部错误信息，也没有区分错误类型
	// @end
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	json.NewEncoder(w).Encode(b)
}

=== web-11-2
title: 参数校验与 typed nil 陷阱
--- desc
实现 `Validate(req)`，按 `name`、`email`、`age` 的顺序检查，收集**所有**字段错误：

- `name`：去掉首尾空白后不能为空（`required`），最多 20 个字符，按 rune 计（`too long`）；
- `email`：必须恰好包含一个 `@`，且前后都不为空（`invalid`）；
- `age`：0 ~ 150（`out of range`）。

有错误时返回 `ValidationErrors`，它的 `Error()` 形如 `name: required; age: out of range`。

**没有错误时必须返回真正的 nil**——如果直接 `return errs`（一个值为 nil 的 `ValidationErrors`），调用方拿到的 `error` 并不等于 nil（参见 GO-10）。
--- solution
package main

import (
	"strings"
	"unicode/utf8"
)

type FieldError struct {
	Field string
	Msg   string
}

type ValidationErrors []FieldError

func (v ValidationErrors) Error() string {
	parts := make([]string, len(v))
	for i, e := range v {
		parts[i] = e.Field + ": " + e.Msg
	}
	return strings.Join(parts, "; ")
}

type CreateUserReq struct {
	Name  string
	Email string
	Age   int
}

func Validate(r CreateUserReq) error {
	var errs ValidationErrors
	// @solution
	name := strings.TrimSpace(r.Name)
	switch {
	case name == "":
		errs = append(errs, FieldError{"name", "required"})
	case utf8.RuneCountInString(name) > 20:
		errs = append(errs, FieldError{"name", "too long"})
	}
	local, domain, ok := strings.Cut(r.Email, "@")
	if !ok || local == "" || domain == "" || strings.Contains(domain, "@") {
		errs = append(errs, FieldError{"email", "invalid"})
	}
	if r.Age < 0 || r.Age > 150 {
		errs = append(errs, FieldError{"age", "out of range"})
	}
	if len(errs) == 0 {
		return nil
	}
	// @starter
	// // TODO: 校验各个字段
	// @end
	return errs
}
--- test
package main

import (
	"errors"
	"strings"
	"testing"
)

func TestValid(t *testing.T) {
	if err := Validate(CreateUserReq{Name: "张三", Email: "a@b.com", Age: 30}); err != nil {
		t.Fatalf("合法请求应返回 nil，实际 %#v", err)
	}
	if err := Validate(CreateUserReq{Name: strings.Repeat("名", 20), Email: "x@y", Age: 0}); err != nil {
		t.Fatalf("20 个汉字是合法的（按 rune 计），实际 %v", err)
	}
}

func TestInvalid(t *testing.T) {
	err := Validate(CreateUserReq{Name: "   ", Email: "no-at-sign", Age: 200})
	var ve ValidationErrors
	if !errors.As(err, &ve) || len(ve) != 3 {
		t.Fatalf("应返回 3 个字段错误，实际 %v", err)
	}
	if err.Error() != "name: required; email: invalid; age: out of range" {
		t.Fatalf("Error() = %q", err.Error())
	}
	cases := []struct {
		req  CreateUserReq
		want string
	}{
		{CreateUserReq{Name: strings.Repeat("a", 21), Email: "a@b", Age: 1}, "name: too long"},
		{CreateUserReq{Name: "ok", Email: "@b.com", Age: 1}, "email: invalid"},
		{CreateUserReq{Name: "ok", Email: "a@b@c", Age: 1}, "email: invalid"},
		{CreateUserReq{Name: "ok", Email: "a@b", Age: -1}, "age: out of range"},
	}
	for _, c := range cases {
		if err := Validate(c.req); err == nil || err.Error() != c.want {
			t.Errorf("Validate(%+v) = %v，期望 %q", c.req, err, c.want)
		}
	}
}
