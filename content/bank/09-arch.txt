=== arch-01-1
title: 服务依赖的发布顺序
--- desc
微服务拆分后，服务之间有依赖关系，循环依赖意味着拆分不合理。实现 `DeployOrder(deps)`：

- `deps[a]` 是服务 a 依赖的服务列表，被依赖的服务要先发布；
- 返回一个发布顺序（拓扑排序）。有多个服务可以同时发布时，按名字字典序选择，保证结果确定；
- 存在循环依赖时返回错误。

提示：Kahn 算法——统计每个服务还有多少依赖没发布，依赖数为 0 的可以发布。
--- solution
package main

import (
	"errors"
	"slices"
)

func DeployOrder(deps map[string][]string) ([]string, error) {
	// @solution
	pending := map[string]int{}       // 还没发布的依赖数
	dependents := map[string][]string{} // 被谁依赖
	for svc, ds := range deps {
		if _, ok := pending[svc]; !ok {
			pending[svc] = 0
		}
		for _, d := range ds {
			pending[svc]++
			dependents[d] = append(dependents[d], svc)
			if _, ok := pending[d]; !ok {
				pending[d] = 0
			}
		}
	}
	var ready, order []string
	for svc, n := range pending {
		if n == 0 {
			ready = append(ready, svc)
		}
	}
	for len(ready) > 0 {
		slices.Sort(ready)
		svc := ready[0]
		ready = ready[1:]
		order = append(order, svc)
		for _, d := range dependents[svc] {
			pending[d]--
			if pending[d] == 0 {
				ready = append(ready, d)
			}
		}
	}
	if len(order) != len(pending) {
		return nil, errors.New("cyclic dependency")
	}
	return order, nil
	// @starter
	// return nil, errors.New("TODO")
	// @end
}
--- test

import (
	"slices"
	"testing"
)

	deps := map[string][]string{
		"gateway": {"order", "user"},
		"order":   {"user", "stock"},
		"stock":   {},
		"user":    {"config"},
	}

}

=== arch-02-1
title: Quorum 读写：W + R > N
--- desc
N 个副本，写入至少 W 个成功才算成功，读取至少读 R 个副本、取版本号最大的值。当 `W + R > N` 时，读写集合必然有交集，能读到最新写入。

- `Overlaps(n, w, r)`：是否满足 `W + R > N`；
- `Write(rs, w, version, value)`：写入所有**存活**的副本；存活数少于 w 时返回 `ErrNoQuorum`（简化：已写入的不回滚）；
- `Read(rs, r)`：按顺序从**存活**的副本中读取前 r 个，返回其中版本最大的值；存活数不足 r 返回 `ErrNoQuorum`。
--- solution
package main

import "errors"

var ErrNoQuorum = errors.New("not enough replicas")

type Replica struct {
	Alive   bool
	Version int
	Value   string
}

func Overlaps(n, w, r int) bool {
	// @solution
	return w+r > n
	// @starter
	// return false // TODO
	// @end
}

func Write(rs []*Replica, w int, version int, value string) error {
	// @solution
	ok := 0
	for _, r := range rs {
		if r.Alive {
			r.Version, r.Value = version, value
			ok++
		}
	}
	if ok < w {
		return ErrNoQuorum
	}
	return nil
	// @starter
	// return ErrNoQuorum // TODO
	// @end
}

func Read(rs []*Replica, r int) (string, error) {
	// @solution
	var best *Replica
	read := 0
	for _, rep := range rs {
		if read == r {
			break
		}
		if !rep.Alive {
			continue
		}
		read++
		if best == nil || rep.Version > best.Version {
			best = rep
		}
	}
	if read < r {
		return "", ErrNoQuorum
	}
	return best.Value, nil
	// @starter
	// return "", ErrNoQuorum // TODO
	// @end
}
--- test
package main

import (
	"errors"
	"testing"
)

func TestQuorum(t *testing.T) {
	if !Overlaps(3, 2, 2) || Overlaps(3, 1, 2) || !Overlaps(5, 3, 3) {
		t.Fatal("Overlaps 判断不对")
	}
	rs := []*Replica{{Alive: true}, {Alive: true}, {Alive: true}}
	if err := Write(rs, 2, 1, "v1"); err != nil {
		t.Fatal(err)
	}
	rs[0].Alive = false
	if err := Write(rs, 2, 2, "v2"); err != nil {
		t.Fatalf("2 个副本存活时 W=2 应成功：%v", err)
	}
	rs[0].Alive = true // 副本 0 恢复，但数据还是 v1
	rs[2].Alive = false
	if v, err := Read(rs, 2); err != nil || v != "v2" {
		t.Fatalf("R=2 应读到最新的 v2，实际 %q, %v", v, err)
	}
	if v, _ := Read(rs, 1); v != "v1" {
		t.Fatalf("W+R<=N 时可能读到旧数据：R=1 读到 %q，期望 v1", v)
	}
	rs[1].Alive = false
	if _, err := Read(rs, 2); !errors.Is(err, ErrNoQuorum) {
		t.Fatal("存活副本不足时应返回 ErrNoQuorum")
	}
	if err := Write(rs, 2, 3, "v3"); !errors.Is(err, ErrNoQuorum) {
		t.Fatal("存活副本不足时写入应失败")
	}
}

title: 基于心跳 TTL 的服务注册中心
--- desc
实现一个简化的注册中心：服务实例定期发送心跳，超过 TTL 没有心跳的实例自动下线。

- `Register(service, addr)`：注册或续约（心跳也调用它）；
- `Deregister(service, addr)`：优雅下线时主动注销；
- `Instances(service)`：返回 TTL 内有心跳的实例地址（升序），并顺便清理过期实例。

时间通过注入的 `now` 获取。

	// @solution
	m, ok := r.instances[service]
	if !ok {
		m = map[string]time.Time{}
		r.instances[service] = m
	}
	m[addr] = r.now()

	// @end

}

func (r *Registry) Instances(service string) []string {
	// @solution
	var out []string
	now := r.now()
	for addr, last := range r.instances[service] {
		if now.Sub(last) > r.ttl {

=== arch-04-1
title: 平滑加权轮询（Nginx 算法）
--- desc
普通加权轮询在权重 `a:5, b:1, c:1` 时会产生 `a a a a a b c`，a 会在短时间内集中收到大量请求。Nginx 的**平滑加权轮询**能把它们打散：

每次选择时：

1. 每个节点的 `current += weight`；
2. 选出 `current` 最大的节点（相同时选靠前的）；
3. 被选中节点的 `current -= 所有权重之和`。

权重 `a:5, b:1, c:1` 时，前 7 次的结果是 `a a b a c a a`。
--- solution
package main

type node struct {
	name    string
	weight  int
	current int
}

type SWRR struct {
	nodes []*node
}

func (s *SWRR) Add(name string, weight int) {
	s.nodes = append(s.nodes, &node{name: name, weight: weight})
}

func (s *SWRR) Next() string {
	// @solution
	total := 0
	var best *node
	for _, n := range s.nodes {
		n.current += n.weight
		total += n.weight
		if best == nil || n.current > best.current {
			best = n
		}
	}
	if best == nil {
		return ""
	}
	best.current -= total
	return best.name
	// @starter
	// return "" // TODO
	// @end
}
--- test

	s.Add("c", 1)

	}

=== arch-04-2
title: P2C：随机选两个，挑负载低的
--- desc
P2C（Power of Two Choices）：随机选出两个**不同**的节点，选择在途请求数（`Inflight`）更少的那个，相同时选第一个。实现简单，效果接近「全局最少连接」，又不需要扫描所有节点。

`PickP2C(nodes, randIntn)`：随机数通过 `randIntn(n)`（返回 `[0, n)`）注入。只有一个节点时直接返回它。
--- solution
package main

type Node struct {
	Name     string
	Inflight int
}

func PickP2C(nodes []*Node, randIntn func(int) int) *Node {
	// @solution
	if len(nodes) == 1 {
		return nodes[0]
	}
	i := randIntn(len(nodes))
	j := randIntn(len(nodes) - 1)
	if j >= i {
		j++ // 保证 j != i
	}
	a, b := nodes[i], nodes[j]
	if b.Inflight < a.Inflight {
		return b
	}
	return a
	// @starter
	// return nodes[randIntn(len(nodes))] // TODO: 纯随机
	// @end
}
--- test
package main

import (

func TestDistinctAndLess(t *testing.T) {
	nodes := []*Node{{"a", 5}, {"b", 1}, {"c", 3}}
	seq := []int{0, 0} // i=0, j=0 → 应映射成 1

=== arch-05-1
title: 超时降级：返回兜底数据
--- desc
降级是在下游故障时主动返回兜底数据，保证核心功能可用。实现 `WithFallback(ctx, timeout, primary, fallback)`：

- 调用 `primary`，在 `timeout` 内成功就返回它的结果，`degraded = false`；
- 超时或出错时返回 `fallback()` 的结果，`degraded = true`；
- **即使 `primary` 不响应 ctx（卡住不返回）**，也必须在超时后立即返回——所以要在单独的 goroutine 里调用它，用 `select` 等待；
- 返回前取消传给 `primary` 的 ctx，结果 channel 要有缓冲，避免 goroutine 泄漏。
--- solution
package main

import (
	"context"
	"time"
)

func WithFallback(ctx context.Context, timeout time.Duration,
	primary func(ctx context.Context) (string, error), fallback func() string) (string, bool) {
	// @solution
	ctx, cancel := context.WithTimeout(ctx, timeout)
	defer cancel()
	type result struct {
		v   string
		err error
	}
	ch := make(chan result, 1)
	go func() {
		v, err := primary(ctx)
		ch <- result{v, err}
	}()
	select {
	case r := <-ch:
		if r.err == nil {
			return r.v, false
		}
	case <-ctx.Done():
	}
	return fallback(), true
	// @starter
	// v, err := primary(ctx) // TODO: primary 卡住时会一直等
	// if err != nil {
	// 	return fallback(), true
	// }
	// return v, false
	// @end
}
--- test
package main

import (
	"context"
	"errors"
	"testing"
	"time"
)

func fb() string { return "热门推荐（缓存）" }

func TestPrimaryOK(t *testing.T) {
	v, degraded := WithFallback(context.Background(), time.Second,
		func(ctx context.Context) (string, error) { return "个性化推荐", nil }, fb)
	if v != "个性化推荐" || degraded {
		t.Fatalf("got %q degraded=%v", v, degraded)
	}
}

func TestPrimaryError(t *testing.T) {

		t.Fatalf("出错时应降级：%q %v", v, degraded)
	}
}

	var seen context.Context
	start := time.Now()
	v, degraded := WithFallback(context.Background(), 30*time.Millisecond,

			seen = ctx
			time.Sleep(500 * time.Millisecond) // 完全不理会 ctx
			return "too late", nil
		}, fb)
	if v != fb() || !degraded {
		t.Fatalf("超时应降级：%q %v", v, degraded)
	}
	if d := time.Since(start); d > 200*time.Millisecond {
		t.Fatalf("耗时 %v，应在超时后立即返回", d)
	}
	time.Sleep(10 * time.Millisecond)
	if seen == nil || seen.Err() == nil {
		t.Fatal("返回前应取消传给 primary 的 ctx")
	}
}

=== arch-06-1
title: Saga：失败时逆序补偿
--- desc
Saga 把分布式事务拆成一系列本地事务，每一步都有对应的**补偿操作**。实现编排式 Saga `RunSaga(steps)`：

1. 依次执行每一步的 `Action`；
2. 某一步失败时，**按相反顺序**对已经成功的步骤执行 `Compensate`；
3. 返回的错误用 `fmt.Errorf("step %s: %w", name, err)` 包装原始错误；
4. 补偿本身也可能失败：继续补偿其余步骤，并用 `errors.Join` 把补偿错误也合并进返回值（真实系统中这些需要重试或人工介入）。
--- solution
package main

import (
	"errors"
	"fmt"
)

type Step struct {
	Name       string
	Action     func() error
	Compensate func() error
}

func RunSaga(steps []Step) error {
	// @solution
	for i, s := range steps {
		if err := s.Action(); err != nil {
			errs := []error{fmt.Errorf("step %s: %w", s.Name, err)}
			for j := i - 1; j >= 0; j-- {
				if cerr := steps[j].Compensate(); cerr != nil {
					errs = append(errs, fmt.Errorf("compensate %s: %w", steps[j].Name, cerr))
				}
			}
			return errors.Join(errs...)
		}
	}
	return nil
	// @starter
	// for _, s := range steps {
	// 	if err := s.Action(); err != nil {
	// 		return fmt.Errorf("step %s: %w", s.Name, err) // TODO: 没有补偿
	// 	}
	// }
	// return nil
	// @end
}

=== arch-06-2
title: 本地消息表（Outbox）投递
--- desc
本地消息表：业务数据和待发送的消息在**同一个本地事务**里写入，再由后台任务把消息投递到 MQ。投递可能失败，所以要能重试，下游需要幂等。

实现 `Outbox.Relay(publish)`：按 ID 顺序投递所有未发送的消息：

- `publish` 成功才标记 `Sent = true`；
- 失败的消息 `Attempts++`，保留到下次再试；
- 返回本次成功投递的数量。
--- solution
package main

type Message struct {
	ID       int
	Payload  string
	Sent     bool
	Attempts int
}

type Outbox struct {
	Msgs []*Message // 按 ID 升序
}

func (o *Outbox) Relay(publish func(Message) error) int {
	// @solution
	n := 0
	for _, m := range o.Msgs {
		if m.Sent {
			continue
		}
		if err := publish(*m); err != nil {
			m.Attempts++
			continue
		}
		m.Sent = true
		n++
	}
	return n
	// @starter
	// for _, m := range o.Msgs {
	// 	m.Sent = true // TODO: 先标记再发送，发送失败消息就丢了
	// 	publish(*m)
	// }
	// return len(o.Msgs)
	// @end
}

=== arch-07-1
title: 实现雪花算法
race: true
--- desc
雪花 ID（64 位）：1 位符号 + 41 位毫秒时间戳（相对自定义纪元）+ 10 位机器 ID + 12 位序列号。实现 `Snowflake.Next()`（并发安全）：

1. 取当前毫秒 `ts`；
2. `ts` 小于上次的时间戳（时钟回拨）：回拨不超过 5ms 就**等待**（反复调用 `now()` 直到追上），超过 5ms 返回 `ErrClockBackwards`；
3. 和上次在同一毫秒：序列号加一；序列号用完（超过 4095）时等待到下一毫秒；
4. 进入新的毫秒：序列号归零；
5. `id = (ts - epoch) << 22 | worker << 12 | seq`。

`Decompose(id)` 反向解析出毫秒时间戳、机器 ID 和序列号。时间通过注入的 `now`（毫秒）获取。
--- solution
package main

import (
	"errors"
	"sync"
)

const (
	epoch       int64 = 1704067200000 // 2024-01-01 00:00:00 UTC
	workerBits        = 10
	seqBits           = 12
	maxSeq            = 1<<seqBits - 1
	maxBackward       = 5
)

var ErrClockBackwards = errors.New("clock moved backwards")

type Snowflake struct {
	mu     sync.Mutex
	now    func() int64
	worker int64
	lastMs int64
	seq    int64
}

func New(worker int64, now func() int64) (*Snowflake, error) {
	if worker < 0 || worker >= 1<<workerBits {
		return nil, errors.New("worker id out of range")
	}
	return &Snowflake{now: now, worker: worker}, nil
}

func (s *Snowflake) Next() (int64, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	// @solution
	ts := s.now()
	if ts < s.lastMs {
		if s.lastMs-ts > maxBackward {
			return 0, ErrClockBackwards
		}
		for ts < s.lastMs {
			ts = s.now()
		}
	}
	if ts == s.lastMs {
		s.seq++
		if s.seq > maxSeq {
			for ts <= s.lastMs {
				ts = s.now()
			}
			s.seq = 0
		}
	} else {
		s.seq = 0
	}
	s.lastMs = ts
	return (ts-epoch)<<(workerBits+seqBits) | s.worker<<seqBits | s.seq, nil
	// @starter
	// return 0, errors.New("TODO")
	// @end
}

func Decompose(id int64) (ms, worker, seq int64) {
	// @solution
	return id>>(workerBits+seqBits) + epoch, id >> seqBits & (1<<workerBits - 1), id & maxSeq
	// @starter
	// return 0, 0, 0 // TODO
	// @end
}

	Reads int

	defer s.mu.Unlock()
	s.Reads++
	v, ok := s.data[k]
	return v, ok

title: 直方图与分位数估算
--- desc
Prometheus 的 Histogram 只记录每个桶的计数，分位数是查询时用 `histogram_quantile` 估算出来的。实现：

- `Observe(v)`：v 落入第一个满足 `v <= Bounds[i]` 的桶，都不满足则落入最后的 `+Inf` 桶；
- `Quantile(q)`：和 `histogram_quantile` 一样：
  1. `rank = q × 总数`；
  2. 找到累计计数第一次 `>= rank` 的桶；
  3. 在这个桶的上下界之间**线性插值**：`下界 + (上界 − 下界) × (rank − 前面桶的累计数) / 本桶计数`。第一个桶的下界取 0；
  4. 落在 `+Inf` 桶时返回最大的有限上界；没有数据返回 `NaN`。

注意估算值依赖桶的划分——桶太粗，p99 可能偏差很大。
--- solution
package main

import "math"

type Histogram struct {
	Bounds []float64
	Counts []uint64 // len(Bounds)+1，最后一个是 +Inf 桶
	Total  uint64
}

func NewHistogram(bounds []float64) *Histogram {
	return &Histogram{Bounds: bounds, Counts: make([]uint64, len(bounds)+1)}
}

func (h *Histogram) Observe(v float64) {
	// @solution
	i := 0
	for i < len(h.Bounds) && v > h.Bounds[i] {
		i++
	}
	h.Counts[i]++
	h.Total++
	// @starter
	// // TODO
	// @end
}

func (h *Histogram) Quantile(q float64) float64 {
	// @solution
	if h.Total == 0 {
		return math.NaN()
	}
	rank := q * float64(h.Total)
	var cum uint64
	for i, c := range h.Counts {
		prev := cum
		cum += c
		if float64(cum) < rank || c == 0 {
			continue
		}
		if i == len(h.Bounds) {
			return h.Bounds[len(h.Bounds)-1]
		}
		lower := 0.0
		if i > 0 {
			lower = h.Bounds[i-1]
		}
		return lower + (h.Bounds[i]-lower)*(rank-float64(prev))/float64(c)
	}
	return h.Bounds[len(h.Bounds)-1]
	// @starter
	// return math.NaN() // TODO
	// @end
}

--- desc
实现 Raft 节点处理 `RequestVote` 的规则：

1. 请求的任期小于自己的当前任期 → 拒绝；
2. 请求的任期更大 → 更新自己的任期，清空投票记录（`VotedFor = -1`，变回 Follower）；

第 4 条保证当选的 Leader 一定包含所有已提交的日志。

package main

type Node struct {
	CurrentTerm int
	VotedFor    int   // -1 表示本任期还没投票
	LogTerms    []int // 每条日志的任期，下标从 1 开始对应 LogTerms[0]
}

type VoteReq struct {
	Term         int
	CandidateID  int
	LastLogIndex int
	LastLogTerm  int
}

type VoteResp struct {
	Term    int
	Granted bool
}

func (n *Node) HandleRequestVote(req VoteReq) VoteResp {
	// @solution
	if req.Term < n.CurrentTerm {
		return VoteResp{n.CurrentTerm, false}
	}
	if req.Term > n.CurrentTerm {
		n.CurrentTerm = req.Term
		n.VotedFor = -1
	}
	lastIndex, lastTerm := len(n.LogTerms), 0
	if lastIndex > 0 {
		lastTerm = n.LogTerms[lastIndex-1]
	}
	upToDate := req.LastLogTerm > lastTerm || (req.LastLogTerm == lastTerm && req.LastLogIndex >= lastIndex)
	if (n.VotedFor == -1 || n.VotedFor == req.CandidateID) && upToDate {
		n.VotedFor = req.CandidateID
		return VoteResp{n.CurrentTerm, true}
	}
	return VoteResp{n.CurrentTerm, false}
	// @starter
	// return VoteResp{n.CurrentTerm, false} // TODO
	// @end

func AdvanceCommit(matchIndex []int, logTerms []int, currentTerm, commitIndex int) int {

	for n := majority; n > commitIndex; n-- {
		if logTerms[n-1] == currentTerm {
			return n
		}
	}
	return commitIndex

--- desc
秒杀流量远超下单服务的处理能力。把同步下单改成「请求进队列 + 后台 worker 异步处理」：

- `Submit(o)`：**非阻塞**入队，队列满了立即返回 `ErrBusy`（前端提示稍后重试），不能让请求堆积；

import (
	"errors"
	"sync"
)

var ErrBusy = errors.New("system busy, retry later")

type Order struct{ ID int }

type Queue struct {
	ch chan Order
}

func NewQueue(size int) *Queue { return &Queue{ch: make(chan Order, size)} }

func (q *Queue) Submit(o Order) error {
	// @solution
	select {
	case q.ch <- o:
		return nil
	default:
		return ErrBusy
	}
	// @starter
	// q.ch <- o // TODO: 队列满时会阻塞
	// return nil
	// @end
}

func (q *Queue) Start(n int, handle func(Order)) (stop func()) {
	// @solution
	var wg sync.WaitGroup
	for i := 0; i < n; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for o := range q.ch {
				handle(o)
			}
		}()
	}
	return func() {
		close(q.ch)
		wg.Wait()
	}
	// @starter
	// return func() {} // TODO
	// @end
}

title: Base62 编解码
--- desc
短链接的短码通常是自增 ID 的 Base62 编码。字符表为 `0-9a-zA-Z`（`0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ`），高位在前。

package main

import (
	"errors"
	"math"
	"strings"
)

const alphabet = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ"

func Encode(id uint64) string {
	// @solution
	if id == 0 {
		return "0"
	}
	var buf [11]byte
	i := len(buf)
	for id > 0 {
		i--
		buf[i] = alphabet[id%62]
		id /= 62
	}
	return string(buf[i:])
	// @starter
	// return "" // TODO
	// @end
}

func Decode(s string) (uint64, error) {
	// @solution
	if s == "" {
		return 0, errors.New("empty code")
	}
	var n uint64
	for i := 0; i < len(s); i++ {
		d := strings.IndexByte(alphabet, s[i])
		if d < 0 {
			return 0, errors.New("invalid character")
		}
		if n > (math.MaxUint64-uint64(d))/62 {
			return 0, errors.New("overflow")
		}
		n = n*62 + uint64(d)
	}
	return n, nil
	// @starter
	// return 0, errors.New("TODO")
	// @end
}

import (
	"errors"

	"sync"

const alphabet = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ"

func Encode(id uint64) string {
	if id == 0 {
		return "0"
	}
	var buf [11]byte
	i := len(buf)
	for id > 0 {
		i--
		buf[i] = alphabet[id%62]
		id /= 62
	}

type Shortener struct {
	mu     sync.RWMutex
	nextID uint64
	byCode map[string]string
	byURL  map[string]string
}

func NewShortener() *Shortener {
	return &Shortener{byCode: map[string]string{}, byURL: map[string]string{}}
}

func (s *Shortener) Shorten(rawURL string) (string, error) {
	// @solution
	u, err := url.Parse(rawURL)
	if err != nil || (u.Scheme != "http" && u.Scheme != "https") || u.Host == "" {
		return "", errors.New("invalid url")
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if code, ok := s.byURL[rawURL]; ok {
		return code, nil
	}
	s.nextID++
	code := Encode(s.nextID)
	s.byCode[code] = rawURL
	s.byURL[rawURL] = code
	return code, nil
	// @starter

	// @end
}

	s.mu.RLock()
	defer s.mu.RUnlock()
	u, ok := s.byCode[code]
	return u, ok
}

=== arch-15-1
title: 拉模式：合并多个有序 Feed
--- desc
拉模式下，用户打开首页时，从关注的每个作者那里拉取最新的帖子，再合并成一条时间线。每个作者的帖子已经按时间**从新到旧**排好序。

实现 `MergeFeeds(feeds, limit)`：用 `container/heap` 做 K 路归并，返回最新的 `limit` 条（时间相同按 ID 从大到小）。只需要访问 `limit` 条左右的数据，不用把所有帖子排序。
--- solution
package main

import "container/heap"

type Post struct {
	ID   int
	Time int64
}

func newer(a, b Post) bool {
	if a.Time != b.Time {
		return a.Time > b.Time
	}
	return a.ID > b.ID
}

type cursor struct {
	feed []Post
	pos  int
}

type cursorHeap []*cursor

func (h cursorHeap) Len() int           { return len(h) }
func (h cursorHeap) Less(i, j int) bool { return newer(h[i].feed[h[i].pos], h[j].feed[h[j].pos]) }
func (h cursorHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *cursorHeap) Push(x any)        { *h = append(*h, x.(*cursor)) }
func (h *cursorHeap) Pop() any {
	old := *h
	x := old[len(old)-1]
	*h = old[:len(old)-1]
	return x
}

func MergeFeeds(feeds [][]Post, limit int) []Post {
	// @solution
	h := &cursorHeap{}
	for _, f := range feeds {
		if len(f) > 0 {
			*h = append(*h, &cursor{feed: f})
		}
	}
	heap.Init(h)
	var out []Post
	for h.Len() > 0 && len(out) < limit {
		c := (*h)[0]
		out = append(out, c.feed[c.pos])
		c.pos++
		if c.pos == len(c.feed) {
			heap.Pop(h)
		} else {
			heap.Fix(h, 0)
		}
	}
	return out
	// @starter
	// return nil // TODO
	// @end
}

- `Timeline(user, limit)`：用户收件箱 + 他关注的**大 V** 的发件箱，合并后按时间从新到旧返回前 `limit` 条（时间相同按 ID 从大到小）。

package main

import (
	"cmp"
	"slices"
)

type Post struct {
	ID   int
	Time int64
}

type Feed struct {
	bigV      int
	followers map[string][]string
	following map[string][]string
	inbox     map[string][]Post
	outbox    map[string][]Post
	Pushes    int
}

func NewFeed(bigV int) *Feed {
	return &Feed{bigV: bigV, followers: map[string][]string{}, following: map[string][]string{},
		inbox: map[string][]Post{}, outbox: map[string][]Post{}}
}

func (f *Feed) Follow(user, author string) {
	f.followers[author] = append(f.followers[author], user)
	f.following[user] = append(f.following[user], author)
}

func (f *Feed) isBigV(author string) bool { return len(f.followers[author]) > f.bigV }

func (f *Feed) Publish(author string, p Post) {
	// @solution
	f.outbox[author] = append(f.outbox[author], p)
	if f.isBigV(author) {
		return
	}
	for _, u := range f.followers[author] {
		f.inbox[u] = append(f.inbox[u], p)
		f.Pushes++
	}
	// @starter
	// f.outbox[author] = append(f.outbox[author], p)
	// for _, u := range f.followers[author] { // TODO: 大 V 也全量推送了
	// 	f.inbox[u] = append(f.inbox[u], p)
	// 	f.Pushes++
	// }
	// @end
}

func (f *Feed) Timeline(user string, limit int) []Post {
	// @solution
	posts := slices.Clone(f.inbox[user])
	for _, a := range f.following[user] {
		if f.isBigV(a) {
			posts = append(posts, f.outbox[a]...)
		}
	}
	slices.SortFunc(posts, func(a, b Post) int {
		return cmp.Or(cmp.Compare(b.Time, a.Time), cmp.Compare(b.ID, a.ID))
	})
	return posts[:min(limit, len(posts))]
	// @starter
	// return nil // TODO
	// @end
}

--- desc
延迟任务到期取消订单时，用户可能恰好在同一时刻完成了支付。两边都必须用**带原状态条件**的更新（相当于 `UPDATE ... WHERE status = 'UNPAID'`），保证只有一方成功：

import "sync"

type Store struct {
	mu     sync.Mutex
	status map[string]string
}

func NewStore() *Store { return &Store{status: map[string]string{}} }

func (s *Store) Set(id, st string) { s.mu.Lock(); s.status[id] = st; s.mu.Unlock() }
func (s *Store) Get(id string) string {
	s.mu.Lock()
	defer s.mu.Unlock()
	return s.status[id]
}

func (s *Store) CompareAndSet(id, from, to string) bool {
	// @solution
	s.mu.Lock()
	defer s.mu.Unlock()
	if s.status[id] != from {
		return false
	}
	s.status[id] = to
	return true
	// @starter
	// s.Set(id, to) // TODO: 没有检查原状态
	// return true
	// @end
}

func CancelIfUnpaid(s *Store, id string) string {
	if s.CompareAndSet(id, "UNPAID", "CANCELLED") {
		return "cancelled"
	}
	return "skipped"
}

func Pay(s *Store, id string) bool {
	return s.CompareAndSet(id, "UNPAID", "PAID")
}

--- desc
`0.1 + 0.2 != 0.3`——金额绝对不要用浮点数，统一用 `int64` 表示「分」。实现：

- `ParseYuan(s)`：把 `"12.34"`、`"12"`、`"0.5"`、`"-3.07"` 解析成分（1234、1200、50、-307）。最多两位小数；`"1.234"`、`"abc"`、`""`、`"1e3"`、`".5"`、`"1."` 都返回错误；**不能**用 `strconv.ParseFloat`；
- `FormatCents(c)`：反向格式化，总是两位小数：`5` → `"0.05"`，`-307` → `"-3.07"`；
- `Split(total, n)`：把金额平均分成 n 份，除不尽的分从前往后各多分 1 分，总和必须等于 total（如 100 分 3 份 → 34、33、33）。

package main

import (
	"errors"
	"fmt"
	"strconv"
	"strings"
)

var ErrBadAmount = errors.New("invalid amount")

func allDigits(s string) bool {
	if s == "" {
		return false
	}
	for i := 0; i < len(s); i++ {
		if s[i] < '0' || s[i] > '9' {
			return false
		}
	}
	return true
}

func ParseYuan(s string) (int64, error) {
	// @solution
	neg := strings.HasPrefix(s, "-")
	s = strings.TrimPrefix(s, "-")
	intPart, frac, hasDot := strings.Cut(s, ".")
	if !allDigits(intPart) || (hasDot && (!allDigits(frac) || len(frac) > 2)) {
		return 0, ErrBadAmount
	}
	yuan, err := strconv.ParseInt(intPart, 10, 64)
	if err != nil {
		return 0, ErrBadAmount
	}
	for len(frac) < 2 {
		frac += "0"
	}
	cents, _ := strconv.ParseInt(frac, 10, 64)
	total := yuan*100 + cents
	if neg {
		total = -total
	}
	return total, nil
	// @starter
	// return 0, ErrBadAmount // TODO
	// @end
}

func FormatCents(c int64) string {
	// @solution
	sign := ""
	if c < 0 {
		sign, c = "-", -c
	}
	return fmt.Sprintf("%s%d.%02d", sign, c/100, c%100)
	// @starter
	// return fmt.Sprintf("%.2f", float64(c)/100) // TODO: 不要经过浮点数
	// @end
}

func Split(total int64, n int) []int64 {
	// @solution
	out := make([]int64, n)
	base, rem := total/int64(n), total%int64(n)
	for i := range out {
		out[i] = base
		if int64(i) < rem {
			out[i]++
		}
	}
	return out
	// @starter
	// return nil // TODO
	// @end
}

}

	// @solution
	if !slices.Contains(allowed[o.Status], to) {

package main

import (
	"slices"
	"time"
)

func Percentile(samples []time.Duration, p float64) time.Duration {
	// @solution
	if len(samples) == 0 {
		return 0
	}
	s := slices.Clone(samples)
	slices.Sort(s)
	rank := int(p / 100 * float64(len(s)))
	if float64(rank) < p/100*float64(len(s)) {
		rank++ // 向上取整
	}
	rank = max(rank, 1)
	return s[rank-1]
	// @starter
	// return 0 // TODO
	// @end
}

	// @solution
	need := peakQPS * 100
	capacity := perInstanceQPS * targetUtil
	return (need+capacity-1)/capacity + extra
	// @starter
	// return 0 // TODO
	// @end
}

	if s.inflight >= max {

	}
	s.inflight++
	return true

	// @end
}

func (s *Shedder) Release() {
	s.mu.Lock()
	s.inflight--
	s.mu.Unlock()
}
--- test
package main

import (
	"sync"
	"testing"
)

func TestShedder(t *testing.T) {
	s := NewShedder(10)
	low := 0
	for i := 0; i < 20; i++ {
		if s.Acquire(false) {
			low++
		}
	}
	if low != 8 {
		t.Fatalf("低优先级最多占用 8 个，实际 %d", low)
	}
	high := 0
	for i := 0; i < 20; i++ {
		if s.Acquire(true) {
			high++
		}
	}
	if high != 2 {
		t.Fatalf("剩下的 2 个容量留给高优先级，实际 %d", high)
	}
	s.Release()
	if s.Acquire(false) {
		t.Fatal("总在途 9 个时低优先级仍应被拒绝")
	}
	if !s.Acquire(true) {
		t.Fatal("高优先级应能拿到释放出来的容量")
	}
}

func TestConcurrent(t *testing.T) {
	s := NewShedder(50)
	var wg sync.WaitGroup
	var mu sync.Mutex
	peak, cur := 0, 0
	for i := 0; i < 500; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			if s.Acquire(i%2 == 0) {
				mu.Lock()
				cur++
				peak = max(peak, cur)
				mu.Unlock()
				mu.Lock()
				cur--
				mu.Unlock()
				s.Release()
			}
		}()
	}
	wg.Wait()
	if peak > 50 {
		t.Fatalf("在途峰值 %d 超过上限", peak)
	}
}

=== arch-20-1
title: 单元化路由与故障切换
--- desc
异地多活常按用户 ID 把用户划分到不同的单元：`uid % 100` 落在 `[From, To)` 区间的用户属于这个单元，一个用户的读写都在自己的单元内完成。

实现 `Router`：

- `Route(uid)`：返回用户所属单元；如果这个单元被标记为故障，切换到它配置的备份单元（备份也故障则返回错误）；
- `MarkDown(unit)` / `MarkUp(unit)`。
--- solution
package main

import "errors"

type Unit struct {
	Name     string
	From, To int // uid % 100 ∈ [From, To)
	Backup   string
}

type Router struct {
	units []Unit
	down  map[string]bool
}

func NewRouter(units []Unit) *Router { return &Router{units: units, down: map[string]bool{}} }

func (r *Router) MarkDown(unit string) { r.down[unit] = true }
func (r *Router) MarkUp(unit string)   { delete(r.down, unit) }

func (r *Router) Route(uid int64) (string, error) {
	// @solution
	slot := int(uid % 100)
	for _, u := range r.units {
		if slot >= u.From && slot < u.To {
			if !r.down[u.Name] {
				return u.Name, nil
			}
			if u.Backup != "" && !r.down[u.Backup] {
				return u.Backup, nil
			}
			return "", errors.New("unit and backup are both down")
		}
	}
	return "", errors.New("no unit for uid")
	// @starter
	// return "", errors.New("TODO")
	// @end
}
--- test
package main

import "testing"

func TestRoute(t *testing.T) {
	r := NewRouter([]Unit{
		{"shanghai", 0, 50, "beijing"},
		{"beijing", 50, 100, "shanghai"},
	})
	cases := map[int64]string{10: "shanghai", 149: "shanghai", 1050: "beijing", 99: "beijing"}
	for uid, want := range cases {
		if got, err := r.Route(uid); err != nil || got != want {
			t.Errorf("Route(%d) = %s, %v，期望 %s", uid, got, err, want)
		}
	}
	r.MarkDown("shanghai")
	if got, _ := r.Route(10); got != "beijing" {
		t.Fatalf("上海故障后应切到北京，实际 %s", got)
	}
	if got, _ := r.Route(99); got != "beijing" {
		t.Fatal("北京的用户不受影响")
	}
	r.MarkDown("beijing")
	if _, err := r.Route(10); err == nil {
		t.Fatal("两个单元都故障时应返回错误")
	}
	r.MarkUp("shanghai")
	if got, _ := r.Route(10); got != "shanghai" {
		t.Fatal("恢复后应切回")
	}
}

=== arch-20-2
title: 跨地域数据合并：最后写入者胜
--- desc
多个地域异步同步数据时，同一条记录可能在两边都被修改。最简单的冲突解决策略是 **LWW（Last Write Wins）**：

`Merge(a, b)`：时间戳大的胜出；时间戳相同时，按地域名字典序**大的**胜出。

要求 `Merge` 满足交换律和结合律——无论各地域以什么顺序收到同步消息，最终都收敛到同一个值。这也是 LWW 的代价：被覆盖的那次写入会悄悄丢失。
--- solution
package main

type Versioned struct {
	Value string
	TS    int64
	Unit  string
}

func Merge(a, b Versioned) Versioned {
	// @solution
	if a.TS != b.TS {
		if a.TS > b.TS {
			return a
		}
		return b
	}
	if a.Unit >= b.Unit {
		return a
	}
	return b
	// @starter
	// return b // TODO
	// @end
}
--- test
package main

import "testing"

func TestMerge(t *testing.T) {
	sh := Versioned{"addr-A", 100, "shanghai"}
	bj := Versioned{"addr-B", 105, "beijing"}
	if Merge(sh, bj) != bj || Merge(bj, sh) != bj {
		t.Fatal("时间戳大的胜出，且与参数顺序无关")
	}
	tie1 := Versioned{"x", 100, "beijing"}
	tie2 := Versioned{"y", 100, "shanghai"}
	if Merge(tie1, tie2) != tie2 || Merge(tie2, tie1) != tie2 {
		t.Fatal("时间戳相同时按地域名决胜，结果必须一致")
	}
	a, b, c := Versioned{"a", 3, "u1"}, Versioned{"b", 5, "u2"}, Versioned{"c", 5, "u3"}
	if Merge(Merge(a, b), c) != Merge(a, Merge(b, c)) {
		t.Fatal("Merge 应满足结合律")
	}
}
