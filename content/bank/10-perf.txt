=== perf-01-1
title: 解析 /proc/loadavg 判断负载趋势
--- desc
`uptime` 的数据来自 `/proc/loadavg`，形如 `0.52 0.58 0.59 3/467 12345`：1/5/15 分钟平均负载、当前可运行任务数/总任务数、最近的 PID。

- `ParseLoadAvg(s)`：解析出这些字段，格式不对返回错误；
- `Trend()`：`L1 > L15 × 1.2` 为 `"rising"`（负载在上升，问题正在发生），`L1 < L15 × 0.8` 为 `"falling"`，否则 `"steady"`；
- `Overloaded(cores)`：1 分钟负载是否超过 CPU 核数。
--- solution
package main

import (
	"fmt"
	"strconv"
	"strings"
)

type LoadAvg struct {
	L1, L5, L15     float64
	Running, Total  int
	LastPID         int
}

func ParseLoadAvg(s string) (LoadAvg, error) {
	var l LoadAvg
	// @solution
	f := strings.Fields(s)
	if len(f) != 5 {
		return l, fmt.Errorf("unexpected format %q", s)
	}
	var err error
	if l.L1, err = strconv.ParseFloat(f[0], 64); err != nil {
		return l, err
	}
	if l.L5, err = strconv.ParseFloat(f[1], 64); err != nil {
		return l, err
	}
	if l.L15, err = strconv.ParseFloat(f[2], 64); err != nil {
		return l, err
	}
	run, total, ok := strings.Cut(f[3], "/")
	if !ok {
		return l, fmt.Errorf("bad task field %q", f[3])
	}
	if l.Running, err = strconv.Atoi(run); err != nil {
		return l, err
	}
	if l.Total, err = strconv.Atoi(total); err != nil {
		return l, err
	}
	l.LastPID, err = strconv.Atoi(f[4])
	return l, err
	// @starter
	// return l, fmt.Errorf("TODO")
	// @end
}

func (l LoadAvg) Trend() string {
	// @solution
	switch {
	case l.L1 > l.L15*1.2:
		return "rising"
	case l.L1 < l.L15*0.8:
		return "falling"
	}
	return "steady"
	// @starter
	// return "" // TODO
	// @end
}

func (l LoadAvg) Overloaded(cores int) bool { return l.L1 > float64(cores) }

import "testing"

	if err != nil || l != want {
		t.Fatalf("got %+v, %v", l, err)
	}

		if _, err := ParseLoadAvg(bad); err == nil {

	cases := map[string]string{
		"12.5 6.1 3.0 20/900 1": "rising",
		"1.0 3.0 5.0 1/900 1":   "falling",
		"4.0 4.1 3.9 4/900 1":   "steady",

=== perf-02-1
title: 模拟内核计算 load average
--- desc
Linux 每 5 秒统计一次「活跃任务数」——**R（可运行）和 D（不可中断睡眠，通常在等 IO）状态**的任务都算——然后用指数衰减更新 1/5/15 分钟负载：

`load = load × e + active × (1 − e)`，其中 `e = exp(−5 / 周期秒数)`（周期分别为 60、300、900）。

- `ActiveTasks(states)`：统计状态为 `R` 或 `D` 的任务数——这就是为什么大量进程卡在磁盘 IO 时，CPU 不忙但负载很高；
- `(*Load).Update(active)`：做一次更新。
--- solution
package main

import "math"

func ActiveTasks(states []string) int {
	// @solution
	n := 0
	for _, s := range states {
		if s == "R" || s == "D" {
			n++
		}
	}
	return n
	// @starter
	// return 0 // TODO
	// @end
}

type Load struct {
	L1, L5, L15 float64
}

func decay(load float64, active int, period float64) float64 {
	e := math.Exp(-5 / period)
	return load*e + float64(active)*(1-e)
}

func (l *Load) Update(active int) {
	// @solution
	l.L1 = decay(l.L1, active, 60)
	l.L5 = decay(l.L5, active, 300)
	l.L15 = decay(l.L15, active, 900)
	// @starter
	// // TODO
	// @end
}

	"math"
	"testing"

func TestActiveTasks(t *testing.T) {

		t.Fatalf("ActiveTasks = %d，期望 4（R 和 D）", n)
	}

	var l Load
	for i := 0; i < 12; i++ { // 一分钟，活跃任务一直是 4
		l.Update(4)
	}
	near := func(a, b float64) bool { return math.Abs(a-b) < 1e-6 }
	if !near(l.L1, 4*(1-math.Exp(-1))) || !near(l.L5, 4*(1-math.Exp(-0.2))) || !near(l.L15, 4*(1-math.Exp(-60.0/900))) {
		t.Fatalf("一分钟后负载 = %+v", l)
	}
	for i := 0; i < 2000; i++ {
		l.Update(4)
	}
	if math.Abs(l.L15-4) > 1e-3 {
		t.Fatalf("长时间之后应趋近 4，实际 %v", l.L15)
	}
}

=== perf-03-1
title: 从 /proc/stat 计算 CPU 使用率
--- desc
`top` 里的 us、sy、wa 等百分比，是对 `/proc/stat` 里 `cpu` 行的**两次采样做差**算出来的。这一行的字段依次是 user、nice、system、idle、iowait、irq、softirq、steal（单位是时钟滴答）。

- `ParseCPULine(line)`：解析前 8 个数字；
- `Percent(a, b)`：返回两次采样之间各项占总增量的百分比：`us`（user + nice）、`sy`、`wa`、`si`（softirq）、`st`、`idle`。
--- solution
package main

import (
	"errors"
	"strconv"
	"strings"
)

type CPUTimes struct {
	User, Nice, System, Idle, IOWait, IRQ, SoftIRQ, Steal uint64
}

func (c CPUTimes) total() uint64 {
	return c.User + c.Nice + c.System + c.Idle + c.IOWait + c.IRQ + c.SoftIRQ + c.Steal
}

func ParseCPULine(line string) (CPUTimes, error) {
	var c CPUTimes
	// @solution
	f := strings.Fields(line)
	if len(f) < 9 || f[0] != "cpu" {
		return c, errors.New("not a cpu line")
	}
	ptrs := []*uint64{&c.User, &c.Nice, &c.System, &c.Idle, &c.IOWait, &c.IRQ, &c.SoftIRQ, &c.Steal}
	for i, p := range ptrs {
		v, err := strconv.ParseUint(f[i+1], 10, 64)
		if err != nil {
			return c, err
		}
		*p = v
	}
	return c, nil
	// @starter
	// return c, errors.New("TODO")
	// @end
}

func Percent(a, b CPUTimes) map[string]float64 {
	// @solution
	total := float64(b.total() - a.total())
	pct := func(x, y uint64) float64 { return float64(y-x) / total * 100 }
	return map[string]float64{
		"us":   pct(a.User+a.Nice, b.User+b.Nice),
		"sy":   pct(a.System, b.System),
		"wa":   pct(a.IOWait, b.IOWait),
		"si":   pct(a.SoftIRQ, b.SoftIRQ),
		"st":   pct(a.Steal, b.Steal),
		"idle": pct(a.Idle, b.Idle),
	}
	// @starter
	// return nil // TODO
	// @end
}
--- test
package main

import (
	"math"
	"testing"
)

func TestCPU(t *testing.T) {
	a, err := ParseCPULine("cpu  4705 150 1120 16250 520 0 30 10 0 0")
	if err != nil || a.User != 4705 || a.Steal != 10 {
		t.Fatalf("ParseCPULine = %+v, %v", a, err)
	}
	b, _ := ParseCPULine("cpu  5305 200 1320 16650 720 0 130 50 0 0")
	// 增量：user 600 nice 50 system 200 idle 400 iowait 200 irq 0 softirq 100 steal 40 → 总 1590
	p := Percent(a, b)
	want := map[string]float64{"us": 650.0 / 1590 * 100, "sy": 200.0 / 1590 * 100, "wa": 200.0 / 1590 * 100,
		"si": 100.0 / 1590 * 100, "st": 40.0 / 1590 * 100, "idle": 400.0 / 1590 * 100}
	for k, v := range want {
		if math.Abs(p[k]-v) > 1e-9 {
			t.Errorf("%s = %.3f%%，期望 %.3f%%", k, p[k], v)
		}
	}
	if _, err := ParseCPULine("cpu0 1 2 3"); err == nil {
		t.Error("格式不对应返回错误")
	}
}

=== perf-04-1
title: 找出最耗 CPU 的线程
--- desc
`ps -L -o tid,pcpu,comm -p <pid>` 会列出进程的每个线程及其 CPU 使用率：

```text
  TID %CPU COMMAND
 4101 85.3 app
 4102  0.1 app
```

实现 `TopThreads(output, n)`：解析输出（跳过表头），按 CPU 从高到低返回前 n 个线程（CPU 相同按 TID 升序）。

注意：对 Java 可以把线程 ID 转成十六进制去 jstack 里找；但 **Go 的 goroutine 会在线程之间迁移**，线程 ID 对应不到具体的 goroutine，定位到「哪个进程、是否单线程打满」之后，要用 CPU profile 找热点函数。
--- solution
package main

import (
	"cmp"
	"slices"
	"strconv"
	"strings"
)

type Thread struct {
	TID int
	CPU float64
	Cmd string
}

func TopThreads(output string, n int) ([]Thread, error) {
	// @solution
	var out []Thread
	for i, line := range strings.Split(strings.TrimSpace(output), "\n") {
		f := strings.Fields(line)
		if i == 0 || len(f) < 3 {
			continue
		}
		tid, err := strconv.Atoi(f[0])
		if err != nil {
			return nil, err
		}
		cpu, err := strconv.ParseFloat(f[1], 64)
		if err != nil {
			return nil, err
		}
		out = append(out, Thread{tid, cpu, strings.Join(f[2:], " ")})
	}
	slices.SortFunc(out, func(a, b Thread) int {
		return cmp.Or(cmp.Compare(b.CPU, a.CPU), cmp.Compare(a.TID, b.TID))
	})
	return out[:min(n, len(out))], nil
	// @starter
	// return nil, nil // TODO
	// @end
}
--- test
package main

import (
	"slices"
	"testing"
)

const psOut = `  TID %CPU COMMAND
 4101 85.3 app
 4102  0.1 app
 4110 12.0 app
 4111 12.0 app
 4120  3.2 app worker
`

func TestTopThreads(t *testing.T) {
	got, err := TopThreads(psOut, 3)
	want := []Thread{{4101, 85.3, "app"}, {4110, 12.0, "app"}, {4111, 12.0, "app"}}
	if err != nil || !slices.Equal(got, want) {
		t.Fatalf("TopThreads = %v, %v", got, err)

	}
}

=== perf-05-1
title: 根据 /proc/meminfo 判断内存压力
--- desc
`free` 里最该看的是 **available**（还能分配给新程序的内存，包含可回收的缓存），而不是 free。

- `ParseMeminfo(text)`：解析 `/proc/meminfo`（`MemTotal:       32768000 kB` 这样的行），返回 `字段名 → kB`；
- `Pressure(m)`：返回 `availPct = MemAvailable / MemTotal × 100`，`swapPct = (SwapTotal − SwapFree) / SwapTotal × 100`（没有 swap 时为 0），以及是否紧张：`availPct < 10` 或 `swapPct > 50`。
--- solution
package main

import (
	"strconv"
	"strings"
)

func ParseMeminfo(text string) (map[string]uint64, error) {
	m := map[string]uint64{}
	// @solution
	for _, line := range strings.Split(text, "\n") {
		key, rest, ok := strings.Cut(line, ":")
		if !ok {
			continue
		}
		f := strings.Fields(rest)
		if len(f) == 0 {
			continue
		}
		v, err := strconv.ParseUint(f[0], 10, 64)
		if err != nil {
			return nil, err
		}
		m[strings.TrimSpace(key)] = v
	}
	// @starter
	// // TODO
	// @end
	return m, nil
}

func Pressure(m map[string]uint64) (availPct, swapPct float64, tight bool) {
	// @solution
	availPct = float64(m["MemAvailable"]) / float64(m["MemTotal"]) * 100
	if m["SwapTotal"] > 0 {
		swapPct = float64(m["SwapTotal"]-m["SwapFree"]) / float64(m["SwapTotal"]) * 100
	}
	return availPct, swapPct, availPct < 10 || swapPct > 50
	// @starter
	// return 0, 0, false // TODO
	// @end
}

=== perf-06-1
title: 解析 iostat 并诊断磁盘
--- desc
`iostat -x` 的列在不同版本里顺序不同，应该**按表头的列名**取值，而不是写死下标。实现：

- `ParseIostat(output)`：返回每个设备的 `Device{Name, Fields}`，`Fields` 是「列名 → 数值」；
- `Diagnose(d, ssd)`：按顺序检查：`aqu-sz > 1` → `"queueing"`；`r_await` 或 `w_await` 超过阈值（SSD 为 5ms，机械盘为 20ms）→ `"slow-io"`；`%util > 90` 且**不是** SSD → `"saturated"`（SSD 可以并行处理请求，100% util 不一定饱和）。
--- solution
package main

import (
	"errors"
	"strconv"
	"strings"
)

type Device struct {
	Name   string
	Fields map[string]float64
}

func ParseIostat(output string) ([]Device, error) {
	// @solution
	var header []string
	var out []Device
	for _, line := range strings.Split(output, "\n") {
		f := strings.Fields(line)
		if len(f) == 0 {
			continue
		}
		if f[0] == "Device" || f[0] == "Device:" {
			header = f
			continue
		}
		if header == nil {
			continue
		}
		if len(f) != len(header) {
			return nil, errors.New("column count mismatch")
		}
		d := Device{Name: f[0], Fields: map[string]float64{}}
		for i := 1; i < len(f); i++ {
			v, err := strconv.ParseFloat(f[i], 64)
			if err != nil {
				return nil, err
			}
			d.Fields[header[i]] = v
		}
		out = append(out, d)
	}
	return out, nil
	// @starter
	// return nil, errors.New("TODO")
	// @end
}

func Diagnose(d Device, ssd bool) []string {
	var out []string
	// @solution
	limit := 20.0
	if ssd {
		limit = 5
	}
	if d.Fields["aqu-sz"] > 1 {
		out = append(out, "queueing")
	}
	if d.Fields["r_await"] > limit || d.Fields["w_await"] > limit {
		out = append(out, "slow-io")
	}
	if !ssd && d.Fields["%util"] > 90 {
		out = append(out, "saturated")
	}
	// @starter
	// // TODO
	// @end
	return out
}

=== perf-07-1
title: 从 /proc/net/snmp 计算重传率
--- desc
`/proc/net/snmp` 里 TCP 的统计是两行：第一行是字段名，第二行是对应的值，都以 `Tcp:` 开头。

- `ParseTCP(text)`：找到这两行，返回「字段名 → 值」（值可能是负数，比如 `MaxConn` 为 -1）；
- `RetransRate(before, after)`：两次采样之间的重传率 = `ΔRetransSegs / ΔOutSegs × 100`。一般超过 1% 就值得关注。
--- solution
package main

import (
	"errors"
	"strconv"
	"strings"
)

func ParseTCP(text string) (map[string]int64, error) {
	// @solution
	var rows [][]string
	for _, line := range strings.Split(text, "\n") {
		if strings.HasPrefix(line, "Tcp:") {
			rows = append(rows, strings.Fields(line)[1:])
		}
	}
	if len(rows) != 2 || len(rows[0]) != len(rows[1]) {
		return nil, errors.New("Tcp section not found")
	}
	m := map[string]int64{}
	for i, name := range rows[0] {
		v, err := strconv.ParseInt(rows[1][i], 10, 64)
		if err != nil {
			return nil, err
		}
		m[name] = v
	}
	return m, nil
	// @starter
	// return nil, errors.New("TODO")
	// @end
}

func RetransRate(before, after map[string]int64) float64 {
	// @solution
	out := after["OutSegs"] - before["OutSegs"]
	if out <= 0 {
		return 0
	}
	return float64(after["RetransSegs"]-before["RetransSegs"]) / float64(out) * 100
	// @starter
	// return 0 // TODO
	// @end
}

title: 解析 goroutine dump 定位泄漏
--- desc
`/debug/pprof/goroutine?debug=2` 和 `runtime.Stack(buf, true)` 的输出由一个个 goroutine 块组成：

```text

runtime.gopark(...)
	/usr/local/go/src/runtime/proc.go:435 +0xce
main.worker(0xc000020180)
	/app/main.go:21 +0x25

	/app/main.go:30 +0x45
```

实现：

- `States(dump)`：统计每个状态的 goroutine 数（方括号里逗号之前的部分，如 `chan receive`）；
- `TopFunc(dump)`：统计每个 goroutine **第一个业务函数**（跳过以 `runtime.`、`sync.`、`internal/`、`time.` 开头的帧；函数名去掉括号及之后的参数），返回「函数 → 数量」——找到数量最多的那个，就找到了泄漏点。

测试会启动 5 个阻塞的 goroutine，用真实的 `runtime.Stack` 输出来检验。
--- solution
package main

import "strings"

var skip = []string{"runtime.", "sync.", "internal/", "time."}

func blocks(dump string) [][]string {
	var out [][]string
	for _, b := range strings.Split(strings.TrimSpace(dump), "\n\n") {
		lines := strings.Split(b, "\n")
		if len(lines) > 0 && strings.HasPrefix(lines[0], "goroutine ") {
			out = append(out, lines)
		}
	}
	return out
}

func States(dump string) map[string]int {
	m := map[string]int{}
	// @solution
	for _, b := range blocks(dump) {
		head := b[0]
		start, end := strings.IndexByte(head, '['), strings.IndexByte(head, ']')
		if start < 0 || end < start {
			continue
		}
		state, _, _ := strings.Cut(head[start+1:end], ",")
		m[state]++
	}
	// @starter
	// // TODO
	// @end
	return m
}

func TopFunc(dump string) map[string]int {
	m := map[string]int{}
	// @solution
	for _, b := range blocks(dump) {
	frames:
		for _, line := range b[1:] {
			if strings.HasPrefix(line, "\t") || strings.HasPrefix(line, "created by ") {
				continue
			}
			for _, p := range skip {
				if strings.HasPrefix(line, p) {
					continue frames
				}
			}
			fn := line
			if i := strings.LastIndexByte(fn, '('); i > 0 {
				fn = fn[:i]
			}
			m[fn]++
			break
		}
	}
	// @starter
	// // TODO
	// @end
	return m
}

=== perf-12-1
title: 分阶段计时，找出时间花在哪
--- desc
CPU 不高但接口很慢，说明时间花在了「等待」上。给请求的各个阶段打点计时，是最直接的办法（链路追踪的 span 也是同样的思路）。

实现 `Timer`：

- `Start(stage)` 开始计时并返回一个结束函数，配合 `defer t.Start("db")()` 使用；同一个阶段可以出现多次，耗时累加；
- `Report()`：按总耗时从大到小返回 `"阶段=耗时"`（耗时相同按名字），耗时用 `time.Duration` 的字符串形式。

时间通过注入的 `now` 获取。
--- solution
package main

import (
	"cmp"
	"maps"
	"slices"
	"time"
)

type Timer struct {
	now    func() time.Time
	totals map[string]time.Duration
}

func NewTimer(now func() time.Time) *Timer {
	return &Timer{now: now, totals: map[string]time.Duration{}}
}

func (t *Timer) Start(stage string) func() {
	// @solution
	start := t.now()
	return func() { t.totals[stage] += t.now().Sub(start) }
	// @starter
	// return func() {} // TODO
	// @end
}

func (t *Timer) Report() []string {
	// @solution
	stages := slices.Collect(maps.Keys(t.totals))
	slices.SortFunc(stages, func(a, b string) int {
		return cmp.Or(cmp.Compare(t.totals[b], t.totals[a]), cmp.Compare(a, b))
	})
	out := make([]string, len(stages))
	for i, s := range stages {
		out[i] = s + "=" + t.totals[s].String()
	}
	return out
	// @starter
	// return nil // TODO
	// @end
}

=== perf-14-1
title: 读取 runtime/metrics 的 GC 暂停分布
--- desc
`runtime/metrics` 提供了 GC 暂停时间的直方图（`/sched/pauses/total/gc:seconds`），可以直接导出到监控系统，用来判断延迟毛刺是不是 GC 引起的。

直方图类型是 `metrics.Float64Histogram`：`Counts[i]` 是落在 `[Buckets[i], Buckets[i+1])` 区间的样本数，所以 `len(Buckets) == len(Counts) + 1`，边界可能是 ±Inf。

- `Quantile(h, q)`：返回第一个累计计数达到 `q × 总数` 的桶的**上界**；上界为 +Inf 时返回下界；没有样本返回 0；
- `GCPauses()`：用 `metrics.Read` 读取真实的直方图。
--- solution
package main

import (
	"math"
	"runtime/metrics"
)

func Quantile(h *metrics.Float64Histogram, q float64) float64 {
	// @solution
	var total uint64
	for _, c := range h.Counts {
		total += c
	}
	if total == 0 {
		return 0
	}
	rank := q * float64(total)
	var cum uint64
	for i, c := range h.Counts {
		cum += c
		if c > 0 && float64(cum) >= rank {
			if upper := h.Buckets[i+1]; !math.IsInf(upper, 1) {
				return upper
			}
			return h.Buckets[i]
		}
	}
	return h.Buckets[len(h.Buckets)-1]
	// @starter
	// return 0 // TODO
	// @end
}

func GCPauses() *metrics.Float64Histogram {
	// @solution
	s := []metrics.Sample{{Name: "/sched/pauses/total/gc:seconds"}}
	metrics.Read(s)
	if s[0].Value.Kind() != metrics.KindFloat64Histogram {
		return &metrics.Float64Histogram{}
	}
	return s[0].Value.Float64Histogram()
	// @starter
	// return &metrics.Float64Histogram{} // TODO
	// @end
}

title: 周期采集 profile 并保留最近几份
race: true
--- desc
持续性能剖析的核心很简单：定期采集一份 profile，保留最近的若干份，出问题时回看。实现 `Collector`：

- `Run(ctx, interval, capture)`：每隔 `interval` 调用一次 `capture()`，只保留最近 `keep` 份；`ctx` 取消时返回。用 `time.Ticker`，退出前 `Stop`；
- `Latest()`：返回保存的 profile（从旧到新），返回的切片不能和内部共享，可以被并发调用。
--- solution
package main

import (
	"context"
	"slices"
	"sync"
	"time"
)

type Collector struct {
	mu       sync.Mutex
	keep     int
	profiles [][]byte
}

func NewCollector(keep int) *Collector { return &Collector{keep: keep} }

func (c *Collector) Run(ctx context.Context, interval time.Duration, capture func() []byte) {
	// @solution
	t := time.NewTicker(interval)
	defer t.Stop()
	for {
		select {
		case <-t.C:
			p := capture()
			c.mu.Lock()
			c.profiles = append(c.profiles, p)
			if len(c.profiles) > c.keep {
				c.profiles = slices.Delete(c.profiles, 0, len(c.profiles)-c.keep)
			}
			c.mu.Unlock()
		case <-ctx.Done():
			return
		}
	}
	// @starter
	// // TODO
	// @end
}

func (c *Collector) Latest() [][]byte {
	c.mu.Lock()
	defer c.mu.Unlock()
	return slices.Clone(c.profiles)
}

=== perf-18-1
title: 用内存预算做背压，防止 OOM
race: true
--- desc
OOM 经常是突发的：一批大请求同时把数据加载进内存。与其等着被 OOM Killer 杀掉，不如给「处理中的数据量」设一个上限，超过就快速拒绝（返回 503，让客户端稍后重试）。

实现 `Budget`：

- `TryAcquire(n)`：占用 n 字节，占用后总量不超过 `limit` 时成功返回 true，否则返回 false 且不占用；
- `Release(n)`：释放；
- `Used()`：当前占用量。
--- solution
package main

import "sync"

type Budget struct {
	mu    sync.Mutex
	limit int64
	used  int64
}

func NewBudget(limit int64) *Budget { return &Budget{limit: limit} }

func (b *Budget) TryAcquire(n int64) bool {
	// @solution
	b.mu.Lock()
	defer b.mu.Unlock()
	if b.used+n > b.limit {
		return false
	}
	b.used += n
	return true
	// @starter
	// b.mu.Lock()
	// defer b.mu.Unlock()
	// b.used += n // TODO: 没有检查上限
	// return true
	// @end
}

func (b *Budget) Release(n int64) {
	b.mu.Lock()
	b.used -= n
	b.mu.Unlock()
}

func (b *Budget) Used() int64 {
	b.mu.Lock()
	defer b.mu.Unlock()
	return b.used
}

	b.Release(60)

		t.Fatalf("Used = %d", b.Used())

--- desc

------ ----------- ----------- --------- --------- ----------------

- `Top(rows, n)`：按耗时从大到小取前 n 个。

package main

import (
	"cmp"
	"slices"
	"strconv"
	"strings"
)

type Syscall struct {
	Name    string
	Seconds float64
	Calls   int
	Errors  int
}

func ParseStrace(output string) ([]Syscall, error) {
	var out []Syscall
	// @solution
	for _, line := range strings.Split(output, "\n") {
		f := strings.Fields(line)
		if len(f) < 5 || strings.HasPrefix(f[0], "-") || f[0] == "%" || f[len(f)-1] == "total" {
			continue
		}
		var s Syscall
		var err error
		s.Name = f[len(f)-1]
		if s.Seconds, err = strconv.ParseFloat(f[1], 64); err != nil {
			return nil, err
		}
		if s.Calls, err = strconv.Atoi(f[3]); err != nil {
			return nil, err
		}
		if len(f) == 6 {
			if s.Errors, err = strconv.Atoi(f[4]); err != nil {
				return nil, err
			}
		}
		out = append(out, s)
	}
	// @starter
	// // TODO
	// @end
	return out, nil
}

func Top(rows []Syscall, n int) []Syscall {
	s := slices.Clone(rows)
	slices.SortStableFunc(s, func(a, b Syscall) int { return cmp.Compare(b.Seconds, a.Seconds) })
	return s[:min(n, len(s))]
}

	"io"
	"net/http"
)

	if err != nil {
		return 0, err
	}
	defer resp.Body.Close()
	io.Copy(io.Discard, io.LimitReader(resp.Body, 64<<10))
	return resp.StatusCode, nil
}

实现 `WithSlowLog(threshold, now, log, next)`，返回包装后的查询函数：

package main

import (
	"context"
	"fmt"
	"strings"
	"time"
)

type QueryFunc func(ctx context.Context, sql string, args ...any) error

func WithSlowLog(threshold time.Duration, now func() time.Time, log func(string), next QueryFunc) QueryFunc {
	return func(ctx context.Context, sql string, args ...any) error {
		// @solution
		start := now()
		err := next(ctx, sql, args...)
		if d := now().Sub(start); d >= threshold {
			s := strings.Join(strings.Fields(sql), " ")
			if len(s) > 100 {
				s = s[:100] + "..."
			}
			log(fmt.Sprintf("slow query (%v): %s", d, s))
		}
		return err
		// @starter
		// return next(ctx, sql, args...) // TODO
		// @end
	}
}

=== perf-24-1
title: 批量处理减少唤醒和切换
race: true
--- desc
每来一条消息就唤醒一次消费者、写一次下游，会产生大量的 goroutine 切换和系统调用。常见的优化是**攒批**。

实现 `Batch(in, max, wait)`：返回一个输出 channel，按以下规则输出 `[]int`：

- 攒够 `max` 条立即输出一批；
- 距离这一批的**第一条**数据到达已经过了 `wait`，即使没攒够也输出；
- `in` 关闭时把剩余的数据作为最后一批输出（如果有），然后关闭输出 channel。

每一批都是新的切片，不能复用底层数组。
--- solution
package main

import "time"

func Batch(in <-chan int, max int, wait time.Duration) <-chan []int {
	out := make(chan []int)
	go func() {
		defer close(out)
		// @solution
		var batch []int
		var timer *time.Timer
		var timeout <-chan time.Time
		flush := func() {
			if len(batch) > 0 {
				out <- batch
				batch = nil
			}
			if timer != nil {
				timer.Stop()
				timer, timeout = nil, nil
			}
		}
		for {
			select {
			case v, ok := <-in:
				if !ok {
					flush()
					return
				}
				if len(batch) == 0 {
					timer = time.NewTimer(wait)
					timeout = timer.C
				}
				batch = append(batch, v)
				if len(batch) >= max {
					flush()
				}
			case <-timeout:
				flush()
			}
		}
		// @starter
		// for v := range in {
		// 	out <- []int{v} // TODO: 每条一批
		// }
		// @end
	}()
	return out
}

=== perf-25-1
title: 最近 N 秒的 p99
--- desc
监控面板上的 p99 是**滑动窗口**内的统计。实现一个按秒分桶的 `Window`：

- `Add(at, d)`：记录一个请求在 `at` 时刻完成、耗时 `d`；
- `P99(now)`：统计最近 `size` 秒（即 Unix 秒数在 `(now − size, now]` 之间）的所有样本，用最近秩法返回 p99：排序后取第 `ceil(0.99 × n)` 个；没有样本返回 0。

用 `size` 个桶的环形数组，桶下标为 `秒数 % size`；写入时如果桶里存的是更早的秒，先清空。这样内存是固定的，不会无限增长。
--- solution
package main

import (
	"slices"
	"time"
)

type bucket struct {
	sec     int64
	samples []time.Duration
}

type Window struct {
	size    int
	buckets []bucket
}

func NewWindow(size int) *Window { return &Window{size: size, buckets: make([]bucket, size)} }

func (w *Window) Add(at time.Time, d time.Duration) {
	// @solution
	sec := at.Unix()
	b := &w.buckets[sec%int64(w.size)]
	if b.sec != sec {
		b.sec, b.samples = sec, b.samples[:0]
	}
	b.samples = append(b.samples, d)
	// @starter
	// // TODO
	// @end
}

func (w *Window) P99(now time.Time) time.Duration {
	// @solution
	var all []time.Duration
	cur := now.Unix()
	for _, b := range w.buckets {
		if b.sec > cur-int64(w.size) && b.sec <= cur {
			all = append(all, b.samples...)
		}
	}
	if len(all) == 0 {
		return 0
	}
	slices.Sort(all)
	rank := (len(all)*99 + 99) / 100
	return all[rank-1]
	// @starter
	// return 0 // TODO
	// @end
}

=== perf-27-1
title: 解析 ping 统计判断丢包
--- desc
排查网络问题时，`ping` 的最后两行最有用：

```text
--- 10.0.0.2 ping statistics ---
100 packets transmitted, 97 received, 3% packet loss, time 99130ms
rtt min/avg/max/mdev = 0.210/0.452/12.301/1.203 ms
```

实现 `ParsePing(output)`，返回发送数、接收数、丢包率（自己根据发送和接收计算，单位 %），以及 min/avg/max/mdev（`time.Duration`）。全部丢包时没有 rtt 行，这时 rtt 都为 0。可以用 `fmt.Sscanf` 解析。

`max` 远大于 `avg` 往往意味着偶发的排队或重传；mdev 大说明抖动严重。
--- solution
package main

import (
	"errors"
	"fmt"
	"strings"
	"time"
)

type Ping struct {
	Sent, Received          int
	LossPct                 float64
	Min, Avg, Max, Mdev     time.Duration
}

func ParsePing(output string) (Ping, error) {
	var p Ping
	// @solution
	found := false
	ms := func(v float64) time.Duration { return time.Duration(v * float64(time.Millisecond)) }
	for _, line := range strings.Split(output, "\n") {
		line = strings.TrimSpace(line)
		switch {
		case strings.Contains(line, "packets transmitted"):
			if _, err := fmt.Sscanf(line, "%d packets transmitted, %d received", &p.Sent, &p.Received); err != nil {
				return p, err
			}
			found = true
		case strings.HasPrefix(line, "rtt ") || strings.HasPrefix(line, "round-trip "):
			_, vals, _ := strings.Cut(line, "= ")
			var a, b, c, d float64
			if _, err := fmt.Sscanf(vals, "%f/%f/%f/%f ms", &a, &b, &c, &d); err != nil {
				return p, err
			}
			p.Min, p.Avg, p.Max, p.Mdev = ms(a), ms(b), ms(c), ms(d)
		}
	}
	if !found || p.Sent == 0 {
		return p, errors.New("no statistics line")
	}
	p.LossPct = float64(p.Sent-p.Received) / float64(p.Sent) * 100
	return p, nil
	// @starter
	// return p, errors.New("TODO")
	// @end
}

=== perf-28-1
title: 实现 bcc 风格的 log2 直方图
--- desc
`biolatency`、`runqlat` 这些 eBPF 工具在内核里把延迟按 2 的幂分桶计数，只把桶传回用户态，开销极低。实现同样的分桶和展示：

- `Log2Hist(values)`：0 和 1 落在第 0 个桶 `[0, 1]`；其他值 v 落在第 k 个桶 `[2^k, 2^(k+1) − 1]`，其中 `k = bits.Len64(v) − 1`。返回从第 0 个桶到最高的非空桶的所有桶（中间的空桶也要有）；
- `Render(buckets, width)`：每个桶一行，格式为 `fmt.Sprintf("%8d -> %-8d : %-6d |%s|", Lo, Hi, Count, 星号)`，星号的个数 = `Count × width / 最大Count`（整数除法），右侧用空格补齐到 width。
--- solution
package main

import (
	"fmt"
	"math/bits"
	"strings"
)

type Bucket struct {
	Lo, Hi uint64
	Count  int
}

func Log2Hist(values []uint64) []Bucket {
	// @solution
	var counts []int
	for _, v := range values {
		k := 0
		if v > 1 {
			k = bits.Len64(v) - 1
		}
		for len(counts) <= k {
			counts = append(counts, 0)
		}
		counts[k]++
	}
	out := make([]Bucket, len(counts))
	for k, c := range counts {
		lo := uint64(1) << k
		if k == 0 {
			lo = 0
		}
		out[k] = Bucket{lo, uint64(1)<<(k+1) - 1, c}
	}
	return out
	// @starter
	// return nil // TODO
	// @end
}

func Render(buckets []Bucket, width int) string {
	// @solution
	maxCount := 0
	for _, b := range buckets {
		maxCount = max(maxCount, b.Count)
	}
	var sb strings.Builder
	for _, b := range buckets {
		stars := 0
		if maxCount > 0 {
			stars = b.Count * width / maxCount
		}
		bar := strings.Repeat("*", stars) + strings.Repeat(" ", width-stars)
		fmt.Fprintf(&sb, "%8d -> %-8d : %-6d |%s|\n", b.Lo, b.Hi, b.Count, bar)
	}
	return sb.String()
	// @starter
	// return "" // TODO
	// @end
}
