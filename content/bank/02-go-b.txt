=== go-11-1
title: 用指针接收者修改状态
--- desc
下面的 `Counter.Inc` 用的是值接收者，修改的是副本，所以计数永远是 0。

1. 把 `Inc` 改成指针接收者；
2. 改完后 `Counter` 值类型不再实现 `Incrementer`（`*Counter` 才实现），编译器会提示 `NewIncrementer` 出错，把它改成返回 `&Counter{}`。
--- solution
package main

type Incrementer interface {
	Inc()
	Value() int
}

type Counter struct{ n int }

func (c *Counter) Inc()      { c.n++ }
func (c Counter) Value() int { return c.n }

func NewIncrementer() Incrementer { return &Counter{} }
--- starter
package main

type Incrementer interface {
	Inc()
	Value() int
}

type Counter struct{ n int }

func (c Counter) Inc()       { c.n++ } // BUG: 值接收者修改的是副本
func (c Counter) Value() int { return c.n }

func NewIncrementer() Incrementer { return Counter{} }
--- test
package main

import "testing"

func TestIncrementer(t *testing.T) {
	inc := NewIncrementer()
	for i := 0; i < 3; i++ {
		inc.Inc()
	}
	if got := inc.Value(); got != 3 {
		t.Fatalf("Inc 三次后 Value() = %d，期望 3", got)
	}
}

func TestAddressableValue(t *testing.T) {
	var c Counter
	c.Inc() // 可寻址的变量调用指针方法时，编译器自动取地址 (&c).Inc()
	c.Inc()
	if c.Value() != 2 {
		t.Fatalf("Value() = %d，期望 2", c.Value())
	}
}
--- explain
方法集规则：`T` 的方法集只有值接收者方法，`*T` 包含全部方法。所以 `Inc` 改成指针接收者后，只有 `*Counter` 能赋给 `Incrementer`。

而对一个**可寻址**的变量 `c` 调用 `c.Inc()` 是合法的，编译器会改写成 `(&c).Inc()`；但 map 元素、函数返回值这类不可寻址的值就不行。

=== go-12-1
title: 修复闭包捕获同一个变量
--- desc
`MakeMultipliers(n)` 应该返回 n 个函数，第 i 个函数计算 `x * i`。

起始代码把循环变量声明在了循环**外面**，所有闭包捕获的是同一个 `i`，循环结束后它等于 n。即使在 Go 1.22 之后这种写法也有问题——1.22 的新语义只针对在 `for` 语句里声明的变量。
--- solution
package main

func MakeMultipliers(n int) []func(int) int {
	var fs []func(int) int
	for i := 0; i < n; i++ {
		fs = append(fs, func(x int) int { return x * i })
	}
	return fs
}
--- starter
package main

func MakeMultipliers(n int) []func(int) int {
	var fs []func(int) int
	var i int
	for i = 0; i < n; i++ {
		fs = append(fs, func(x int) int { return x * i })
	}
	return fs
}
--- test
package main

import "testing"

func TestMakeMultipliers(t *testing.T) {
	fs := MakeMultipliers(4)
	if len(fs) != 4 {
		t.Fatalf("应返回 4 个函数，实际 %d 个", len(fs))
	}
	for i, f := range fs {
		if got := f(10); got != 10*i {
			t.Errorf("fs[%d](10) = %d，期望 %d", i, got, 10*i)
		}
	}
}
--- explain
在 `for i := 0; ...` 里声明变量，Go 1.22+ 每次迭代都会创建新的 `i`。如果必须在循环外声明，就在循环体里拷贝一份：`i := i`。

=== go-13-1
title: 泛型去重与结构体做 map key
--- desc
- `Dedup[T comparable](s []T) []T`：去掉重复元素，保留第一次出现的顺序，不修改入参；
- `CountPoints(ps []Point) map[Point]int`：统计每个坐标出现的次数。结构体的字段都可比较，所以它可以直接做 map 的 key。
--- solution
package main

type Point struct{ X, Y int }

func Dedup[T comparable](s []T) []T {
	// @solution
	seen := make(map[T]struct{}, len(s))
	out := make([]T, 0, len(s))
	for _, v := range s {
		if _, ok := seen[v]; ok {
			continue
		}
		seen[v] = struct{}{}
		out = append(out, v)
	}
	return out
	// @starter
	// return nil // TODO
	// @end
}

func CountPoints(ps []Point) map[Point]int {
	// @solution
	m := make(map[Point]int)
	for _, p := range ps {
		m[p]++
	}
	return m
	// @starter
	// return nil // TODO
	// @end
}
--- test
package main

import (
	"slices"
	"testing"
)

func TestDedup(t *testing.T) {
	if got := Dedup([]int{3, 1, 3, 2, 1}); !slices.Equal(got, []int{3, 1, 2}) {
		t.Errorf("Dedup ints = %v，期望 [3 1 2]", got)
	}
	if got := Dedup([]string{"go", "rust", "go"}); !slices.Equal(got, []string{"go", "rust"}) {
		t.Errorf("Dedup strings = %v", got)
	}
	if got := Dedup([][2]int{{1, 2}, {1, 2}, {2, 1}}); len(got) != 2 {
		t.Errorf("数组也可比较，Dedup 结果应有 2 个元素，实际 %v", got)
	}
	in := []int{1, 1}
	Dedup(in)
	if !slices.Equal(in, []int{1, 1}) {
		t.Error("不应修改入参")
	}
}

func TestCountPoints(t *testing.T) {
	m := CountPoints([]Point{{1, 2}, {0, 0}, {1, 2}})
	if m[Point{1, 2}] != 2 || m[Point{0, 0}] != 1 || len(m) != 2 {
		t.Fatalf("CountPoints = %v", m)
	}
}

title: 接口 key 的运行时 panic
--- desc
`map[any]int` 用 `[]int` 做 key 能编译通过，但运行时会 panic：`hash of unhashable type []int`。

实现 `TryInsert(m, k, v)`：尝试执行 `m[k] = v`，如果 key 不可哈希，捕获 panic 并返回错误，而不是让程序崩溃。
--- solution
package main

import "fmt"

func TryInsert(m map[any]int, k any, v int) (err error) {
	// @solution
	defer func() {
		if r := recover(); r != nil {
			err = fmt.Errorf("invalid key: %v", r)
		}
	}()
	// @starter
	// // TODO: 捕获不可哈希 key 引起的 panic
	// @end
	m[k] = v
	return nil
}
--- test
package main

import "testing"

func TestTryInsert(t *testing.T) {
	m := map[any]int{}
	ok := []any{"a", 1, [2]int{1, 2}, struct{ A string }{"x"}, (*int)(nil)}
	for _, k := range ok {
		if err := TryInsert(m, k, 1); err != nil {
			t.Errorf("TryInsert(%#v) 不应出错：%v", k, err)
		}
	}
	bad := []any{[]int{1}, map[string]int{}, func() {}, struct{ S []int }{}}
	for _, k := range bad {
		if err := TryInsert(m, k, 1); err == nil {
			t.Errorf("TryInsert(%T) 应返回错误", k)
		}
	}

		t.Fatalf("map 中应有 %d 个元素，实际 %d", len(ok), len(m))
	}
}

=== go-14-2
title: 用 chan struct{} 广播信号
race: true
--- desc
实现一个只能打开一次的闸门 `Gate`：

- `Wait()` 返回一个 `<-chan struct{}`，闸门打开后所有等待者同时被唤醒；
- `Open()` 打开闸门，可以被多次、并发调用，但不能 panic（重复 close channel 会 panic）。

提示：`close` 一个 channel 会唤醒所有接收者；用 `sync.Once` 保证只关闭一次。
--- solution
package main

import "sync"

type Gate struct {
	ch   chan struct{}
	once sync.Once
}

func NewGate() *Gate { return &Gate{ch: make(chan struct{})} }

func (g *Gate) Wait() <-chan struct{} { return g.ch }

func (g *Gate) Open() {
	// @solution
	g.once.Do(func() { close(g.ch) })
	// @starter
	// // TODO: 关闭 channel，且只能关闭一次
	// @end
}
--- test
package main

import (
	"sync"
	"testing"
	"time"
)

func TestGateBroadcast(t *testing.T) {
	g := NewGate()
	var wg sync.WaitGroup
	released := make(chan int, 10)
	for i := 0; i < 10; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			<-g.Wait()
			released <- i
		}()
	}
	select {
	case <-released:
		t.Fatal("闸门打开之前不应有人通过")
	case <-time.After(20 * time.Millisecond):
	}
	var opens sync.WaitGroup
	for i := 0; i < 5; i++ {
		opens.Add(1)
		go func() { defer opens.Done(); g.Open() }()
	}
	opens.Wait()
	done := make(chan struct{})
	go func() { wg.Wait(); close(done) }()
	select {
	case <-done:
	case <-time.After(2 * time.Second):
		t.Fatal("Open 之后所有等待者都应被唤醒")
	}
}

=== go-15-1
title: 嵌入接口，只重写一个方法
--- desc
`LoggedStore` 嵌入了 `Store` 接口，`Get` 会被自动「提升」，可以直接使用。请为它**只重写** `Set` 方法：先往 `Logs` 里追加 `"set <key>"`，再调用内层 `Store` 的 `Set`。

这就是 Go 用组合代替继承的方式：外层类型复用内层的方法，只覆盖需要改变的部分。
--- solution
package main

type Store interface {
	Get(key string) (string, bool)
	Set(key, value string)
}

type MemStore struct{ m map[string]string }

func NewMemStore() *MemStore { return &MemStore{m: map[string]string{}} }

func (s *MemStore) Get(key string) (string, bool) { v, ok := s.m[key]; return v, ok }
func (s *MemStore) Set(key, value string)          { s.m[key] = value }

type LoggedStore struct {
	Store
	Logs []string
}

// @solution
func (l *LoggedStore) Set(key, value string) {
	l.Logs = append(l.Logs, "set "+key)
	l.Store.Set(key, value)
}

// @starter
// // TODO: 为 *LoggedStore 定义 Set 方法
// @end
--- test
package main

import (
	"slices"
	"testing"
)

func TestLoggedStore(t *testing.T) {
	ls := &LoggedStore{Store: NewMemStore()}
	var s Store = ls
	s.Set("a", "1")
	s.Set("b", "2")
	if v, ok := s.Get("a"); !ok || v != "1" {
		t.Fatalf("Get(a) = %q, %v，期望 1, true", v, ok)
	}
	if !slices.Equal(ls.Logs, []string{"set a", "set b"}) {
		t.Fatalf("Logs = %v，期望 [set a set b]", ls.Logs)
	}
}
--- explain
`l.Store.Set(...)` 显式调用内层对象的方法。注意这不是「重写」意义上的多态：如果 `MemStore` 内部的其他方法调用了 `Set`，调用的仍是 `MemStore.Set`，不会绕回 `LoggedStore.Set`。

title: 用 %w 保留错误链
--- desc
实现两层调用，让上层能用 `errors.Is` / `errors.As` 判断底层错误：

- `FindUser(id)`：用户不存在时返回 `fmt.Errorf("find user %d: %w", id, ErrNotFound)`；
- `LoadProfile(id)`：`FindUser` 出错时包装成 `&QueryError{Query: "load profile", Err: err}`；
- 给 `*QueryError` 实现 `Unwrap() error`，否则错误链在这一层就断了。

最终错误信息应该是 `load profile: find user 3: not found`。

package main

import (
	"errors"
	"fmt"
)

var ErrNotFound = errors.New("not found")

var users = map[int]string{1: "alice", 2: "bob"}

type QueryError struct {
	Query string
	Err   error
}

func (e *QueryError) Error() string { return e.Query + ": " + e.Err.Error() }

// @solution
func (e *QueryError) Unwrap() error { return e.Err }

// @starter
// // TODO: 实现 Unwrap
// @end

func FindUser(id int) (string, error) {
	name, ok := users[id]
	if !ok {
		// @solution
		return "", fmt.Errorf("find user %d: %w", id, ErrNotFound)
		// @starter
		// return "", fmt.Errorf("find user %d: %v", id, ErrNotFound) // TODO: %v 会丢掉错误链
		// @end
	}
	return name, nil
}

func LoadProfile(id int) (string, error) {
	name, err := FindUser(id)
	if err != nil {
		return "", &QueryError{Query: "load profile", Err: err}

=== go-16-2
title: 用 errors.Is / errors.As 给错误分类
--- desc
实现 `Classify(err)`，按以下规则返回分类（timeout 优先）：

| 条件 | 返回 |
| --- | --- |
| `err == nil` | `"ok"` |
| 错误链上有 `ErrTimeout` 或 `context.DeadlineExceeded` | `"timeout"` |
| 错误链上有 `*HTTPError`，Code 为 4xx | `"client"` |
| 错误链上有 `*HTTPError`，Code 为 5xx | `"server"` |
| 其他 | `"unknown"` |

测试里会出现 `fmt.Errorf` 多层包装、`errors.Join`、一个 `fmt.Errorf` 里多个 `%w` 的情况。
--- solution
package main

import (
	"context"
	"errors"
	"fmt"
)

var ErrTimeout = errors.New("timeout")

type HTTPError struct{ Code int }

func (e *HTTPError) Error() string { return fmt.Sprintf("http %d", e.Code) }

func Classify(err error) string {
	// @solution
	if err == nil {
		return "ok"
	}
	if errors.Is(err, ErrTimeout) || errors.Is(err, context.DeadlineExceeded) {
		return "timeout"
	}
	var he *HTTPError
	if errors.As(err, &he) {
		switch {
		case he.Code >= 400 && he.Code < 500:
			return "client"
		case he.Code >= 500:
			return "server"
		}
	}
	return "unknown"
	// @starter
	// return "unknown" // TODO
	// @end
}
--- test
package main

import (
	"context"
	"errors"
	"fmt"
	"testing"
)

func TestClassify(t *testing.T) {
	cases := []struct {
		name string
		err  error
		want string
	}{
		{"nil", nil, "ok"},
		{"timeout", ErrTimeout, "timeout"},
		{"wrapped timeout", fmt.Errorf("call api: %w", ErrTimeout), "timeout"},
		{"deadline", fmt.Errorf("db: %w", context.DeadlineExceeded), "timeout"},
		{"404", &HTTPError{404}, "client"},
		{"wrapped 503", fmt.Errorf("a: %w", fmt.Errorf("b: %w", &HTTPError{503})), "server"},
		{"join", errors.Join(errors.New("x"), &HTTPError{400}), "client"},
		{"timeout wins", errors.Join(&HTTPError{500}, ErrTimeout), "timeout"},
		{"multi %w", fmt.Errorf("%w; %w", errors.New("x"), &HTTPError{502}), "server"},
		{"plain", errors.New("boom"), "unknown"},
		{"3xx", &HTTPError{302}, "unknown"},
	}
	for _, c := range cases {
		if got := Classify(c.err); got != c.want {
			t.Errorf("%s: Classify = %q，期望 %q", c.name, got, c.want)
		}
	}
}

=== go-20-1
title: 实现一个会扩容的哈希表
--- desc
用「拉链法」实现一个字符串到 int 的哈希表，体会 map 底层的桶和扩容：

- 初始 8 个桶，用给定的 `hash` 函数（FNV-1a）对桶数取模定位；
- `Put` 新增或更新、`Get`、`Delete`、`Len`；
- **负载因子**（元素数 / 桶数）超过 6.5 时，桶数翻倍，并把所有元素重新分布到新桶里。

所以插入第 52 个元素后仍是 8 个桶（52/8 = 6.5），插入第 53 个时扩容到 16 个桶。
--- solution
package main

type entry struct {
	key   string
	value int
	next  *entry
}

type HashMap struct {
	buckets []*entry
	count   int
}

func NewHashMap() *HashMap { return &HashMap{buckets: make([]*entry, 8)} }

func hash(key string) uint32 {
	h := uint32(2166136261)
	for i := 0; i < len(key); i++ {
		h ^= uint32(key[i])
		h *= 16777619
	}
	return h
}

func (m *HashMap) BucketCount() int { return len(m.buckets) }

func (m *HashMap) index(key string, n int) int { return int(hash(key) % uint32(n)) }

func (m *HashMap) Put(key string, value int) {
	// @solution
	i := m.index(key, len(m.buckets))
	for e := m.buckets[i]; e != nil; e = e.next {
		if e.key == key {
			e.value = value
			return
		}
	}
	m.buckets[i] = &entry{key: key, value: value, next: m.buckets[i]}
	m.count++
	if float64(m.count)/float64(len(m.buckets)) > 6.5 {
		m.grow()
	}
	// @starter
	// // TODO: 定位桶 → 已存在则更新 → 否则头插新节点 → 检查负载因子
	// @end
}

// @solution
func (m *HashMap) grow() {
	nb := make([]*entry, len(m.buckets)*2)
	for _, head := range m.buckets {
		for e := head; e != nil; {
			next := e.next
			i := m.index(e.key, len(nb))
			e.next = nb[i]
			nb[i] = e
			e = next
		}
	}
	m.buckets = nb
}

// @end

func (m *HashMap) Get(key string) (int, bool) {
	// @solution
	for e := m.buckets[m.index(key, len(m.buckets))]; e != nil; e = e.next {
		if e.key == key {
			return e.value, true
		}
	}
	return 0, false
	// @starter
	// return 0, false // TODO
	// @end
}

func (m *HashMap) Delete(key string) bool {
	// @solution
	i := m.index(key, len(m.buckets))
	for p := &m.buckets[i]; *p != nil; p = &(*p).next {
		if (*p).key == key {
			*p = (*p).next
			m.count--
			return true
		}
	}
	return false
	// @starter
	// return false // TODO
	// @end
}

func (m *HashMap) Len() int { return m.count }

package main

import (
	"reflect"
	"strings"
)

func JSONFields(v any) []string {
	// @solution
	t := reflect.TypeOf(v)
	if t == nil {
		return nil
	}
	if t.Kind() == reflect.Pointer {
		t = t.Elem()
	}
	if t.Kind() != reflect.Struct {
		return nil
	}
	var names []string
	for i := 0; i < t.NumField(); i++ {
		f := t.Field(i)
		if !f.IsExported() {
			continue
		}
		name := f.Name
		if tag, ok := f.Tag.Lookup("json"); ok {
			tagName, _, _ := strings.Cut(tag, ",")
			if tagName == "-" {
				continue
			}
			if tagName != "" {
				name = tagName
			}
		}
		names = append(names, name)
	}
	return names
	// @starter
	// return nil // TODO: reflect.TypeOf → 遍历字段 → 读取 Tag.Get("json")
	// @end
}

package main

import (
	"errors"
	"fmt"
	"reflect"
)

func SetField(obj any, name string, value any) error {
	// @solution
	v := reflect.ValueOf(obj)
	if v.Kind() != reflect.Pointer || v.IsNil() || v.Elem().Kind() != reflect.Struct {
		return errors.New("obj 必须是非 nil 的结构体指针")
	}
	f := v.Elem().FieldByName(name)
	if !f.IsValid() {
		return fmt.Errorf("字段 %s 不存在", name)
	}
	if !f.CanSet() {
		return fmt.Errorf("字段 %s 不可设置", name)
	}
	val := reflect.ValueOf(value)
	if !val.IsValid() || !val.Type().AssignableTo(f.Type()) {
		return fmt.Errorf("类型不匹配：%T 不能赋值给 %s", value, f.Type())
	}
	f.Set(val)
	return nil
	// @starter
	// return errors.New("TODO")
	// @end
}

import "strings"

func Join(parts []string, sep string) string {
	// @solution
	if len(parts) == 0 {

		n += len(p)
	}

	b.Grow(n)
	b.WriteString(parts[0])

		b.WriteString(sep)
		b.WriteString(p)
	}
	return b.String()

title: 批量调用，摊薄跨边界开销
--- desc
cgo 调用每次都有固定开销，频繁调用小函数得不偿失，常见做法是**批量化**。这里用一个接口模拟这类「每次调用都很贵」的后端。

实现 `ProcessAll(b, items, batchSize)`：把 `items` 按每批最多 `batchSize` 个切分，逐批调用 `b.Call`，按原顺序拼接结果。`batchSize < 1` 时按 1 处理；`items` 为空时不调用。
--- solution
package main

type Backend interface {
	Call(batch []int) []int
}

func ProcessAll(b Backend, items []int, batchSize int) []int {
	// @solution
	if batchSize < 1 {
		batchSize = 1
	}
	out := make([]int, 0, len(items))
	for start := 0; start < len(items); start += batchSize {
		end := min(start+batchSize, len(items))
		out = append(out, b.Call(items[start:end])...)
	}
	return out
	// @starter
	// var out []int
	// for _, it := range items {
	// 	out = append(out, b.Call([]int{it})...) // TODO: 每个元素调用一次，开销太大
	// }
	// return out
	// @end
}
