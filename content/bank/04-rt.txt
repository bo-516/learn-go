=== rt-01-1
title: 模拟调度器的 findRunnable
--- desc
用简化模型实现调度器为 P 挑选下一个 G 的过程。G 用正整数编号，0 表示「没有」。

`FindRunnable(s, p)` 的规则：

1. 先把 `p.SchedTick` 加一；如果加一后 `SchedTick % 61 == 0` 且全局队列非空，**先从全局队列**取队首（防止全局队列饥饿）；
2. `p.RunNext` 非 0 时返回它并清零；
3. 本地队列非空时取队首；
4. 全局队列非空时取队首；
5. 按 `s.Ps` 的顺序找第一个本地队列非空的其他 P，从它的**队首偷走一半**（向上取整）：第一个偷来的 G 直接返回，其余追加到 `p` 的本地队列；
6. 都没有则返回 0。
--- solution
package main

type P struct {
	ID        int
	RunNext   int
	Local     []int
	SchedTick int
}

type Sched struct {
	Global []int
	Ps     []*P
}

func FindRunnable(s *Sched, p *P) int {
	// @solution
	p.SchedTick++
	if p.SchedTick%61 == 0 && len(s.Global) > 0 {
		g := s.Global[0]
		s.Global = s.Global[1:]
		return g
	}
	if p.RunNext != 0 {
		g := p.RunNext
		p.RunNext = 0
		return g
	}
	if len(p.Local) > 0 {
		g := p.Local[0]
		p.Local = p.Local[1:]
		return g
	}
	if len(s.Global) > 0 {
		g := s.Global[0]
		s.Global = s.Global[1:]
		return g
	}
	for _, victim := range s.Ps {
		if victim == p || len(victim.Local) == 0 {
			continue
		}
		n := len(victim.Local) - len(victim.Local)/2
		stolen := victim.Local[:n:n]
		victim.Local = victim.Local[n:]
		p.Local = append(p.Local, stolen[1:]...)
		return stolen[0]
	}
	return 0
	// @starter
	// p.SchedTick++
	// return 0 // TODO
	// @end
}
--- test

	"slices"
	"testing"

		t.Fatalf("第 61 次调度应先查全局队列，得到 %d", g)
	}
	if g := FindRunnable(s, p); g != 1 {
		t.Fatalf("之后回到本地队列，得到 %d", g)
	}

	idle := &P{ID: 0}

	busy := &P{ID: 2, Local: []int{1, 2, 3, 4, 5}}

=== rt-02-1
title: 模拟时间片抢占
--- desc
一个 P 上有多个任务，`Work` 表示需要的 CPU 时间（ms）。实现 `Schedule(tasks, quantum)` 模拟**抢占式**调度：

- 按队列顺序轮流执行，每个任务每次最多运行 `quantum` ms（Go 的抢占阈值是 10ms）；
- 没做完的任务放回队尾；
- 返回每个任务的完成时刻，按完成顺序排列。

把 `quantum` 设得非常大就退化成了协作式调度：一个长任务会一直霸占 P，短任务只能干等。
--- solution
package main

type Task struct {
	Name string
	Work int
}

type Finish struct {
	Name string
	At   int
}

func Schedule(tasks []Task, quantum int) []Finish {
	// @solution
	queue := append([]Task(nil), tasks...)
	var out []Finish
	now := 0
	for len(queue) > 0 {
		t := queue[0]
		queue = queue[1:]
		run := min(t.Work, quantum)
		now += run
		t.Work -= run
		if t.Work == 0 {
			out = append(out, Finish{t.Name, now})
		} else {
			queue = append(queue, t)
		}
	}
	return out
	// @starter
	// return nil // TODO
	// @end
}

=== rt-03-1
title: 根据 cgroup 限制计算 GOMAXPROCS
--- desc
容器里 Go 1.25 之前读取的是宿主机核数。实现 `DefaultProcs(cpuMax, numCPU, env)` 计算 GOMAXPROCS，规则参考 Go 1.25 的做法：

1. 环境变量 `GOMAXPROCS`（`env`）是合法的正整数时，直接使用它；
2. `cpuMax` 是 cgroup v2 `cpu.max` 文件的内容，格式为 `"<quota> <period>"`，`quota` 为 `max` 表示不限制，此时返回 `numCPU`；
3. 有限制时，`limit = ceil(quota / period)`，再取 `max(limit, 2)`（GOMAXPROCS=1 的性能断崖太大），最后不超过 `numCPU`；
4. 格式不合法返回错误。

| cpuMax | numCPU | 结果 |
| --- | --- | --- |
| `max 100000` | 64 | 64 |
| `200000 100000` | 64 | 2 |
| `150000 100000` | 64 | 2 |
| `450000 100000` | 64 | 5 |
| `50000 100000` | 64 | 2 |
| `400000 100000` | 2 | 2 |
--- solution
package main

import (
	"errors"
	"strconv"
	"strings"
)

func DefaultProcs(cpuMax string, numCPU int, env string) (int, error) {
	// @solution
	if n, err := strconv.Atoi(env); err == nil && n > 0 {
		return n, nil
	}
	fields := strings.Fields(cpuMax)
	if len(fields) != 2 {
		return 0, errors.New("invalid cpu.max")
	}
	period, err := strconv.Atoi(fields[1])
	if err != nil || period <= 0 {
		return 0, errors.New("invalid period")
	}
	if fields[0] == "max" {
		return numCPU, nil
	}
	quota, err := strconv.Atoi(fields[0])
	if err != nil || quota <= 0 {
		return 0, errors.New("invalid quota")
	}
	limit := (quota + period - 1) / period
	return min(max(limit, 2), numCPU), nil
	// @starter
	// return numCPU, nil // TODO
	// @end
}

=== rt-04-1
title: 模拟 netpoller：goroutine 挂在 fd 上等待
race: true
--- desc
网络 IO 没有就绪时，goroutine 不会阻塞线程，而是挂在 netpoller 上，fd 就绪后再被放回运行队列。实现一个简化的 `Poller`：

- `Wait(ctx, fd)`：把当前 goroutine 挂在 `fd` 上，直到 `Ready(fd)` 被调用（返回 nil），或者 `ctx` 被取消（返回 `ctx.Err()`，并把自己从等待列表中移除）；
- `Ready(fd)`：唤醒所有等待 `fd` 的 goroutine，返回唤醒的数量；
- `Waiting(fd)`：当前等待 `fd` 的 goroutine 数量。
--- solution
package main

import (
	"context"
	"slices"
	"sync"
)

type Poller struct {
	mu      sync.Mutex
	waiters map[int][]chan struct{}
}

func NewPoller() *Poller { return &Poller{waiters: map[int][]chan struct{}{}} }

func (p *Poller) Wait(ctx context.Context, fd int) error {
	// @solution
	ch := make(chan struct{})
	p.mu.Lock()
	p.waiters[fd] = append(p.waiters[fd], ch)
	p.mu.Unlock()
	select {
	case <-ch:
		return nil
	case <-ctx.Done():
		p.mu.Lock()
		defer p.mu.Unlock()
		ws := p.waiters[fd]
		if i := slices.Index(ws, ch); i >= 0 {
			p.waiters[fd] = slices.Delete(ws, i, i+1)
			return ctx.Err()
		}
		return nil // 取消的同时刚好被 Ready 唤醒
	}
	// @starter
	// return nil // TODO
	// @end
}

func (p *Poller) Ready(fd int) int {
	// @solution
	p.mu.Lock()
	ws := p.waiters[fd]
	delete(p.waiters, fd)
	p.mu.Unlock()
	for _, ch := range ws {
		close(ch)
	}
	return len(ws)
	// @starter
	// return 0 // TODO
	// @end
}

func (p *Poller) Waiting(fd int) int {
	p.mu.Lock()
	defer p.mu.Unlock()
	return len(p.waiters[fd])
}
--- test

title: 深递归与显式栈
--- desc
goroutine 的栈从几 KB 开始，不够时整体拷贝到 2 倍大小的新栈上，最大可以到 1GB。

- `SumTo(n)`：用**递归**计算 `1 + 2 + … + n`。测试会在一个新的 goroutine 里以 n = 1,000,000 调用它，栈会自动增长到几十 MB；
- `MaxDepth(root)`：用**显式栈**（切片）计算多叉树的最大深度（只有根节点时深度为 1，nil 为 0），不用递归。
--- solution
package main

type Node struct {
	Children []*Node
}

func SumTo(n int) int {
	// @solution
	if n == 0 {
		return 0
	}
	return n + SumTo(n-1)
	// @starter
	// return 0 // TODO: 用递归实现
	// @end
}

func MaxDepth(root *Node) int {
	// @solution
	if root == nil {
		return 0
	}
	type item struct {
		n     *Node
		depth int
	}
	best := 0
	stack := []item{{root, 1}}
	for len(stack) > 0 {
		it := stack[len(stack)-1]
		stack = stack[:len(stack)-1]
		best = max(best, it.depth)
		for _, c := range it.n.Children {
			stack = append(stack, item{c, it.depth + 1})
		}
	}
	return best
	// @starter
	// return 0 // TODO: 用切片模拟栈
	// @end
}

=== rt-06-1
title: 三色标记与清除
--- desc
用三色标记实现一次（非并发的）GC。`heap[x]` 是对象 `x` 引用的对象列表。

- `Mark(heap, roots)`：根对象先标灰放进工作队列；不断取出一个灰色对象，把它引用的**白色**对象标灰入队，然后把它自己标黑。返回所有黑色对象的集合；
- `Sweep(heap, marked)`：返回所有未被标记（白色）的对象编号，升序。

注意循环引用：不可达的环也必须被回收，可达的环不能导致死循环。
--- solution
package main

import "slices"

func Mark(heap map[int][]int, roots []int) map[int]bool {
	// @solution
	black := map[int]bool{}
	grey := map[int]bool{}
	var queue []int
	for _, r := range roots {
		if !grey[r] {
			grey[r] = true
			queue = append(queue, r)
		}
	}
	for len(queue) > 0 {
		x := queue[0]
		queue = queue[1:]
		for _, y := range heap[x] {
			if !grey[y] && !black[y] {
				grey[y] = true
				queue = append(queue, y)
			}
		}
		delete(grey, x)
		black[x] = true
	}
	return black
	// @starter
	// return nil // TODO
	// @end
}

func Sweep(heap map[int][]int, marked map[int]bool) []int {
	// @solution
	var garbage []int
	for x := range heap {
		if !marked[x] {
			garbage = append(garbage, x)
		}
	}
	slices.Sort(garbage)
	return garbage
	// @starter
	// return nil // TODO
	// @end
}

=== rt-06-2
title: 写屏障：防止并发标记漏标
--- desc
并发标记时用户代码还在修改指针。经典的漏标场景：

1. 对象 1（已黑）→ 2（灰）→ 3（白）；
2. 用户代码执行 `1.f = 3`（黑对象引用了白对象），然后删除 `2 → 3`；
3. 继续标记：2 变黑，但 3 再也不会被扫描到，最终被误回收，而它明明还被 1 引用着。

`Heap` 已经实现了增量标记 `Step`。请实现两个写屏障：

- `WritePointer(src, dst)`：添加引用 `src → dst`；**插入屏障**：如果标记正在进行且 dst 是白色，把它标灰；
- `DeletePointer(src, dst)`：删除引用 `src → dst`；**删除屏障**：如果标记正在进行且 dst 是白色，把它标灰。

Go 1.8 起用的混合写屏障就是这两种屏障的结合。
--- solution
package main

import "slices"

type Color int

const (
	White Color = iota
	Grey
	Black
)

type Heap struct {
	refs    map[int][]int
	color   map[int]Color
	grey    []int
	marking bool
}

func NewHeap(refs map[int][]int, roots []int) *Heap {
	h := &Heap{refs: refs, color: map[int]Color{}, marking: true}
	for _, r := range roots {
		h.shade(r)
	}
	return h
}

func (h *Heap) shade(x int) {
	if h.color[x] == White {
		h.color[x] = Grey
		h.grey = append(h.grey, x)
	}
}

// Step 最多处理 n 个灰色对象，返回标记是否已经完成。
func (h *Heap) Step(n int) bool {
	for ; n > 0 && len(h.grey) > 0; n-- {
		x := h.grey[0]
		h.grey = h.grey[1:]
		for _, y := range h.refs[x] {
			h.shade(y)
		}
		h.color[x] = Black
	}
	if len(h.grey) == 0 {
		h.marking = false
	}
	return !h.marking
}

func (h *Heap) WritePointer(src, dst int) {
	// @solution
	if h.marking {
		h.shade(dst)
	}
	// @starter
	// // TODO: 插入写屏障
	// @end
	h.refs[src] = append(h.refs[src], dst)
}

func (h *Heap) DeletePointer(src, dst int) {
	// @solution
	if h.marking {
		h.shade(dst)
	}
	// @starter
	// // TODO: 删除写屏障
	// @end
	if i := slices.Index(h.refs[src], dst); i >= 0 {
		h.refs[src] = slices.Delete(h.refs[src], i, i+1)
	}
}

// Garbage 返回标记完成后仍为白色的对象。
func (h *Heap) Garbage() []int {
	var out []int
	for x := range h.refs {
		if h.color[x] == White {
			out = append(out, x)
		}
	}
	slices.Sort(out)
	return out
}
--- test

)

	h := NewHeap(map[int][]int{1: {2}, 2: {3}, 3: {}, 4: {}}, []int{1})
	h.Step(1) // 1 变黑，2 变灰
	h.WritePointer(1, 3)
	h.DeletePointer(2, 3)
	for !h.Step(1) {
	}

title: 限制阻塞调用的并发数
race: true
--- desc
GOMAXPROCS 不限制总线程数：每个阻塞在文件 IO 或 cgo 里的调用都占着一个线程，并发多了线程会暴涨（默认上限 10000，超过直接崩溃）。

实现 `BlockingPool`：

- `NewBlockingPool(n)`：同时最多 `n` 个调用在执行；
- `Do(ctx, fn)`：拿到名额后执行 `fn` 并返回 nil；排队期间 `ctx` 被取消，则放弃等待、**不执行** `fn`，返回 `ctx.Err()`。
--- solution
package main

import "context"

type BlockingPool struct {
	sem chan struct{}
}

func NewBlockingPool(n int) *BlockingPool {
	return &BlockingPool{sem: make(chan struct{}, n)}
}

func (p *BlockingPool) Do(ctx context.Context, fn func()) error {
	// @solution
	select {
	case p.sem <- struct{}{}:
	case <-ctx.Done():
		return ctx.Err()
	}
	defer func() { <-p.sem }()
	fn()
	return nil
	// @starter
	// fn() // TODO: 没有任何限制
	// return nil
	// @end
}

title: 模拟 sysmon 的抢占检查与休眠间隔
--- desc
sysmon 是不需要 P 的监控线程。实现它的两项逻辑：

`Preemptable(gs, now, limit)`：返回正在运行（`Running`）且连续运行时间**超过** `limit`（Go 里是 10ms）的 G 的 ID，升序。

`NextDelay(prev, idle)`：计算下一次休眠时间。`idle` 是连续「什么也没做」的轮数：

- `idle == 0` → 20µs；
- `idle > 50` → 上次间隔翻倍；
- 其他情况保持上次间隔；
- 最长 10ms。
--- solution
package main

import (
	"slices"
	"time"
)

type G struct {
	ID           int
	Running      bool
	RunningSince time.Time
}

func Preemptable(gs []G, now time.Time, limit time.Duration) []int {
	// @solution
	var ids []int
	for _, g := range gs {
		if g.Running && now.Sub(g.RunningSince) > limit {
			ids = append(ids, g.ID)
		}
	}
	slices.Sort(ids)
	return ids
	// @starter
	// return nil // TODO
	// @end
}

func NextDelay(prev time.Duration, idle int) time.Duration {
	// @solution
	d := prev
	if idle == 0 {
		d = 20 * time.Microsecond
	} else if idle > 50 {
		d *= 2
	}
	return min(d, 10*time.Millisecond)
	// @starter
	// return prev // TODO
	// @end
}

		d = NextDelay(d, idle)
	}
	if d != 10*time.Millisecond {
		t.Fatalf("长时间空闲后间隔应增长到 10ms，实际 %v", d)
	}
}

=== rt-14-1
title: 用 container/heap 实现定时器队列
--- desc
Go 的每个 P 都用最小堆管理定时器。用 `container/heap` 实现一个支持停止和重置的定时器队列：

- `Add(id, when)`：添加定时器；
- `Stop(id) bool`：删除定时器（用 `heap.Remove`，需要记住每个元素在堆里的下标）；
- `Reset(id, when) bool`：修改触发时间（用 `heap.Fix`）；
- `PopExpired(now) []int`：按触发时间顺序弹出所有 `when <= now` 的定时器 ID；
- `Len()`。

关键：实现 `heap.Interface` 的 `Swap` 时同步更新元素的 `index` 字段。
--- solution
package main

import (
	"container/heap"
	"time"
)

type timer struct {
	id    int
	when  time.Time
	index int
}

type timerHeap []*timer

func (h timerHeap) Len() int           { return len(h) }
func (h timerHeap) Less(i, j int) bool { return h[i].when.Before(h[j].when) }

func (h timerHeap) Swap(i, j int) {
	// @solution
	h[i], h[j] = h[j], h[i]
	h[i].index = i
	h[j].index = j
	// @starter
	// h[i], h[j] = h[j], h[i] // TODO: 同步更新 index
	// @end
}

func (h *timerHeap) Push(x any) {
	t := x.(*timer)
	t.index = len(*h)
	*h = append(*h, t)
}

func (h *timerHeap) Pop() any {
	old := *h
	t := old[len(old)-1]
	old[len(old)-1] = nil
	*h = old[:len(old)-1]
	t.index = -1
	return t
}

type TimerQueue struct {
	h    timerHeap
	byID map[int]*timer
}

func NewTimerQueue() *TimerQueue { return &TimerQueue{byID: map[int]*timer{}} }

func (q *TimerQueue) Add(id int, when time.Time) {
	t := &timer{id: id, when: when}
	q.byID[id] = t
	heap.Push(&q.h, t)
}

func (q *TimerQueue) Stop(id int) bool {
	// @solution
	t, ok := q.byID[id]
	if !ok {
		return false
	}
	heap.Remove(&q.h, t.index)
	delete(q.byID, id)
	return true
	// @starter
	// return false // TODO
	// @end
}

func (q *TimerQueue) Reset(id int, when time.Time) bool {
	// @solution
	t, ok := q.byID[id]
	if !ok {
		return false
	}
	t.when = when
	heap.Fix(&q.h, t.index)
	return true
	// @starter
	// return false // TODO
	// @end
}

func (q *TimerQueue) PopExpired(now time.Time) []int {
	// @solution
	var ids []int
	for len(q.h) > 0 && !q.h[0].when.After(now) {
		t := heap.Pop(&q.h).(*timer)
		delete(q.byID, t.id)
		ids = append(ids, t.id)
	}
	return ids
	// @starter
	// return nil // TODO
	// @end
}

func (q *TimerQueue) Len() int { return len(q.h) }

=== rt-15-1
title: 模拟辅助标记的记账
--- desc
GC 标记期间，分配内存的 goroutine 要「还债」：每分配 1 字节需要完成 `ratio` 单位的扫描工作。用一个简化模型实现 `Assist(n, ratio, credit, bg, overAssist)`：

1. 本次债务 `debt = ceil(n * ratio)`；
2. 先用自己的 `credit` 抵扣；
3. 还不够，再从后台标记积累的全局信用 `bg` 中「偷」；
4. 仍然不够，就必须亲自做扫描工作，而且一次至少做 `overAssist` 单位（多做的部分存为自己的 credit，下次分配就不用再停下来）。

返回 `(work, credit, bg)`：本次亲自做的扫描量，以及更新后的个人信用和全局信用。
--- solution
package main

import "math"

func Assist(n int64, ratio float64, credit, bg, overAssist int64) (work, newCredit, newBg int64) {
	// @solution
	debt := int64(math.Ceil(float64(n) * ratio))
	use := min(debt, credit)
	debt -= use
	credit -= use
	steal := min(debt, bg)
	debt -= steal
	bg -= steal
	if debt > 0 {
		work = max(debt, overAssist)
		credit += work - debt
	}
	return work, credit, bg
	// @starter
	// return 0, credit, bg // TODO
	// @end
}

	}{
		{"自己的信用够用", 100, 0.5, 80, 0, 64, 0, 30, 0},
		{"借用后台信用", 100, 1, 30, 100, 64, 0, 0, 30},
		{"必须亲自扫描，多做的存起来", 100, 1, 10, 20, 64, 70, 0, 0},
		{"欠得多就做得多", 1000, 1, 0, 0, 64, 1000, 0, 0},

=== rt-16-1
title: 按 span 分组扫描（Green Tea 的思路）
--- desc
传统标记按发现对象的顺序扫描，内存访问在不同的页之间跳来跳去；Green Tea 以 span 为单位批量扫描。

- `SpanOrder(objs)`：把待扫描对象按 span 分组：span 按**第一次出现**的顺序排列，同一 span 内保持原来的顺序；
- `SpanSwitches(order)`：统计扫描顺序中相邻两个对象 span 不同的次数（近似「换页」的次数）。

分组后，换页次数等于不同 span 数减一。
--- solution
package main

type Obj struct {
	ID   int
	Span int
}

func SpanOrder(objs []Obj) []Obj {
	// @solution
	var spans []int
	groups := map[int][]Obj{}
	for _, o := range objs {
		if _, ok := groups[o.Span]; !ok {
			spans = append(spans, o.Span)
		}
		groups[o.Span] = append(groups[o.Span], o)
	}
	out := make([]Obj, 0, len(objs))
	for _, s := range spans {
		out = append(out, groups[s]...)
	}
	return out
	// @starter
	// return objs // TODO
	// @end
}

func SpanSwitches(order []Obj) int {
	// @solution
	n := 0
	for i := 1; i < len(order); i++ {
		if order[i].Span != order[i-1].Span {
			n++
		}
	}
	return n
	// @starter
	// return 0 // TODO
	// @end
}
--- test

import (
	"slices"
	"testing"
)

func TestSpanOrder(t *testing.T) {
	objs := []Obj{{1, 7}, {2, 3}, {3, 7}, {4, 9}, {5, 3}, {6, 7}}
	got := SpanOrder(objs)
	want := []Obj{{1, 7}, {3, 7}, {6, 7}, {2, 3}, {5, 3}, {4, 9}}
	if !slices.Equal(got, want) {
		t.Fatalf("SpanOrder = %v\n期望 %v", got, want)
	}
	if SpanSwitches(objs) != 5 || SpanSwitches(got) != 2 {
		t.Fatalf("换页次数：原顺序 %d（期望 5），分组后 %d（期望 2）", SpanSwitches(objs), SpanSwitches(got))

	}
}

title: 显式 Close + AddCleanup 兜底
--- desc
文件、连接这类资源必须显式关闭，`runtime.AddCleanup`（Go 1.24+）只能用来兜底和发现泄漏。实现：

- `Open(fd)`：创建资源，并用 `runtime.AddCleanup` 注册清理函数：资源被 GC 回收时如果还没关闭，就把全局计数 `leaked` 加一；

注意：清理函数**不能引用资源对象本身**（否则对象永远不可达），所以要把 fd 作为单独的参数传给它。
--- solution
package main

import (
	"errors"
	"runtime"
	"sync/atomic"
)

var ErrClosed = errors.New("already closed")

var leaked atomic.Int32

type Resource struct {
	fd      int
	closed  bool
	cleanup runtime.Cleanup
	name    *string
}

func Open(fd int) *Resource {
	name := "res"
	r := &Resource{fd: fd, name: &name}
	// @solution
	r.cleanup = runtime.AddCleanup(r, func(fd int) { leaked.Add(1) }, fd)
	// @starter
	// // TODO: 注册兜底清理函数
	// @end
	return r
}

func (r *Resource) Close() error {
	// @solution
	if r.closed {
		return ErrClosed
	}
	r.closed = true
	r.cleanup.Stop()
	return nil
	// @starter
	// r.closed = true // TODO: 重复关闭要报错，并取消兜底清理
	// return nil
	// @end
}

=== rt-18-1
title: 解析 schedtrace
--- desc
解析 `GODEBUG=schedtrace=1000` 的一行输出：

```text
SCHED 2003ms: gomaxprocs=8 idleprocs=0 threads=25 spinningthreads=0 needspinning=0 idlethreads=10 runqueue=127 [34 51 12 40 66 9 28 80]
```

- `ParseSchedTrace(line)`：解析出时间（ms）、各个 `key=value` 字段和方括号里每个 P 的本地队列长度。不同 Go 版本的字段可能不同（比如 `needspinning` 是新加的），用通用的 `key=value` 解析，不要依赖字段顺序；
- `Runnable()`：全局队列 + 所有本地队列的长度之和；
- `Diagnose()`：按顺序返回问题：`idleprocs == 0` 且 `Runnable() > 2 * gomaxprocs` → `"cpu-saturated"`；`threads > 4 * gomaxprocs` → `"too-many-threads"`。
--- solution
package main

import (
	"errors"
	"strconv"
	"strings"
)

type SchedTrace struct {
	AtMS   int
	Fields map[string]int
	Local  []int
}

func ParseSchedTrace(line string) (SchedTrace, error) {
	st := SchedTrace{Fields: map[string]int{}}
	// @solution
	head, rest, ok := strings.Cut(line, ": ")
	if !ok || !strings.HasPrefix(head, "SCHED ") {
		return st, errors.New("not a schedtrace line")
	}
	ms, err := strconv.Atoi(strings.TrimSuffix(strings.TrimPrefix(head, "SCHED "), "ms"))
	if err != nil {
		return st, err
	}
	st.AtMS = ms
	kv, local, _ := strings.Cut(rest, "[")
	for _, f := range strings.Fields(kv) {
		k, v, ok := strings.Cut(f, "=")
		if !ok {
			continue
		}
		n, err := strconv.Atoi(v)
		if err != nil {
			return st, err
		}
		st.Fields[k] = n
	}
	for _, f := range strings.Fields(strings.TrimSuffix(strings.TrimSpace(local), "]")) {
		n, err := strconv.Atoi(f)
		if err != nil {
			return st, err
		}
		st.Local = append(st.Local, n)
	}
	return st, nil
	// @starter
	// return st, errors.New("TODO")
	// @end
}

func (s SchedTrace) Runnable() int {
	// @solution
	n := s.Fields["runqueue"]
	for _, l := range s.Local {
		n += l
	}
	return n
	// @starter
	// return 0 // TODO
	// @end
}

func (s SchedTrace) Diagnose() []string {
	var out []string
	// @solution
	procs := s.Fields["gomaxprocs"]
	if s.Fields["idleprocs"] == 0 && s.Runnable() > 2*procs {
		out = append(out, "cpu-saturated")
	}
	if s.Fields["threads"] > 4*procs {
		out = append(out, "too-many-threads")
	}
	// @starter
	// // TODO
	// @end
	return out
}
