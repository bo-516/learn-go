=== code-01-1
title: 修复带超时查询的 goroutine 泄漏
--- desc
下面是原题里的代码（为了便于测试，`slowCall` 和超时时间改成了参数）。它有两个问题：

1. 超时返回后没人再接收 `ch`，执行 `slowCall` 的 goroutine 永远阻塞在发送上——**泄漏**；
2. 没有使用传入的 `ctx`，上游取消时不会停止，`slowCall` 也拿不到截止时间。

修复它：基于 `ctx` 派生超时 context 并传给 `slowCall`，channel 带 1 个缓冲，超时或取消时返回 `ctx.Err()`。
--- solution
package main

import (
	"context"
	"time"
)

func Query(ctx context.Context, slowCall func(context.Context) string, timeout time.Duration) (string, error) {
	ctx, cancel := context.WithTimeout(ctx, timeout)
	defer cancel()
	ch := make(chan string, 1)
	go func() {
		ch <- slowCall(ctx)
	}()
	select {
	case r := <-ch:
		return r, nil
	case <-ctx.Done():
		return "", ctx.Err()
	}
}

=== code-02-1
title: 修复值接收者导致的锁失效
race: true
--- desc
原题的计数器用了**值接收者**：每次调用 `Inc` 时 `c` 是一个副本，加锁的是副本里的 Mutex，goroutine 之间根本没有互斥；而 map 是引用类型，所有副本指向同一个哈希表，于是并发写 map 直接崩溃。

修复它（`go vet` 的 copylocks 检查可以发现这类问题）。
--- solution
package main

import "sync"

type Counter struct {
	mu sync.Mutex
	n  map[string]int
}

func NewCounter() *Counter { return &Counter{n: map[string]int{}} }

func (c *Counter) Inc(key string) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.n[key]++
}

func (c *Counter) Get(key string) int {
	c.mu.Lock()
	defer c.mu.Unlock()
	return c.n[key]
}

=== code-03-1
title: 修复循环里的 defer
--- desc
原题的 `ProcessFiles` 在循环里 `defer f.Close()`：defer 要等**函数返回**时才执行，处理上万个文件时所有文件都一直开着，最终 `too many open files`。

修复它：把每次循环的内容放进一个单独的函数（或闭包），让文件在每轮结束时就关闭。测试用一个假的 `open` 统计同时打开的文件数，必须始终不超过 1 个。
--- solution
package main

import "io"

type File interface {
	io.Reader
	Close() error
}

func ProcessFiles(open func(path string) (File, error), paths []string, process func(File) error) error {
	for _, p := range paths {
		if err := func() error {
			f, err := open(p)
			if err != nil {
				return err
			}
			defer f.Close()
			return process(f)
		}(); err != nil {
			return err
		}
	}
	return nil
}

=== code-04-1
title: 修复并发抓取 URL
race: true
--- desc
原题代码有三个问题：`wg.Add(1)` 写在 goroutine 里；多个 goroutine 并发 `append` 同一个切片（数据竞争）；忽略错误且不限制并发数。

重写 `FetchAll(ctx, urls, limit, fetch)`：

- 结果按 `urls` 的顺序返回（预分配切片，每个 goroutine 写自己的下标）；
- 同时最多 `limit` 个请求；
- 任何一个请求出错时，返回 `fetch <url>: <原始错误>`（用 `%w`），并取消 ctx 让其他请求尽快停止。

标准库里没有 errgroup，可以用 WaitGroup + 信号量 + `sync.Once` 记录第一个错误自己实现。
--- solution
package main

import (
	"context"
	"fmt"
	"sync"
)

func FetchAll(ctx context.Context, urls []string, limit int,
	fetch func(ctx context.Context, url string) (string, error)) ([]string, error) {
	ctx, cancel := context.WithCancel(ctx)
	defer cancel()
	results := make([]string, len(urls))
	sem := make(chan struct{}, limit)
	var (
		wg       sync.WaitGroup
		once     sync.Once
		firstErr error
	)
	for i, u := range urls {
		wg.Add(1)
		go func() {
			defer wg.Done()
			select {
			case sem <- struct{}{}:
			case <-ctx.Done():
				return
			}
			defer func() { <-sem }()
			body, err := fetch(ctx, u)
			if err != nil {
				once.Do(func() {
					firstErr = fmt.Errorf("fetch %s: %w", u, err)
					cancel()
				})
				return
			}
			results[i] = body
		}()
	}
	wg.Wait()
	if firstErr != nil {
		return nil, firstErr
	}
	return results, nil
}

func FetchAll(ctx context.Context, urls []string, limit int,

title: 两个 goroutine 交替打印奇偶数
--- desc
实现 `Alternate(n, odd, even)`：启动**两个** goroutine，一个只负责奇数、一个只负责偶数，按 `1, 2, 3, …, n` 的顺序交替调用 `odd(i)` / `even(i)`，全部完成后返回。

测试会检查：所有 `odd` 调用来自同一个 goroutine，所有 `even` 调用来自另一个 goroutine，并且顺序严格交替。

提示：用两个无缓冲 channel 互相「传球」。
--- solution
package main

import "sync"

func Alternate(n int, odd, even func(int)) {
	// @solution
	oddTurn := make(chan struct{})
	evenTurn := make(chan struct{})
	var wg sync.WaitGroup
	wg.Add(2)
	go func() {
		defer wg.Done()
		for i := 1; i <= n; i += 2 {
			<-oddTurn
			odd(i)
			if i+1 <= n {
				evenTurn <- struct{}{}
			}
		}
	}()
	go func() {
		defer wg.Done()
		for i := 2; i <= n; i += 2 {
			<-evenTurn
			even(i)
			if i+1 <= n {
				oddTurn <- struct{}{}
			}
		}
	}()
	if n >= 1 {
		oddTurn <- struct{}{}
	}
	wg.Wait()

title: 三个 goroutine 轮流打印 ABC
--- desc
扩展上一题：启动三个 goroutine，分别只打印 `A`、`B`、`C`，按 `ABCABC…` 的顺序轮流调用 `print`，共 `rounds` 轮。

思路：准备三个 channel，第 i 个 goroutine 等待自己的 channel，打印后通知下一个。注意最后一轮结束时不要向没人接收的 channel 发送，否则会死锁或泄漏。
--- solution
package main

import "sync"

func ABC(rounds int, print func(string)) {
	// @solution
	letters := []string{"A", "B", "C"}
	turns := make([]chan struct{}, 3)
	for i := range turns {
		turns[i] = make(chan struct{})
	}
	var wg sync.WaitGroup
	for i, l := range letters {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for r := 0; r < rounds; r++ {
				<-turns[i]
				print(l)
				if r == rounds-1 && i == 2 {
					return
				}
				turns[(i+1)%3] <- struct{}{}
			}
		}()
	}
	if rounds > 0 {
		turns[0] <- struct{}{}
	}
	wg.Wait()

	// @end
}

=== code-06-1
title: 带优雅关闭的 worker 池
race: true
--- desc
实现 worker 池 `Pool`：

- `NewPool(workers, queue)`：启动 `workers` 个 worker，任务队列容量为 `queue`；
- `Submit(task)`：提交任务，队列满时阻塞；池已关闭返回 `ErrClosed`（不能 panic）；
- `Shutdown(ctx)`：停止接收新任务，等待**已提交的任务全部执行完**；`ctx` 先到期则返回 `ctx.Err()`。

难点：`Submit` 和 `Shutdown` 并发执行时，不能向已关闭的 channel 发送。可以用读写锁：`Submit` 持读锁发送，`Shutdown` 持写锁设置关闭标记并关闭 channel。
--- solution
package main

import (
	"context"
	"errors"
	"sync"
)

var ErrClosed = errors.New("pool closed")

type Pool struct {
	mu     sync.RWMutex
	closed bool
	tasks  chan func()
	wg     sync.WaitGroup
}

func NewPool(workers, queue int) *Pool {
	p := &Pool{tasks: make(chan func(), queue)}
	for i := 0; i < workers; i++ {
		p.wg.Add(1)
		go func() {
			defer p.wg.Done()
			for task := range p.tasks {
				task()
			}
		}()
	}
	return p
}

func (p *Pool) Submit(task func()) error {
	// @solution
	p.mu.RLock()
	defer p.mu.RUnlock()
	if p.closed {
		return ErrClosed
	}
	p.tasks <- task
	return nil
	// @starter
	// p.tasks <- task // TODO: 关闭后再提交会 panic
	// return nil
	// @end
}

func (p *Pool) Shutdown(ctx context.Context) error {
	// @solution
	p.mu.Lock()
	if !p.closed {
		p.closed = true
		close(p.tasks)
	}
	p.mu.Unlock()
	done := make(chan struct{})
	go func() {
		p.wg.Wait()
		close(done)
	}()
	select {
	case <-done:
		return nil
	case <-ctx.Done():
		return ctx.Err()
	}
	// @starter
	// return nil // TODO
	// @end
}
--- test
package main

import (
	"context"
	"errors"
	"sync/atomic"
	"testing"
	"time"
)

func TestPool(t *testing.T) {
	p := NewPool(3, 10)
	var done, running, peak atomic.Int32
	for i := 0; i < 30; i++ {
		err := p.Submit(func() {
			n := running.Add(1)
			for {
				m := peak.Load()
				if n <= m || peak.CompareAndSwap(m, n) {
					break
				}
			}
			time.Sleep(2 * time.Millisecond)
			running.Add(-1)
			done.Add(1)
		})
		if err != nil {
			t.Fatal(err)
		}
	}
	if err := p.Shutdown(context.Background()); err != nil {
		t.Fatal(err)
	}
	if done.Load() != 30 {
		t.Fatalf("Shutdown 返回时只完成了 %d 个任务", done.Load())
	}
	if peak.Load() > 3 {
		t.Fatalf("并发峰值 %d 超过 worker 数", peak.Load())
	}
	if err := p.Submit(func() {}); !errors.Is(err, ErrClosed) {
		t.Fatalf("关闭后提交应返回 ErrClosed，实际 %v", err)
	}
	if err := p.Shutdown(context.Background()); err != nil {
		t.Fatal("重复 Shutdown 不应出错")
	}
}

--- desc
实现 `CallAll(ctx, timeout, calls)`：

- 并发执行所有调用，整体超时 `timeout`；
- 全部成功：按顺序返回结果；
- 任何一个失败：**立即**返回这个错误，并取消其余调用（它们收到 ctx 取消后会退出）；

package main

import (
	"context"
	"time"
)

func CallAll(ctx context.Context, timeout time.Duration, calls []func(ctx context.Context) (string, error)) ([]string, error) {
	// @solution
	ctx, cancel := context.WithTimeout(ctx, timeout)
	defer cancel()
	type result struct {
		i   int
		v   string
		err error
	}
	ch := make(chan result, len(calls))
	for i, call := range calls {
		go func() {
			v, err := call(ctx)
			ch <- result{i, v, err}
		}()
	}
	out := make([]string, len(calls))
	for range calls {
		select {
		case r := <-ch:
			if r.err != nil {
				return nil, r.err
			}
			out[r.i] = r.v
		case <-ctx.Done():
			return nil, ctx.Err()
		}
	}
	return out, nil

--- desc
实现令牌桶 `TokenBucket`：

- 以每秒 `rate` 个的速度生成令牌，桶里最多 `burst` 个；初始时桶是满的；
- `AllowN(n)`：先根据距离上次的时间补充令牌（不超过 burst），令牌够 n 个就扣减并返回 true，否则返回 false（不扣减）；
- `Allow()` 等价于 `AllowN(1)`；
- 并发安全，时间通过注入的 `now` 获取。

和漏桶相比，令牌桶允许一定程度的突发（最多 burst 个），是最常用的限流算法（`golang.org/x/time/rate`）。
--- solution
package main

import (
	"sync"
	"time"
)

type TokenBucket struct {
	mu     sync.Mutex
	rate   float64
	burst  float64
	tokens float64
	last   time.Time
	now    func() time.Time
}

func NewTokenBucket(rate float64, burst int, now func() time.Time) *TokenBucket {
	return &TokenBucket{rate: rate, burst: float64(burst), tokens: float64(burst), last: now(), now: now}
}

func (b *TokenBucket) AllowN(n int) bool {
	// @solution
	b.mu.Lock()
	defer b.mu.Unlock()
	now := b.now()
	b.tokens = min(b.burst, b.tokens+now.Sub(b.last).Seconds()*b.rate)
	b.last = now
	if b.tokens < float64(n) {
		return false
	}
	b.tokens -= float64(n)
	return true
	// @starter
	// return true // TODO
	// @end
}

=== code-09-1
title: 滑动窗口限流器
--- desc
按原题的「滑动窗口计数器」实现单机版限流：只保留当前窗口和上一个窗口的计数，用上一个窗口的计数乘以**重叠比例**近似滑动窗口内的请求数：

1. 当前时刻距离当前窗口起点超过一个窗口时，切换窗口：超过两个窗口（中间空了一个完整窗口）`prev = 0`，否则 `prev = cur`；`cur = 0`，新的窗口起点为 `now.Truncate(window)`；
2. `weight = 1 − (now − 当前窗口起点) / window`；
3. `prev × weight + cur >= limit` 时拒绝，否则 `cur++` 并放行。

时间通过注入的 `now` 获取。Redis 版本的实现（ZSET 滑动日志 + Lua）见原题答案。
--- solution
package main

import (
	"sync"
	"time"
)

type SlidingWindow struct {
	mu        sync.Mutex
	limit     int
	window    time.Duration
	now       func() time.Time
	curStart  time.Time
	cur, prev int
}

func NewSlidingWindow(limit int, window time.Duration, now func() time.Time) *SlidingWindow {
	return &SlidingWindow{limit: limit, window: window, now: now}
}

func (s *SlidingWindow) Allow() bool {
	s.mu.Lock()
	defer s.mu.Unlock()
	// @solution
	now := s.now()
	if elapsed := now.Sub(s.curStart); elapsed >= s.window {
		if elapsed >= 2*s.window {
			s.prev = 0
		} else {
			s.prev = s.cur
		}
		s.cur = 0
		s.curStart = now.Truncate(s.window)
	}
	weight := 1 - float64(now.Sub(s.curStart))/float64(s.window)
	if float64(s.prev)*weight+float64(s.cur) >= float64(s.limit) {
		return false
	}
	s.cur++
	return true
	// @starter
	// return true // TODO
	// @end
}

实现简化版的 `singleflight.Group`：

- `Do(key, fn)`：同一时刻对同一个 key 的多个调用只执行一次 `fn`，其他调用者等待并共享结果。返回 `(v, err, shared)`，`shared` 表示这个调用者**没有亲自执行** fn，而是拿到了别人的结果；

package main

import "sync"

type call struct {
	wg  sync.WaitGroup
	val any
	err error
}

type Group struct {
	mu sync.Mutex
	m  map[string]*call
}

func (g *Group) Do(key string, fn func() (any, error)) (v any, err error, shared bool) {
	// @solution
	g.mu.Lock()
	if g.m == nil {
		g.m = map[string]*call{}
	}
	if c, ok := g.m[key]; ok {
		g.mu.Unlock()
		c.wg.Wait()
		return c.val, c.err, true
	}
	c := &call{}
	c.wg.Add(1)
	g.m[key] = c
	g.mu.Unlock()

	c.val, c.err = fn()
	c.wg.Done()

	g.mu.Lock()
	if g.m[key] == c { // 可能已经被 Forget，并被新的调用替换
		delete(g.m, key)
	}
	g.mu.Unlock()
	return c.val, c.err, false
	// @starter
	// v, err = fn() // TODO
	// return v, err, false
	// @end
}

--- desc
实现进程内的发布订阅 `PubSub`：

- `Subscribe(topic, buf)`：返回一个缓冲大小为 `buf` 的只读 channel 和一个取消函数；取消函数会退订并关闭 channel，可以重复调用；
- `Publish(topic, msg)`：**非阻塞**地投递给该主题的所有订阅者，某个订阅者缓冲满了就丢弃给它的这条消息（不能拖慢发布者），返回成功投递的数量；
- `Close()`：关闭所有订阅者的 channel；之后 `Publish` 返回 0，`Subscribe` 返回一个已关闭的 channel。

注意并发安全，以及「关闭后不能再向 channel 发送」。
--- solution
package main

import "sync"

type PubSub struct {
	mu     sync.RWMutex
	subs   map[string]map[int]chan string
	nextID int
	closed bool
}

func NewPubSub() *PubSub { return &PubSub{subs: map[string]map[int]chan string{}} }

func (ps *PubSub) Subscribe(topic string, buf int) (<-chan string, func()) {
	// @solution
	ps.mu.Lock()
	defer ps.mu.Unlock()
	ch := make(chan string, buf)
	if ps.closed {
		close(ch)
		return ch, func() {}
	}
	id := ps.nextID
	ps.nextID++
	if ps.subs[topic] == nil {
		ps.subs[topic] = map[int]chan string{}
	}
	ps.subs[topic][id] = ch
	cancel := func() {
		ps.mu.Lock()
		defer ps.mu.Unlock()
		if c, ok := ps.subs[topic][id]; ok {
			delete(ps.subs[topic], id)
			close(c)
		}
	}
	return ch, cancel
	// @starter
	// ch := make(chan string, buf) // TODO
	// return ch, func() {}
	// @end
}

func (ps *PubSub) Publish(topic, msg string) int {
	// @solution
	ps.mu.RLock()
	defer ps.mu.RUnlock()
	n := 0
	for _, ch := range ps.subs[topic] {
		select {
		case ch <- msg:
			n++
		default:
		}
	}
	return n
	// @starter
	// return 0 // TODO
	// @end
}

func (ps *PubSub) Close() {
	// @solution
	ps.mu.Lock()
	defer ps.mu.Unlock()
	if ps.closed {
		return
	}
	ps.closed = true
	for _, subs := range ps.subs {
		for _, ch := range subs {
			close(ch)
		}
	}
	ps.subs = map[string]map[int]chan string{}
	// @starter
	// // TODO
	// @end
}

title: 多生产者多消费者，收到退出信号时优雅停止

--- desc
实现 `Run(ctx, producers, consumers, produce, consume)`：

- 生产者循环调用 `produce(ctx)` 获取任务，返回 `false` 表示没有更多任务；发送任务时要同时监听 `ctx.Done()`，收到退出信号就停止；
- 所有生产者退出后，由**单独的协调者**关闭任务 channel（只关闭一次）；
- 消费者用 `for range` 消费，把**已经入队的任务处理完**再退出；
- `Run` 等所有消费者退出后返回。

package main

import (
	"context"
	"sync"
)

func Run(ctx context.Context, producers, consumers int,
	produce func(ctx context.Context) (int, bool), consume func(int)) {
	jobs := make(chan int, 16)
	// @solution
	var prod, cons sync.WaitGroup
	for p := 0; p < producers; p++ {
		prod.Add(1)
		go func() {
			defer prod.Done()
			for {
				job, ok := produce(ctx)
				if !ok {
					return
				}
				select {
				case jobs <- job:
				case <-ctx.Done():
					return
				}
			}
		}()
	}
	go func() {
		prod.Wait()
		close(jobs)
	}()
	for c := 0; c < consumers; c++ {
		cons.Add(1)
		go func() {
			defer cons.Done()
			for job := range jobs {
				consume(job)
			}
		}()
	}
	cons.Wait()
	// @starter
	// // TODO
	// _ = jobs
	// @end
}

--- desc
用 `container/list`（双向链表）+ map 实现 LRU 缓存：

- `Get(key)`：命中时把元素移到链表头部（最近使用）；
- `Put(key, val)`：已存在则更新并移到头部；不存在则插入头部，超过容量时淘汰**尾部**（最久未使用）的元素；

- 所有方法并发安全。注意 `Get` 也会修改链表顺序，所以要用写锁（`sync.Mutex`），不能用读锁。

package main

import (
	"container/list"
	"sync"
)

type entry struct {
	key string
	val int
}

type LRU struct {
	mu    sync.Mutex
	cap   int
	ll    *list.List
	items map[string]*list.Element
}

func NewLRU(capacity int) *LRU {
	return &LRU{cap: capacity, ll: list.New(), items: map[string]*list.Element{}}
}

func (c *LRU) Get(key string) (int, bool) {
	// @solution
	c.mu.Lock()
	defer c.mu.Unlock()
	if e, ok := c.items[key]; ok {
		c.ll.MoveToFront(e)
		return e.Value.(*entry).val, true
	}
	return 0, false
	// @starter
	// return 0, false // TODO
	// @end
}

func (c *LRU) Put(key string, val int) {
	// @solution
	c.mu.Lock()
	defer c.mu.Unlock()
	if e, ok := c.items[key]; ok {
		e.Value.(*entry).val = val
		c.ll.MoveToFront(e)
		return
	}
	c.items[key] = c.ll.PushFront(&entry{key, val})
	if c.ll.Len() > c.cap {
		oldest := c.ll.Back()
		c.ll.Remove(oldest)
		delete(c.items, oldest.Value.(*entry).key)
	}
	// @starter
	// // TODO
	// @end
}

func (c *LRU) Len() int {
	c.mu.Lock()
	defer c.mu.Unlock()
	return c.ll.Len()
}

title: 指数退避 + 随机抖动的重试
--- desc
实现 `Retry(ctx, attempts, base, maxDelay, randN, sleep, fn)`：

1. 最多执行 `attempts` 次 `fn`，成功立即返回 nil；
2. 只重试可重试的错误：错误链上有 `ErrUnavailable`，或者实现了 `Timeout() bool` 且返回 true（如网络超时）；其他错误直接返回；
3. 每次失败后等待 `backoff/2 + randN(backoff/2 + 1)`（Equal Jitter），`backoff` 从 `base` 开始每次翻倍，不超过 `maxDelay`；最后一次失败后不再等待；
4. 等待通过注入的 `sleep(ctx, d)` 完成（测试会记录等待时长）；它返回错误（ctx 被取消）时，返回 `errors.Join(最后一次的错误, ctx 错误)`。

`TimerSleep` 已经给出：它用 `select` 同时等待 Timer 和 `ctx.Done()`——不要用 `time.Sleep`，否则无法及时响应取消。
--- solution
package main

import (
	"context"
	"errors"
	"time"
)

var ErrUnavailable = errors.New("service unavailable")

func TimerSleep(ctx context.Context, d time.Duration) error {
	t := time.NewTimer(d)
	defer t.Stop()
	select {
	case <-t.C:
		return nil
	case <-ctx.Done():
		return ctx.Err()
	}
}

func retryable(err error) bool {
	var te interface{ Timeout() bool }
	if errors.As(err, &te) && te.Timeout() {
		return true
	}
	return errors.Is(err, ErrUnavailable)
}

func Retry(ctx context.Context, attempts int, base, maxDelay time.Duration,
	randN func(n int64) int64,
	sleep func(context.Context, time.Duration) error,
	fn func(context.Context) error) error {
	// @solution
	var err error
	backoff := base
	for i := 0; i < attempts; i++ {
		if err = fn(ctx); err == nil {
			return nil
		}
		if !retryable(err) || i == attempts-1 {
			return err
		}
		d := backoff/2 + time.Duration(randN(int64(backoff/2)+1))
		if serr := sleep(ctx, d); serr != nil {
			return errors.Join(err, serr)
		}
		backoff = min(backoff*2, maxDelay)
	}
	return err
	// @starter
	// var err error
	// for i := 0; i < attempts; i++ {
	// 	if err = fn(ctx); err == nil {
	// 		return nil
	// 	}
	// 	sleep(ctx, base) // TODO: 固定间隔、没有抖动、不区分错误类型
	// }
	// return err
	// @end
}

package main

import (
	"net/http"
	"time"
)

type Entry struct {
	Method   string
	Path     string
	Status   int
	Duration time.Duration
	Panic    bool
}

type recorder struct {
	http.ResponseWriter
	status      int
	wroteHeader bool
}

func (r *recorder) WriteHeader(code int) {
	if !r.wroteHeader {
		r.status, r.wroteHeader = code, true
	}
	r.ResponseWriter.WriteHeader(code)
}

func (r *recorder) Write(b []byte) (int, error) {
	if !r.wroteHeader {
		r.WriteHeader(http.StatusOK)
	}
	return r.ResponseWriter.Write(b)
}

func Logging(log func(Entry), now func() time.Time, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		// @solution
		start := now()
		rec := &recorder{ResponseWriter: w, status: http.StatusOK}
		e := Entry{Method: r.Method, Path: r.URL.Path}
		defer func() {
			if p := recover(); p != nil {
				e.Panic = true
				if !rec.wroteHeader {
					http.Error(rec, "internal server error", http.StatusInternalServerError)
				}
				rec.status = http.StatusInternalServerError
			}
			e.Status = rec.status
			e.Duration = now().Sub(start)
			log(e)
		}()
		next.ServeHTTP(rec, r)
		// @starter
		// start := now()

按原题实现一个简化版熔断器 `Breaker.Do(fn)`：

- **关闭**：正常调用，统计连续失败次数，达到 `threshold` 时打开；成功则清零；
- **打开**：直接返回 `ErrOpen`，不调用 fn；经过 `coolDown` 后进入半开；
- **半开**：只放行**一个**探测请求（其他请求返回 `ErrOpen`）；探测成功则关闭，失败则重新打开并重新计时。

时间通过注入的 `now` 获取。并发安全：判断是否放行（allow）和记录结果（record）分别加锁，调用 fn 时不持有锁。
--- solution
package main

import (
	"errors"
	"sync"
	"time"
)

type State int

const (
	Closed State = iota
	Open
	HalfOpen
)

var ErrOpen = errors.New("circuit breaker is open")

type Breaker struct {
	mu        sync.Mutex
	state     State
	failures  int
	threshold int
	coolDown  time.Duration
	openedAt  time.Time
	probing   bool
	now       func() time.Time
}

func NewBreaker(threshold int, coolDown time.Duration, now func() time.Time) *Breaker {
	return &Breaker{threshold: threshold, coolDown: coolDown, now: now}
}

func (b *Breaker) State() State {
	b.mu.Lock()
	defer b.mu.Unlock()
	return b.state
}

func (b *Breaker) Do(fn func() error) error {
	if !b.allow() {
		return ErrOpen
	}
	err := fn()
	b.record(err)
	return err
}

func (b *Breaker) allow() bool {
	b.mu.Lock()
	defer b.mu.Unlock()
	// @solution
	switch b.state {
	case Open:
		if b.now().Sub(b.openedAt) < b.coolDown {
			return false
		}
		b.state, b.probing = HalfOpen, true
		return true
	case HalfOpen:
		if b.probing {
			return false
		}
		b.probing = true
		return true
	default:
		return true
	}
	// @starter
	// return true // TODO
	// @end
}

func (b *Breaker) record(err error) {
	b.mu.Lock()
	defer b.mu.Unlock()
	// @solution
	switch {
	case b.state == HalfOpen && err == nil:
		b.state, b.failures, b.probing = Closed, 0, false
	case b.state == HalfOpen:
		b.state, b.openedAt, b.probing = Open, b.now(), false
	case err == nil:
		b.failures = 0
	default:
		b.failures++
		if b.failures >= b.threshold {
			b.state, b.openedAt = Open, b.now()
		}
	}
	// @starter
	// // TODO
	// @end
}

package main

import (
	"errors"
	"sync"
	"sync/atomic"
	"testing"
	"time"
)

var boom = errors.New("downstream error")

func TestStateMachine(t *testing.T) {
	now := time.Unix(0, 0)
	var mu sync.Mutex
	clock := func() time.Time { mu.Lock(); defer mu.Unlock(); return now }
	advance := func(d time.Duration) { mu.Lock(); now = now.Add(d); mu.Unlock() }

	b := NewBreaker(3, 5*time.Second, clock)
	fail := func() error { return boom }
	ok := func() error { return nil }

	b.Do(fail)
	b.Do(fail)
	b.Do(ok) // 成功清零
	b.Do(fail)
	b.Do(fail)
	if b.State() != Closed {
		t.Fatal("没有连续 3 次失败，不应打开")
	}
	b.Do(fail)
	if b.State() != Open {
		t.Fatal("连续 3 次失败应打开")
	}
	called := false
	if err := b.Do(func() error { called = true; return nil }); !errors.Is(err, ErrOpen) || called {
		t.Fatal("打开状态应直接返回 ErrOpen，不调用 fn")
	}

	advance(6 * time.Second)
	if err := b.Do(fail); !errors.Is(err, boom) || b.State() != Open {
		t.Fatal("半开探测失败应重新打开")
	}
	advance(3 * time.Second)
	if err := b.Do(ok); !errors.Is(err, ErrOpen) {
		t.Fatal("重新打开后要重新计时冷却时间")
	}
	advance(3 * time.Second)
	if err := b.Do(ok); err != nil || b.State() != Closed {
		t.Fatal("半开探测成功应关闭")
	}
}

func TestSingleProbe(t *testing.T) {
	now := time.Unix(0, 0)
	b := NewBreaker(1, time.Second, func() time.Time { return now })
	b.Do(func() error { return boom })
	now = now.Add(2 * time.Second)

	release := make(chan struct{})
	var probes atomic.Int32
	var wg sync.WaitGroup
	for i := 0; i < 20; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			b.Do(func() error { probes.Add(1); <-release; return nil })
		}()
	}
	time.Sleep(30 * time.Millisecond)
	close(release)
	wg.Wait()
	if probes.Load() != 1 {
		t.Fatalf("半开状态只应放行 1 个探测请求，实际 %d 个", probes.Load())
	}
	if b.State() != Closed {
		t.Fatal("探测成功后应关闭")
	}
}
