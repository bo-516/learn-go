=== db-01-1
title: 行锁：按 key 加锁的锁管理器
race: true
--- desc
MyISAM 只有表锁，InnoDB 支持行锁：不同的行可以被不同事务同时修改。实现一个按 key 加锁的 `RowLocker`：

- `Lock(key)` / `Unlock(key)`：同一个 key 互斥，不同 key 互不影响；
- 不能给每个出现过的 key 永久保留一把锁，否则内存会无限增长：用引用计数，没有人持有或等待某个 key 时删除它的锁；
- `Size()` 返回当前内部保存的锁数量。
--- solution
package main

import "sync"

type entry struct {
	mu   sync.Mutex
	refs int
}

type RowLocker struct {
	mu    sync.Mutex
	locks map[string]*entry
}

func NewRowLocker() *RowLocker { return &RowLocker{locks: map[string]*entry{}} }

func (l *RowLocker) Lock(key string) {
	// @solution
	l.mu.Lock()
	e, ok := l.locks[key]
	if !ok {
		e = &entry{}
		l.locks[key] = e
	}
	e.refs++
	l.mu.Unlock()
	e.mu.Lock() // 在全局锁之外等待，不阻塞其他 key
	// @starter
	// // TODO
	// @end
}

func (l *RowLocker) Unlock(key string) {
	// @solution
	l.mu.Lock()
	defer l.mu.Unlock()
	e := l.locks[key]
	e.refs--
	if e.refs == 0 {
		delete(l.locks, key)
	}
	e.mu.Unlock()
	// @starter
	// // TODO
	// @end
}

func (l *RowLocker) Size() int {
	l.mu.Lock()
	defer l.mu.Unlock()
	return len(l.locks)
}

func TestCleanup(t *testing.T) {
	l := NewRowLocker()
	// 每个 key 一个计数器；map 本身只读，计数器由对应 key 的锁保护
	counter := map[string]*int{}
	for i := 0; i < 5; i++ {
		counter[fmt.Sprint("row:", i)] = new(int)
	}
	var wg sync.WaitGroup
	for g := 0; g < 8; g++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for i := 0; i < 200; i++ {
				k := fmt.Sprint("row:", i%5)
				l.Lock(k)
				*counter[k]++
				l.Unlock(k)
			}
		}()
	}
	wg.Wait()
	if *counter["row:0"] != 8*40 {
		t.Fatalf("row:0 计数 = %d，期望 %d", *counter["row:0"], 8*40)
	}
	if n := l.Size(); n != 0 {
		t.Fatalf("所有锁都释放后，内部还剩 %d 把锁", n)
	}
}
--- explain
关键在 `Lock`：先在全局锁里找到（或创建）这个 key 的锁并增加引用计数，然后**释放全局锁之后**再等待 key 锁。如果在持有全局锁时等待，一个 key 被长时间占用就会卡住所有 key，退化成表锁。

=== db-02-1
title: B+ 树能存多少行
--- desc
估算 InnoDB 一棵 B+ 树的容量：

- `Fanout(pageSize, keySize, ptrSize)`：一个非叶子页能放多少个「键 + 子页指针」，即 `pageSize / (keySize + ptrSize)`；
- `Capacity(height, fanout, rowsPerLeaf)`：高度为 `height` 的树最多能存多少行：`fanout^(height-1) * rowsPerLeaf`；
- `MinHeight(rows, fanout, rowsPerLeaf)`：存下 `rows` 行最少需要几层（至少 1 层）。

经典结论：16KB 页、BIGINT 主键（8 字节）+ 6 字节指针，扇出约 1170；每个叶子页放 16 行（每行约 1KB），3 层就能存两千多万行，所以主键查询一般只需要 2～3 次 IO。
--- solution
package main

func Fanout(pageSize, keySize, ptrSize int) int {
	// @solution
	return pageSize / (keySize + ptrSize)
	// @starter
	// return 0 // TODO
	// @end
}

func Capacity(height, fanout, rowsPerLeaf int) int {
	// @solution
	c := rowsPerLeaf
	for i := 1; i < height; i++ {
		c *= fanout
	}
	return c
	// @starter
	// return 0 // TODO
	// @end
}

func MinHeight(rows, fanout, rowsPerLeaf int) int {
	// @solution
	h := 1
	for Capacity(h, fanout, rowsPerLeaf) < rows {
		h++
	}
	return h
	// @starter
	// return 0 // TODO
	// @end
}

=== db-02-2
title: B+ 树的范围扫描
--- desc
B+ 树适合范围查询：先通过上层索引二分定位到起始叶子页，然后沿着叶子之间的链表顺序扫描。

`Index` 里 `Leaves` 是按顺序排列的叶子页，叶子之间用 `Next` 连接；`Seps[i]` 是第 `i+1` 个叶子的最小 key（相当于上层的分隔键）。

实现 `Range(lo, hi)`：返回 `[lo, hi]` 范围内的所有 key（升序），以及读取的叶子页数量。要求：

- 用二分查找（`sort.Search` 或 `slices.BinarySearch`）定位起始叶子，不能从第一个叶子开始扫；
- 沿 `Next` 指针扫描，遇到大于 `hi` 的 key 立即停止。
--- solution
package main

import "sort"

type Leaf struct {
	Keys []int
	Next *Leaf
}

type Index struct {
	Seps   []int
	Leaves []*Leaf
}

func Build(keys []int, perLeaf int) *Index {
	ix := &Index{}
	for i := 0; i < len(keys); i += perLeaf {
		leaf := &Leaf{Keys: keys[i:min(i+perLeaf, len(keys))]}
		if n := len(ix.Leaves); n > 0 {
			ix.Leaves[n-1].Next = leaf
			ix.Seps = append(ix.Seps, leaf.Keys[0])
		}
		ix.Leaves = append(ix.Leaves, leaf)
	}
	return ix
}

func (ix *Index) Range(lo, hi int) (keys []int, leavesRead int) {
	if len(ix.Leaves) == 0 {
		return nil, 0
	}
	// @solution
	// 第一个分隔键 > lo 的位置，就是起始叶子的下标
	start := sort.Search(len(ix.Seps), func(i int) bool { return ix.Seps[i] > lo })
	for leaf := ix.Leaves[start]; leaf != nil; leaf = leaf.Next {
		leavesRead++
		for _, k := range leaf.Keys {
			if k > hi {
				return keys, leavesRead
			}
			if k >= lo {
				keys = append(keys, k)
			}
		}
	}
	return keys, leavesRead
	// @starter
	// for leaf := ix.Leaves[0]; leaf != nil; leaf = leaf.Next { // TODO: 从头扫描太慢
	// 	leavesRead++
	// 	for _, k := range leaf.Keys {
	// 		if k >= lo && k <= hi {
	// 			keys = append(keys, k)
	// 		}
	// 	}
	// }
	// return keys, leavesRead
	// @end
}

--- test

=== db-03-1
title: 回表与覆盖索引
--- desc
用 Go 模拟 InnoDB 的两种索引：

- **聚簇索引** `rows`：主键 → 整行数据；
- **二级索引** `byName`：按 `(name, id)` 排序的切片，叶子节点只存索引列和主键。

实现：

- `FindByName(name) []Row`：先在二级索引里二分找到所有匹配的主键，再逐个去聚簇索引取整行——每取一次计为一次**回表**（`t.Lookups++`）；
- `IDsByName(name) []int`：只需要主键，二级索引里已经有了，这就是**覆盖索引**，不能回表。
--- solution
package main

import (
	"slices"
	"strings"
)

type Row struct {
	ID   int
	Name string
	Age  int
}

type secEntry struct {
	Name string
	ID   int
}

type Table struct {
	rows    map[int]Row
	byName  []secEntry
	Lookups int
}

func NewTable(rows []Row) *Table {
	t := &Table{rows: map[int]Row{}}
	for _, r := range rows {
		t.rows[r.ID] = r
		t.byName = append(t.byName, secEntry{r.Name, r.ID})
	}
	slices.SortFunc(t.byName, func(a, b secEntry) int {
		if c := strings.Compare(a.Name, b.Name); c != 0 {
			return c
		}
		return a.ID - b.ID
	})
	return t
}

func (t *Table) IDsByName(name string) []int {
	// @solution
	i, _ := slices.BinarySearchFunc(t.byName, name, func(e secEntry, name string) int {
		return strings.Compare(e.Name, name)
	})
	var ids []int
	for ; i < len(t.byName) && t.byName[i].Name == name; i++ {
		ids = append(ids, t.byName[i].ID)
	}
	return ids
	// @starter
	// return nil // TODO: 二分查找二级索引
	// @end
}

func (t *Table) FindByName(name string) []Row {
	// @solution
	var out []Row
	for _, id := range t.IDsByName(name) {
		t.Lookups++
		out = append(out, t.rows[id])
	}
	return out
	// @starter
	// return nil // TODO: 用 IDsByName 拿到主键后回表
	// @end
}
--- test

=== db-04-1
title: 最左前缀：能用到索引的几列
--- desc
联合索引 `(a, b, c)` 按 a、b、c 依次排序。实现 `UsableColumns(index, conds)`，返回查询条件能用到索引的前几列：

1. 按索引列的顺序检查；
2. 这一列有等值条件（`=`、`in`）→ 可用，继续看下一列；
3. 这一列有范围条件（`>`、`<`、`between`、`like-prefix` 即 `LIKE 'abc%'`）→ 可用，但**后面的列都用不上了**；
4. 这一列没有条件，或者条件让索引失效（`like-suffix` 即 `LIKE '%abc'`、`func` 即对列用了函数）→ 停止。

WHERE 里条件的书写顺序不影响结果（优化器会调整）。
--- solution
package main

type Cond struct {
	Col string
	Op  string // "=", "in", ">", "<", "between", "like-prefix", "like-suffix", "func"
}

func UsableColumns(index []string, conds []Cond) int {
	// @solution
	ops := map[string]string{}
	for _, c := range conds {
		if _, seen := ops[c.Col]; !seen || c.Op == "=" || c.Op == "in" {
			ops[c.Col] = c.Op
		}
	}
	n := 0
	for _, col := range index {
		switch ops[col] {
		case "=", "in":
			n++
		case ">", "<", "between", "like-prefix":
			return n + 1
		default:
			return n
		}
	}
	return n
	// @starter
	// return 0 // TODO
	// @end
}
--- test
package main

import "testing"

func TestUsableColumns(t *testing.T) {
	idx := []string{"a", "b", "c"}
	cases := []struct {
		name  string
		conds []Cond
		want  int
	}{
		{"全部等值", []Cond{{"a", "="}, {"b", "="}, {"c", "="}}, 3},
		{"顺序无关", []Cond{{"c", "="}, {"a", "="}, {"b", "in"}}, 3},
		{"跳过中间列", []Cond{{"a", "="}, {"c", "="}}, 1},
		{"没有最左列", []Cond{{"b", "="}, {"c", "="}}, 0},

=== db-05-1
title: 用 iota 定义隔离级别
--- desc
用 Go 的常量和方法给事务隔离级别建模：

- `Level` 类型用 `iota` 定义四个级别，实现 `String()` 返回 `READ-UNCOMMITTED`、`READ-COMMITTED`、`REPEATABLE-READ`、`SERIALIZABLE`；
- `ParseLevel(s)`：不区分大小写，空格和连字符都可以，如 `"read committed"`、`"REPEATABLE-READ"`；
- `Allows(a Anomaly)`：按 SQL 标准，该级别是否**可能出现**某种异常：读未提交三种都可能，读已提交可能出现不可重复读和幻读，可重复读可能出现幻读，串行化都不会。

（InnoDB 在 RR 下通过 MVCC + 临键锁在大多数情况下也避免了幻读，这里按标准回答。）
--- solution
package main

import (
	"fmt"
	"strings"
)

type Level int

const (
	ReadUncommitted Level = iota
	ReadCommitted
	RepeatableRead
	Serializable
)

type Anomaly int

const (
	DirtyRead Anomaly = iota
	NonRepeatableRead
	PhantomRead
)

var levelNames = [...]string{"READ-UNCOMMITTED", "READ-COMMITTED", "REPEATABLE-READ", "SERIALIZABLE"}

func (l Level) String() string {
	// @solution
	if l < 0 || int(l) >= len(levelNames) {
		return fmt.Sprintf("Level(%d)", int(l))
	}
	return levelNames[l]
	// @starter
	// return "" // TODO
	// @end
}

func ParseLevel(s string) (Level, error) {
	// @solution
	norm := strings.ToUpper(strings.ReplaceAll(strings.TrimSpace(s), " ", "-"))
	for i, name := range levelNames {
		if name == norm {
			return Level(i), nil
		}
	}
	return 0, fmt.Errorf("unknown isolation level %q", s)
	// @starter
	// return 0, fmt.Errorf("TODO")
	// @end
}

func (l Level) Allows(a Anomaly) bool {
	// @solution
	// 级别越高，能避免的异常越多：RU 避免 0 种，RC 避免脏读，RR 再避免不可重复读，S 全部避免
	return int(a) >= int(l)
	// @starter
	// return false // TODO
	// @end
}

实现 `Diagnose(row)`，根据 EXPLAIN 的一行结果给出问题列表（按下面的顺序检查）：

| --- | --- |

package main

type ExplainRow struct {
	Type     string
	Key      string
	Rows     int
	Filtered float64
	Extra    string
}

func Diagnose(r ExplainRow) []string {
	var out []string
	// @solution
	switch r.Type {
	case "ALL":
		out = append(out, "full-table-scan")
	case "index":
		out = append(out, "full-index-scan")

	if r.Key == "" {
		out = append(out, "no-index")
	}
	if strings.Contains(r.Extra, "Using filesort") {
		out = append(out, "filesort")
	}
	if strings.Contains(r.Extra, "Using temporary") {
		out = append(out, "temporary-table")
	}
	if r.Rows > 10000 && r.Filtered < 10 {
		out = append(out, "low-selectivity")

- `NewReadView(active, nextTrxID, creator)`：`active` 是创建视图时的活跃事务列表；`MinTrxID` 取其中最小值（没有活跃事务时等于 `nextTrxID`）；`MaxTrxID = nextTrxID`；

title: RC 和 RR 的区别：ReadView 何时创建
--- desc
`Engine` 模拟一行数据和事务系统，`ReadView` 已经实现好。实现 `Select(txID, level)`：

- `"RC"`：**每次** Select 都基于当前活跃事务创建新的 ReadView；
- `"RR"`：事务里**第一次** Select 时创建 ReadView，保存在 `e.views[txID]` 里，之后一直复用。

测试会重现经典场景：T1 开始后，T2 修改并提交，T1 在 RC 下能读到新值，在 RR 下读到的还是旧值。
--- solution
package main

import "slices"

type ReadView struct {
	active   []int
	min, max int
	creator  int
}

func (v ReadView) visible(id int) bool {
	switch {
	case id == v.creator, id < v.min:
		return true
	case id >= v.max:
		return false
	}
	return !slices.Contains(v.active, id)
}

type Version struct {
	TrxID int
	Value string
}

type Engine struct {
	nextID int
	active map[int]bool
	chain  []Version // 最新版本在前
	views  map[int]ReadView
}

func NewEngine(initial string) *Engine {
	return &Engine{nextID: 2, active: map[int]bool{}, chain: []Version{{1, initial}}, views: map[int]ReadView{}}
}

func (e *Engine) Begin() int {
	id := e.nextID
	e.nextID++
	e.active[id] = true
	return id
}

func (e *Engine) Commit(id int) { delete(e.active, id) }

func (e *Engine) Write(id int, v string) {
	e.chain = append([]Version{{id, v}}, e.chain...)
}

func (e *Engine) newView(creator int) ReadView {
	v := ReadView{min: e.nextID, max: e.nextID, creator: creator}
	for id := range e.active {
		v.active = append(v.active, id)
		v.min = min(v.min, id)
	}
	return v
}

func (e *Engine) Select(txID int, level string) string {
	// @solution
	var view ReadView
	if level == "RR" {
		v, ok := e.views[txID]
		if !ok {
			v = e.newView(txID)
			e.views[txID] = v
		}
		view = v
	} else {
		view = e.newView(txID)
	}
	for _, ver := range e.chain {
		if view.visible(ver.TrxID) {
			return ver.Value
		}
	}
	return ""
	// @starter
	// return e.chain[0].Value // TODO: 直接读最新版本会读到未提交的数据
	// @end
}

title: 临键锁锁住了哪些区间
--- desc
用简化模型理解 RR 下范围查询的加锁范围。`keys` 是**非唯一二级索引**上已有的值（升序）。执行 `SELECT ... WHERE k BETWEEN lo AND hi FOR UPDATE` 时：

- 对扫描到的每条满足条件的记录加**临键锁** `(前一个值, 当前值]`；
- 扫描还会多看一条：第一条大于 `hi` 的记录也加临键锁 `(前一个值, 它]`；如果后面没有记录，锁住 `(最后一个值, +∞)`。

用 `math.MinInt` 和 `math.MaxInt` 表示 -∞ 和 +∞。实现：

- `NextKeyLocks(keys, lo, hi) []Interval`：按顺序返回加锁的区间；
- `Blocked(locks, v) bool`：插入值 `v` 会不会被这些锁阻塞（落在某个区间里）。
--- solution
package main

import "math"

type Interval struct {
	Lo, Hi int // 左开右闭 (Lo, Hi]；Hi == math.MaxInt 表示 +∞
}

func NextKeyLocks(keys []int, lo, hi int) []Interval {
	var locks []Interval
	// @solution
	prev := math.MinInt
	for _, k := range keys {
		if k < lo {
			prev = k
			continue
		}
		locks = append(locks, Interval{prev, k})
		if k > hi {
			return locks
		}
		prev = k
	}
	locks = append(locks, Interval{prev, math.MaxInt})
	// @starter
	// // TODO
	// @end
	return locks
}

func Blocked(locks []Interval, v int) bool {
	// @solution
	for _, l := range locks {
		if v > l.Lo && v <= l.Hi {
			return true
		}
	}
	return false
	// @starter
	// return false // TODO
	// @end
}

	}
}

title: 两阶段提交与崩溃恢复
--- desc
一次提交按顺序执行三步：`redo-prepare` → `binlog-write` → `redo-commit`。实现：

- `Commit(crashAfter)`：返回实际执行了的步骤；`crashAfter` 表示在第几步之后崩溃（0 表示不崩溃，正常执行完三步）；
- `StateAfter(steps)`：根据执行过的步骤推出崩溃后看到的状态：redo 是 `none` / `prepare` / `commit`，binlog 是否完整；
- `Recover(redo, binlogComplete)`：崩溃恢复的决策：redo 已 commit → `commit`；redo 是 prepare 且 binlog 完整 → `commit`；其他情况 → `rollback`。

测试会验证：无论在哪一步崩溃，恢复后「事务是否提交」都和「binlog 里有没有这个事务」一致，主从数据因此不会分叉。
--- solution
package main

import "slices"

var allSteps = []string{"redo-prepare", "binlog-write", "redo-commit"}

func Commit(crashAfter int) []string {
	// @solution
	if crashAfter <= 0 || crashAfter > len(allSteps) {
		crashAfter = len(allSteps)
	}
	return slices.Clone(allSteps[:crashAfter])
	// @starter
	// return nil // TODO
	// @end
}

func StateAfter(steps []string) (redo string, binlogComplete bool) {
	// @solution
	redo = "none"
	for _, s := range steps {
		switch s {
		case "redo-prepare":
			redo = "prepare"
		case "binlog-write":
			binlogComplete = true
		case "redo-commit":
			redo = "commit"
		}
	}
	return redo, binlogComplete
	// @starter
	// return "none", false // TODO
	// @end
}

func Recover(redo string, binlogComplete bool) string {
	// @solution
	if redo == "commit" || (redo == "prepare" && binlogComplete) {
		return "commit"
	}
	return "rollback"
	// @starter
	// return "rollback" // TODO
	// @end
}

title: 解析慢查询日志
--- desc
慢查询日志的每条记录形如：

```text
# Time: 2026-10-13T10:00:00
# User@Host: app[app] @ [10.0.0.1]
# Query_time: 2.500000  Lock_time: 0.000100 Rows_sent: 10  Rows_examined: 1000000
SET timestamp=1760320800;
SELECT * FROM orders WHERE status = 1 LIMIT 10;
```

实现：

- `ParseSlowLog(text)`：解析出每条记录的 `QueryTime`、`LockTime`、`RowsSent`、`RowsExamined` 和 SQL（跳过 `SET timestamp=` 和 `use xxx;` 行，多行 SQL 用空格拼接）；
- `Worst(qs, n)`：按 `QueryTime` 从大到小返回前 n 条（不修改入参）；
- `Suspicious(q)`：扫描行数超过返回行数 1000 倍（返回 0 行按 1 行算）时为 true——这通常意味着缺索引。
--- solution
package main

import (
	"cmp"
	"slices"
	"strconv"
	"strings"
)

type SlowQuery struct {
	QueryTime    float64
	LockTime     float64
	RowsSent     int
	RowsExamined int
	SQL          string
}

func ParseSlowLog(text string) ([]SlowQuery, error) {
	// @solution
	var out []SlowQuery
	var cur *SlowQuery
	var sql []string
	flush := func() {
		if cur != nil {
			cur.SQL = strings.Join(sql, " ")
			out = append(out, *cur)
		}
		cur, sql = nil, nil
	}
	for _, line := range strings.Split(text, "\n") {
		line = strings.TrimSpace(line)
		switch {
		case line == "":
		case strings.HasPrefix(line, "# Query_time:"):
			flush()
			cur = &SlowQuery{}
			f := strings.Fields(strings.TrimPrefix(line, "#"))
			for i := 0; i+1 < len(f); i += 2 {
				v := f[i+1]
				var err error
				switch f[i] {
				case "Query_time:":
					cur.QueryTime, err = strconv.ParseFloat(v, 64)
				case "Lock_time:":
					cur.LockTime, err = strconv.ParseFloat(v, 64)
				case "Rows_sent:":
					cur.RowsSent, err = strconv.Atoi(v)
				case "Rows_examined:":
					cur.RowsExamined, err = strconv.Atoi(v)
				}
				if err != nil {
					return nil, err
				}
			}
		case strings.HasPrefix(line, "#"):
			flush()
		case cur == nil, strings.HasPrefix(line, "SET timestamp="), strings.HasPrefix(strings.ToLower(line), "use "):
		default:
			sql = append(sql, line)
		}
	}
	flush()
	return out, nil
	// @starter
	// return nil, nil // TODO
	// @end
}

func Worst(qs []SlowQuery, n int) []SlowQuery {
	// @solution
	s := slices.Clone(qs)
	slices.SortStableFunc(s, func(a, b SlowQuery) int { return cmp.Compare(b.QueryTime, a.QueryTime) })
	return s[:min(n, len(s))]
	// @starter
	// return nil // TODO
	// @end
}

func Suspicious(q SlowQuery) bool {
	// @solution
	return q.RowsExamined > 1000*max(q.RowsSent, 1)
	// @starter
	// return false // TODO
	// @end
}

title: 游标分页代替 OFFSET
--- desc
`LIMIT 1000000, 10` 要先扫描并丢弃一百万行。游标分页记住上一页最后一条的 ID，下一页直接从它之后开始。

`rows` 按 ID **降序**排列（最新的在前）。实现：

- `PageByOffset(rows, offset, size)`：传统分页，返回结果和「扫描过的行数」（offset + 返回行数）；
- `PageAfter(rows, lastID, size)`：返回 ID **小于** `lastID` 的前 `size` 行，`lastID <= 0` 表示第一页。用二分查找定位起点，扫描行数只算真正返回的行。
--- solution
package main

import "sort"

type Row struct {
	ID    int
	Title string
}

func PageByOffset(rows []Row, offset, size int) ([]Row, int) {
	if offset >= len(rows) {
		return nil, len(rows)
	}
	end := min(offset+size, len(rows))
	return rows[offset:end], end
}

func PageAfter(rows []Row, lastID, size int) ([]Row, int) {
	// @solution
	start := 0
	if lastID > 0 {
		// rows 按 ID 降序，找到第一个 ID < lastID 的位置
		start = sort.Search(len(rows), func(i int) bool { return rows[i].ID < lastID })
	}
	end := min(start+size, len(rows))
	return rows[start:end], end - start
	// @starter
	// return nil, 0 // TODO
	// @end
}

=== db-12-1
title: 等待图中的死锁检测
--- desc
InnoDB 维护一张「等待图」：事务 A 等待事务 B 持有的锁，就有一条 A → B 的边，出现环就是死锁，需要回滚其中一个事务。

实现 `DetectDeadlock(waits)`：`waits[a]` 是事务 a 正在等待的事务列表。

- 按事务 ID 从小到大作为起点做深度优先搜索（邻居也按从小到大），找到的**第一个环**就是死锁；
- 返回环里 ID 最大的事务作为牺牲者（简化成「回滚最年轻的事务」），以及 `true`；
- 没有环返回 `0, false`。
--- solution
package main

import (
	"maps"
	"slices"
)

func DetectDeadlock(waits map[int][]int) (victim int, found bool) {
	// @solution
	const (
		white = iota
		grey
		black
	)
	color := map[int]int{}
	var path []int
	var dfs func(u int) []int
	dfs = func(u int) []int {
		color[u] = grey
		path = append(path, u)
		next := slices.Sorted(slices.Values(waits[u]))
		for _, v := range next {
			switch color[v] {
			case grey:
				i := slices.Index(path, v)
				return slices.Clone(path[i:])
			case white:
				if cycle := dfs(v); cycle != nil {
					return cycle
				}
			}
		}
		path = path[:len(path)-1]
		color[u] = black
		return nil
	}
	for _, u := range slices.Sorted(maps.Keys(waits)) {
		if color[u] == white {
			if cycle := dfs(u); cycle != nil {
				return slices.Max(cycle), true
			}
		}
	}
	return 0, false
	// @starter
	// return 0, false // TODO
	// @end
}
--- test

func TestDetectDeadlock(t *testing.T) {
	cases := []struct {
		name   string
		waits  map[int][]int
		victim int
		found  bool
	}{
		{"两个事务互相等待", map[int][]int{1: {2}, 2: {1}}, 2, true},
		{"三角环", map[int][]int{3: {7}, 7: {5}, 5: {3}, 9: {3}}, 7, true},
		{"没有环", map[int][]int{1: {2}, 2: {3}, 4: {3}}, 0, false},
		{"多条路径汇合但无环", map[int][]int{1: {2, 3}, 2: {3}, 3: {}}, 0, false},
		{"环不经过起点", map[int][]int{1: {2}, 2: {3}, 3: {4}, 4: {2}}, 4, true},
		{"空图", map[int][]int{}, 0, false},
	}
	for _, c := range cases {
		v, ok := DetectDeadlock(c.waits)
		if v != c.victim || ok != c.found {
			t.Errorf("%s: DetectDeadlock = %d, %v，期望 %d, %v", c.name, v, ok, c.victim, c.found)
		}
	}
}

title: 主从延迟下的读写路由
--- desc
主从复制有延迟，用户刚写完立刻去从库读，可能读不到自己写的数据。实现一个 `Router`：

- `RecordWrite(user)`：记录用户最近一次写入的时间；
- `ReadTarget(user)`：
  1. 用户在最近 `window` 时间内写过 → 读主库 `"primary"`；

	return &Router{now: now, window: window, maxLag: maxLag, replicas: replicas, lastWrite: map[string]time.Time{}}
}

	// @solution
	r.lastWrite[user] = r.now()
	// @starter

	// @solution

		return "primary"

=== db-14-1
title: 模拟 gh-ost 在线改表
--- desc
gh-ost 这类工具在线加字段的思路：建一张影子表，**分批**把原表数据拷过去，同时把拷贝期间产生的 binlog 变更也应用到影子表，最后原子地切换表名。

实现 `Migrate(orig, chunk, batches)`：

1. 按原表**开始时**的最小、最大 ID，每 `chunk` 个 ID 为一批；
2. 每拷贝一批（从原表**当前**数据里取这个 ID 范围内的行，用 INSERT IGNORE 语义：影子表已有的行不覆盖），就把 `batches[i]`（这段时间产生的变更）应用到**原表和影子表**：`upsert` 写入，`delete` 删除；
3. 拷贝结束后还剩下的 batches 也要应用；
4. 返回影子表。最终影子表必须和原表完全一致。
--- solution
package main

import (
	"maps"
	"slices"
)

type Event struct {
	Op  string // "upsert" | "delete"
	ID  int
	Val string
}

func apply(m map[int]string, e Event) {
	if e.Op == "delete" {
		delete(m, e.ID)
	} else {
		m[e.ID] = e.Val
	}
}

func Migrate(orig map[int]string, chunk int, batches [][]Event) map[int]string {
	shadow := map[int]string{}
	if len(orig) == 0 && len(batches) == 0 {
		return shadow
	}
	// @solution
	keys := slices.Sorted(maps.Keys(orig))
	b := 0
	applyNext := func() {
		if b < len(batches) {
			for _, e := range batches[b] {
				apply(orig, e)
				apply(shadow, e)
			}
			b++
		}
	}
	if len(keys) > 0 {
		lo, hi := keys[0], keys[len(keys)-1]
		for start := lo; start <= hi; start += chunk {
			for id := start; id < start+chunk && id <= hi; id++ {
				if v, ok := orig[id]; ok {
					if _, exists := shadow[id]; !exists {
						shadow[id] = v
					}
				}
			}
			applyNext()
		}
	}
	for b < len(batches) {
		applyNext()
	}
	// @starter
	// for id, v := range orig { // TODO: 只拷贝了数据，没有处理拷贝期间的变更
	// 	shadow[id] = v
	// }
	// @end
	return shadow
}

--- desc
订单表按 `user_id` 分成 4 个库 × 8 张表，一共 32 个分片：

- `Route(userID)`：`slot = userID % 32`，库号 `slot / 8`，表号 `slot % 8`；
- 按订单号查询时不知道 user_id，怎么路由？**基因法**：生成订单号时把 user_id 的低 5 位（32 = 2⁵）嵌进订单号的低 5 位：`OrderID(seq, userID) = seq << 5 | userID & 31`；

	tablesPerDB = 8
	slots      = dbCount * tablesPerDB // 32
	geneBits   = 5

func Route(userID int64) (db, table int) {
	// @solution
	slot := int(userID % slots)
	return slot / tablesPerDB, slot % tablesPerDB
	// @starter
	// return 0, 0 // TODO
	// @end
}

func OrderID(seq, userID int64) int64 {
	// @solution
	return seq<<geneBits | userID&(slots-1)
	// @starter
	// return seq // TODO
	// @end
}

func RouteByOrder(orderID int64) (db, table int) {

	// @end

--- desc
大表 `count(*)` 很慢，常见做法是单独维护计数（Redis 或计数表）。但每次插入都去更新同一行计数，会形成热点。

- `Add(key, delta)`：并发安全地累加增量；
