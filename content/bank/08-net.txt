=== net-01-1
title: TCP 状态机
--- desc
用一张转移表实现 TCP 连接的状态机 `Next(state, event)`，非法转移返回错误：

| 当前状态 | 事件 | 新状态 |
| --- | --- | --- |
| CLOSED | listen | LISTEN |
| CLOSED | connect（发 SYN） | SYN_SENT |
| LISTEN | recv_syn | SYN_RCVD |
| SYN_SENT | recv_synack | ESTABLISHED |
| SYN_RCVD | recv_ack | ESTABLISHED |
| ESTABLISHED | close（主动关闭，发 FIN） | FIN_WAIT_1 |
| ESTABLISHED | recv_fin（被动关闭） | CLOSE_WAIT |
| FIN_WAIT_1 | recv_ack | FIN_WAIT_2 |
| FIN_WAIT_2 | recv_fin | TIME_WAIT |
| CLOSE_WAIT | close | LAST_ACK |
| LAST_ACK | recv_ack | CLOSED |
| TIME_WAIT | timeout_2msl | CLOSED |

注意 CLOSE_WAIT 只有在**本方程序调用 close** 后才会离开——这就是线上 CLOSE_WAIT 堆积的原因。
--- solution
package main

import "fmt"

type State string

const (
	Closed      State = "CLOSED"
	Listen      State = "LISTEN"
	SynSent     State = "SYN_SENT"
	SynRcvd     State = "SYN_RCVD"
	Established State = "ESTABLISHED"
	FinWait1    State = "FIN_WAIT_1"
	FinWait2    State = "FIN_WAIT_2"
	CloseWait   State = "CLOSE_WAIT"
	LastAck     State = "LAST_ACK"
	TimeWait    State = "TIME_WAIT"
)

// @solution
var transitions = map[State]map[string]State{
	Closed:      {"listen": Listen, "connect": SynSent},
	Listen:      {"recv_syn": SynRcvd},
	SynSent:     {"recv_synack": Established},
	SynRcvd:     {"recv_ack": Established},
	Established: {"close": FinWait1, "recv_fin": CloseWait},
	FinWait1:    {"recv_ack": FinWait2},
	FinWait2:    {"recv_fin": TimeWait},
	CloseWait:   {"close": LastAck},
	LastAck:     {"recv_ack": Closed},
	TimeWait:    {"timeout_2msl": Closed},
}

// @end

func Next(s State, event string) (State, error) {
	// @solution
	if next, ok := transitions[s][event]; ok {
		return next, nil
	}
	return s, fmt.Errorf("invalid event %q in state %s", event, s)
	// @starter
	// return s, fmt.Errorf("TODO")
	// @end
}
--- test
package main

import "testing"

func run(t *testing.T, s State, events ...string) State {
	t.Helper()
	for _, e := range events {
		var err error
		if s, err = Next(s, e); err != nil {
			t.Fatalf("事件 %s：%v", e, err)
		}
	}
	return s
}

func TestHandshake(t *testing.T) {
	if s := run(t, Closed, "connect", "recv_synack"); s != Established {
		t.Fatalf("客户端握手后 = %s", s)
	}
	if s := run(t, Closed, "listen", "recv_syn", "recv_ack"); s != Established {
		t.Fatalf("服务端握手后 = %s", s)
	}
}

func TestActiveClose(t *testing.T) {
	s := run(t, Established, "close", "recv_ack", "recv_fin")
	if s != TimeWait {
		t.Fatalf("主动关闭方应进入 TIME_WAIT，实际 %s", s)
	}
	if s := run(t, s, "timeout_2msl"); s != Closed {
		t.Fatalf("2MSL 后应为 CLOSED，实际 %s", s)
	}
}

func TestPassiveClose(t *testing.T) {
	s := run(t, Established, "recv_fin")
	if s != CloseWait {
		t.Fatalf("被动关闭方收到 FIN 后应为 CLOSE_WAIT，实际 %s", s)
	}
	if _, err := Next(s, "timeout_2msl"); err == nil {
		t.Fatal("CLOSE_WAIT 不会因为超时自动离开，必须由程序调用 close")
	}
	if s := run(t, s, "close", "recv_ack"); s != Closed {
		t.Fatalf("被动关闭最终应为 CLOSED，实际 %s", s)
	}
}

func TestInvalid(t *testing.T) {
	if _, err := Next(Closed, "recv_fin"); err == nil {
		t.Fatal("非法转移应返回错误")
	}
}

=== net-02-1
title: 按序号重组数据报
--- desc
UDP 不保证到达和顺序，TCP 帮你做的事情之一就是**按序号重组**。实现 `Reassemble(pkts)`：

- 包的序号从 0 开始连续编号，可能乱序、重复、丢失；
- 返回从序号 0 开始**连续收到**的那一段数据拼接结果（遇到第一个缺口就停止）；
- 以及 `0..最大序号` 之间所有缺失的序号（升序），这些需要请求重传。
--- solution
package main

type Packet struct {
	Seq  int
	Data string
}

func Reassemble(pkts []Packet) (data string, missing []int) {
	// @solution
	got := map[int]string{}
	maxSeq := -1
	for _, p := range pkts {
		if _, dup := got[p.Seq]; !dup {
			got[p.Seq] = p.Data
		}
		maxSeq = max(maxSeq, p.Seq)
	}
	contiguous := true
	for seq := 0; seq <= maxSeq; seq++ {
		d, ok := got[seq]
		if !ok {
			contiguous = false
			missing = append(missing, seq)
			continue
		}
		if contiguous {
			data += d
		}
	}
	return data, missing
	// @starter
	// return "", nil // TODO
	// @end
}
--- test
package main

import (
	"slices"
	"testing"
)

func TestReassemble(t *testing.T) {
	pkts := []Packet{{2, "C"}, {0, "A"}, {5, "F"}, {1, "B"}, {2, "C"}, {4, "E"}}
	data, missing := Reassemble(pkts)
	if data != "ABC" || !slices.Equal(missing, []int{3}) {
		t.Fatalf("data=%q missing=%v，期望 ABC [3]", data, missing)
	}
	pkts = append(pkts, Packet{3, "D"})
	if data, missing := Reassemble(pkts); data != "ABCDEF" || len(missing) != 0 {
		t.Fatalf("补齐后 data=%q missing=%v", data, missing)
	}
	if data, missing := Reassemble([]Packet{{1, "x"}}); data != "" || !slices.Equal(missing, []int{0}) {
		t.Fatalf("丢了第一个包：data=%q missing=%v", data, missing)
	}
	if data, missing := Reassemble(nil); data != "" || missing != nil {
		t.Fatal("空输入")
	}
}

=== net-03-1
title: 长度前缀分帧，解决粘包和拆包
--- desc
TCP 是字节流，没有消息边界。实现最常用的「4 字节长度 + 消息体」分帧：

- `WriteFrame(w, payload)`：先写 4 字节大端序的长度，再写消息体；
- `ReadFrame(r)`：读出一条完整的消息。必须用 `io.ReadFull`——一次 `Read` 不保证读满（拆包）；长度超过 `maxFrame` 返回 `ErrFrameTooLarge`，防止恶意的超大长度把内存耗尽。

测试会把多条消息连在一起（粘包），还会用 `iotest.OneByteReader` 让每次 Read 只返回 1 个字节（拆包）。
--- solution
package main

import (
	"bufio"
	"encoding/binary"
	"errors"
	"io"
)

const maxFrame = 1 << 20

var ErrFrameTooLarge = errors.New("frame too large")

func WriteFrame(w io.Writer, payload []byte) error {
	// @solution
	var hdr [4]byte
	binary.BigEndian.PutUint32(hdr[:], uint32(len(payload)))
	if _, err := w.Write(hdr[:]); err != nil {
		return err
	}
	_, err := w.Write(payload)
	return err
	// @starter
	// _, err := w.Write(payload) // TODO: 没有长度前缀，接收方无法区分消息边界
	// return err
	// @end
}

func ReadFrame(r *bufio.Reader) ([]byte, error) {
	// @solution
	var hdr [4]byte
	if _, err := io.ReadFull(r, hdr[:]); err != nil {
		return nil, err
	}
	n := binary.BigEndian.Uint32(hdr[:])
	if n > maxFrame {
		return nil, ErrFrameTooLarge
	}
	buf := make([]byte, n)
	if _, err := io.ReadFull(r, buf); err != nil {
		return nil, err
	}
	return buf, nil
	// @starter
	// buf := make([]byte, 1024)
	// n, err := r.Read(buf) // TODO: 一次 Read 可能只读到半条，也可能读到多条
	// return buf[:n], err
	// @end
}
--- test

=== net-03-2
title: 用 bufio.Scanner 自定义分帧
--- desc
`bufio.Scanner` 支持自定义切分函数。为「4 字节大端长度 + 消息体」的协议实现 `SplitFrames`，签名和 `bufio.SplitFunc` 一致：

- 数据不够一个完整帧时返回 `0, nil, nil`，让 Scanner 继续读取；
- 够一帧时返回 `advance = 4 + 长度`、`token = 消息体`；
- 已经到达 EOF（`atEOF`）但还剩不完整的数据时返回 `io.ErrUnexpectedEOF`；
- 没有剩余数据且 `atEOF` 时返回 `0, nil, nil` 结束扫描。

注意：Scanner 默认单个 token 最大 64KB，需要更大时调用 `scanner.Buffer` 调整。
--- solution
package main

import (
	"encoding/binary"
	"io"
)

func SplitFrames(data []byte, atEOF bool) (advance int, token []byte, err error) {
	// @solution
	if len(data) >= 4 {
		n := int(binary.BigEndian.Uint32(data[:4]))
		if len(data) >= 4+n {
			return 4 + n, data[4 : 4+n], nil
		}
	}
	if atEOF && len(data) > 0 {
		return 0, nil, io.ErrUnexpectedEOF
	}
	return 0, nil, nil
	// @starter
	// return 0, nil, io.EOF // TODO
	// @end
}

=== net-04-1
title: 队头阻塞：HTTP/1.1 vs HTTP/2
--- desc
用一个调度模型对比两种协议加载一组资源的总耗时。`costs` 是每个请求的服务端处理时间（按发起顺序）。

- `HTTP1Time(costs, conns)`：浏览器对一个域名开 `conns` 个连接，每个连接**同一时间只能处理一个请求**。请求按顺序分配给**最早空闲**的连接，返回所有请求完成的时间；
- `HTTP2Time(costs)`：一个连接上多路复用，所有请求同时进行，返回最慢的那个。

例如第一个请求很慢、只有 1 个连接时，后面的小请求全都被它堵住——这就是 HTTP 层的队头阻塞。
--- solution
package main

import "time"

func HTTP1Time(costs []time.Duration, conns int) time.Duration {
	// @solution
	free := make([]time.Duration, conns) // 每个连接的空闲时刻
	var end time.Duration
	for _, c := range costs {
		i := 0
		for j := range free {
			if free[j] < free[i] {
				i = j
			}
		}
		free[i] += c
		end = max(end, free[i])
	}
	return end
	// @starter
	// return 0 // TODO
	// @end
}

func HTTP2Time(costs []time.Duration) time.Duration {
	// @solution
	var m time.Duration
	for _, c := range costs {
		m = max(m, c)
	}
	return m
	// @starter
	// return 0 // TODO
	// @end
}
--- test

=== net-05-1
title: ECDHE 密钥交换 + AES-GCM
--- desc
TLS 的核心是两步：用 ECDHE 在不安全的信道上协商出共享密钥，再用对称加密传输数据。用标准库实现简化版：

- `NewParty()`：生成一个 X25519 私钥（`ecdh.X25519().GenerateKey(rand.Reader)`）；
- `SessionKey(priv, peerPub)`：用自己的私钥和对方的公钥算出共享秘密（`priv.ECDH`），再做一次 SHA-256 得到 32 字节会话密钥；
- `Seal(key, plaintext)`：AES-256-GCM 加密，随机生成 nonce 并放在密文**前面**；
- `Open(key, sealed)`：拆出 nonce 并解密，密文被篡改时返回错误（GCM 自带完整性校验）。
--- solution
package main

import (
	"crypto/aes"
	"crypto/cipher"
	"crypto/ecdh"
	"crypto/rand"
	"crypto/sha256"
	"errors"
)

func NewParty() (*ecdh.PrivateKey, error) {
	return ecdh.X25519().GenerateKey(rand.Reader)
}

func SessionKey(priv *ecdh.PrivateKey, peerPub *ecdh.PublicKey) ([]byte, error) {
	// @solution
	shared, err := priv.ECDH(peerPub)
	if err != nil {
		return nil, err
	}
	sum := sha256.Sum256(shared)
	return sum[:], nil
	// @starter
	// return nil, errors.New("TODO")
	// @end
}

func newGCM(key []byte) (cipher.AEAD, error) {
	block, err := aes.NewCipher(key)
	if err != nil {
		return nil, err
	}
	return cipher.NewGCM(block)
}

func Seal(key, plaintext []byte) ([]byte, error) {
	// @solution
	gcm, err := newGCM(key)
	if err != nil {
		return nil, err
	}
	nonce := make([]byte, gcm.NonceSize())
	if _, err := rand.Read(nonce); err != nil {
		return nil, err
	}
	return gcm.Seal(nonce, nonce, plaintext, nil), nil
	// @starter
	// return nil, errors.New("TODO")
	// @end
}

func Open(key, sealed []byte) ([]byte, error) {
	// @solution
	gcm, err := newGCM(key)
	if err != nil {
		return nil, err
	}
	if len(sealed) < gcm.NonceSize() {
		return nil, errors.New("ciphertext too short")
	}
	nonce, ct := sealed[:gcm.NonceSize()], sealed[gcm.NonceSize():]
	return gcm.Open(nil, nonce, ct, nil)
	// @starter
	// return nil, errors.New("TODO")
	// @end
}

func TestHandshakeAndEncrypt(t *testing.T) {

1. `arrivals[i]` 个新连接进入队列，队列已满（达到 `backlog`）的部分被丢弃；
2. 应用最多 accept `acceptRate` 个连接。

--- solution
package main

func Simulate(arrivals []int, acceptRate, backlog int) (accepted, dropped, maxQueue int) {
	// @solution
	queue := 0
	for _, a := range arrivals {
		room := backlog - queue
		in := min(a, room)
		queue += in
		dropped += a - in
		maxQueue = max(maxQueue, queue)
		n := min(queue, acceptRate)
		queue -= n
		accepted += n
	}
	return accepted, dropped, maxQueue
	// @starter
	// return 0, 0, 0 // TODO
	// @end
}

gRPC 的流式调用在 Go 里表现为 `Send` / `Recv` 方法。这里用函数参数模拟，实现三种模式的服务端逻辑：

- **服务端流** `Factorize(n, send)`：把 n 分解质因数，从小到大逐个 `send`；`send` 返回错误（客户端断开）时立即停止并返回该错误；

import (
	"errors"
	"io"

func Factorize(n int, send func(int) error) error {
	// @solution
	for p := 2; p*p <= n; p++ {
		for n%p == 0 {
			if err := send(p); err != nil {
				return err
			}
			n /= p
		}
	}
	if n > 1 {
		return send(n)
	}
	return nil
	// @starter
	// return errors.New("TODO")
	// @end
}

	// @solution
	sum := 0
	for {
		v, err := recv()
		if errors.Is(err, io.EOF) {
			return sum, nil
		}
		if err != nil {
			return 0, err
		}
		sum += v
	}
	// @starter
	// return 0, errors.New("TODO")
	// @end
}

--- desc
protobuf 用 varint 编码整数：每个字节低 7 位存数据，最高位为 1 表示后面还有字节，低位在前。实现：

- `AppendUvarint(dst, v)`：把 v 追加到 dst，例如 300 → `AC 02`；

func AppendUvarint(dst []byte, v uint64) []byte {
	// @solution
	for v >= 0x80 {
		dst = append(dst, byte(v)|0x80)
		v >>= 7
	}
	return append(dst, byte(v))
	// @starter
	// return dst // TODO
	// @end

func Uvarint(b []byte) (uint64, int) {
	// @solution
	var v uint64
	var shift uint
	for i, c := range b {
		if i == 10 {
			return 0, 0
		}
		v |= uint64(c&0x7f) << shift
		if c < 0x80 {
			return v, i + 1
		}
		shift += 7
	}
	return 0, 0
	// @starter
	// return 0, 0 // TODO
	// @end
}

func ZigZag(v int64) uint64 {
	// @solution

	// return uint64(v) // TODO
	// @end
}

--- desc
protobuf 的每个字段编码为 `tag + 值`，`tag = 字段编号 << 3 | wire type`。只考虑两种 wire type：

- 0：varint（整数）；
- 2：length-delimited（字符串、bytes、嵌套消息），格式为 `varint 长度 + 内容`。

- `Decode(b)`：解析出所有字段。遇到不认识的 wire type 或数据截断返回错误。

import (
	"encoding/binary"
	"errors"
)

type Field struct {
	Num    int
	Wire   int
	Varint uint64
	Bytes  []byte
}

func AppendVarintField(dst []byte, num int, v uint64) []byte {
	// @solution
	dst = binary.AppendUvarint(dst, uint64(num)<<3|0)
	return binary.AppendUvarint(dst, v)
	// @starter
	// return dst // TODO
	// @end
}

func AppendBytesField(dst []byte, num int, b []byte) []byte {
	// @solution
	dst = binary.AppendUvarint(dst, uint64(num)<<3|2)
	dst = binary.AppendUvarint(dst, uint64(len(b)))
	return append(dst, b...)
	// @starter
	// return dst // TODO
	// @end
}

func Decode(b []byte) ([]Field, error) {
	var fields []Field
	// @solution
	for len(b) > 0 {
		tag, n := binary.Uvarint(b)
		if n <= 0 {
			return nil, errors.New("bad tag")
		}
		b = b[n:]
		f := Field{Num: int(tag >> 3), Wire: int(tag & 7)}
		switch f.Wire {
		case 0:
			v, n := binary.Uvarint(b)
			if n <= 0 {
				return nil, errors.New("bad varint")
			}
			f.Varint, b = v, b[n:]
		case 2:
			l, n := binary.Uvarint(b)
			if n <= 0 || uint64(len(b)-n) < l {
				return nil, errors.New("truncated field")
			}
			f.Bytes, b = b[n:n+int(l)], b[n+int(l):]
		default:
			return nil, errors.New("unsupported wire type")
		}
		fields = append(fields, f)
	}
	// @starter
	// // TODO
	// @end
	return fields, nil
}

	"fmt"

type Handler func(ctx context.Context, req any) (any, error)

type Interceptor func(ctx context.Context, req any, method string, next Handler) (any, error)

func Chain(method string, h Handler, ics ...Interceptor) Handler {
	// @solution
	for i := len(ics) - 1; i >= 0; i-- {
		ic, next := ics[i], h
		h = func(ctx context.Context, req any) (any, error) {
			return ic(ctx, req, method, next)
		}
	}
	return h
	// @starter

	// @end

import (
	"errors"
	"sync"
	"sync/atomic"
)

var ErrNoAddr = errors.New("no healthy address")

type Picker struct {
	mu      sync.RWMutex
	addrs   []string
	healthy map[string]bool
	ready   []string // 健康地址的缓存
	next    atomic.Uint64

func (p *Picker) rebuild() {
	p.ready = p.ready[:0:0]
	for _, a := range p.addrs {
		if p.healthy[a] {
			p.ready = append(p.ready, a)
		}
	}
}

	p.mu.Lock()
	defer p.mu.Unlock()
	p.addrs = append([]string(nil), addrs...)
	p.healthy = map[string]bool{}
	for _, a := range addrs {
		p.healthy[a] = true
	}
	p.rebuild()
}

	// @solution
	p.mu.Lock()
	defer p.mu.Unlock()
	if _, known := p.healthy[addr]; known {
		p.healthy[addr] = ok
		p.rebuild()
	}

	// // TODO
	// @end

	defer p.mu.RUnlock()

		return "", ErrNoAddr

title: 重试策略与重试预算
--- desc
重试用得不好会放大故障（重试风暴）。实现：

- `Retryable(err, idempotent)`：只有**幂等**请求、并且错误链上是 `ErrTimeout` 或 `ErrUnavailable` 时才可以重试；
- `Budget`：重试预算，重试次数不能超过请求次数的 `percent`%。`OnRequest()` 每个请求调用一次；`AllowRetry()` 在 `(retries+1) × 100 ≤ percent × requests` 时允许并计数（用整数运算）；
- `Call(b, maxAttempts, idempotent, fn)`：调用 `b.OnRequest()` 后执行 `fn`，失败且可重试、且预算允许时重试，总尝试次数不超过 `maxAttempts`，返回最后一次的错误。

测试模拟下游完全不可用：100 个请求、每个最多尝试 3 次、预算 10%，总调用次数必须是 110 而不是 300。
--- solution
package main

import "errors"

var (
	ErrTimeout     = errors.New("timeout")
	ErrUnavailable = errors.New("unavailable")
)

func Retryable(err error, idempotent bool) bool {
	// @solution
	return idempotent && (errors.Is(err, ErrTimeout) || errors.Is(err, ErrUnavailable))
	// @starter
	// return err != nil // TODO
	// @end
}

type Budget struct {
	percent  int
	requests int
	retries  int
}

func NewBudget(percent int) *Budget { return &Budget{percent: percent} }

func (b *Budget) OnRequest() { b.requests++ }

func (b *Budget) AllowRetry() bool {
	// @solution
	if (b.retries+1)*100 > b.percent*b.requests {
		return false
	}
	b.retries++
	return true
	// @starter
	// return true // TODO
	// @end
}

func Call(b *Budget, maxAttempts int, idempotent bool, fn func() error) error {
	b.OnRequest()
	var err error
	// @solution
	for attempt := 1; ; attempt++ {
		err = fn()
		if err == nil || attempt >= maxAttempts || !Retryable(err, idempotent) || !b.AllowRetry() {
			return err
		}
	}
	// @starter
	// for attempt := 1; attempt <= maxAttempts; attempt++ { // TODO: 没有判断错误类型和预算
	// 	if err = fn(); err == nil {
	// 		return nil
	// 	}
	// }
	// return err
	// @end
}

- `Register(name, fn)`：注册方法，`fn` 接收原始的 `params` JSON；
- `Handle(data)`：处理一个请求 `{"id":1,"method":"add","params":[1,2]}`，返回响应 JSON：
  - 成功：`{"id":1,"result":3}`；
  - 请求解析失败：`{"id":0,"error":"parse error"}`；
  - 方法不存在：`{"id":1,"error":"method not found"}`；
  - 方法返回错误：`{"id":1,"error":"<错误信息>"}`；
  - 方法 panic：`{"id":1,"error":"internal error"}`，服务不能崩溃。
--- solution
package main

import (
	"encoding/json"
	"fmt"
)

type Request struct {
	ID     int             `json:"id"`
	Method string          `json:"method"`
	Params json.RawMessage `json:"params"`
}

type Response struct {
	ID     int    `json:"id"`
	Result any    `json:"result,omitempty"`
	Error  string `json:"error,omitempty"`
}

type Server struct {
	methods map[string]func(json.RawMessage) (any, error)
}

func NewServer() *Server {
	return &Server{methods: map[string]func(json.RawMessage) (any, error){}}
}

func (s *Server) Register(name string, fn func(json.RawMessage) (any, error)) {
	s.methods[name] = fn
}

func (s *Server) Handle(data []byte) []byte {
	// @solution
	var req Request
	if err := json.Unmarshal(data, &req); err != nil {
		return encode(Response{Error: "parse error"})
	}
	fn, ok := s.methods[req.Method]
	if !ok {
		return encode(Response{ID: req.ID, Error: "method not found"})
	}
	result, err := call(fn, req.Params)
	if err != nil {
		return encode(Response{ID: req.ID, Error: err.Error()})
	}
	return encode(Response{ID: req.ID, Result: result})
	// @starter
	// return encode(Response{Error: "TODO"})
	// @end
}

// @solution
func call(fn func(json.RawMessage) (any, error), params json.RawMessage) (result any, err error) {
	defer func() {
		if r := recover(); r != nil {
			result, err = nil, fmt.Errorf("internal error")
		}
	}()
	return fn(params)
}

// @end

func encode(r Response) []byte {
	b, _ := json.Marshal(r)
	return b
}

func headerHasToken(h http.Header, name, token string) bool {

--- desc
WebSocket 服务通常给每个连接一个带缓冲的发送 channel，由该连接的写 goroutine 消费。实现 `Hub`：

- `Register(c)` / `Unregister(id)`：注销时关闭这个客户端的 `Send` channel（写 goroutine 会因此退出）；
- `Broadcast(msg)`：**非阻塞地**发给所有客户端，返回成功投递的数量。某个客户端的缓冲已满（慢消费者），就把它注销——不能因为一个慢客户端阻塞整个广播；
- `Len()`：当前客户端数。

注意：不要在持有读锁时调用需要写锁的 `Unregister`，先收集要踢掉的客户端，释放锁后再处理。
--- solution
package main

import "sync"

type Client struct {
	ID   string
	Send chan []byte
}

type Hub struct {
	mu      sync.RWMutex
	clients map[string]*Client
}

func NewHub() *Hub { return &Hub{clients: map[string]*Client{}} }

func (h *Hub) Register(c *Client) {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.clients[c.ID] = c
}

func (h *Hub) Unregister(id string) {
	// @solution
	h.mu.Lock()
	defer h.mu.Unlock()
	if c, ok := h.clients[id]; ok {
		delete(h.clients, id)
		close(c.Send)
	}
	// @starter
	// // TODO
	// @end
}

func (h *Hub) Broadcast(msg []byte) int {
	// @solution
	var slow []string
	delivered := 0
	h.mu.RLock()
	for id, c := range h.clients {
		select {
		case c.Send <- msg:
			delivered++
		default:
			slow = append(slow, id)
		}
	}
	h.mu.RUnlock()
	for _, id := range slow {
		h.Unregister(id)
	}
	return delivered
	// @starter
	// h.mu.RLock()
	// defer h.mu.RUnlock()
	// for _, c := range h.clients {
	// 	c.Send <- msg // TODO: 慢客户端会阻塞整个广播
	// }
	// return len(h.clients)
	// @end
}

func (h *Hub) Len() int {
	h.mu.RLock()
	defer h.mu.RUnlock()
	return len(h.clients)
}

--- desc
Kafka 消费者组内，每个分区只能被一个消费者消费，所以**消费者数量超过分区数时，多出来的消费者是空闲的**。实现 Range 分配策略 `RangeAssign(partitions, consumers)`：

package main

import "slices"

func RangeAssign(partitions int, consumers []string) map[string][]int {
	out := map[string][]int{}
	// @solution
	sorted := slices.Sorted(slices.Values(consumers))
	n := len(sorted)
	if n == 0 {
		return out
	}
	per, extra := partitions/n, partitions%n
	next := 0
	for i, c := range sorted {
		count := per
		if i < extra {
			count++
		}
		ps := []int{}
		for j := 0; j < count; j++ {
			ps = append(ps, next)
			next++
		}
		out[c] = ps
	}

	// @end
	return out
}

--- desc
Kafka 默认是「至少一次」：消费者处理完消息、提交位点之前崩溃，重启后会再次收到这些消息。所以消费端要做到：

2. 用消息的业务 ID 去重：已经处理过的消息直接跳过（但仍然推进位点）。

package main

type Msg struct {
	Offset int64
	ID     string
}

type Consumer struct {
	Committed int64
	processed map[string]bool
	handle    func(Msg) error
}

func NewConsumer(handle func(Msg) error) *Consumer {
	return &Consumer{processed: map[string]bool{}, handle: handle}
}

func (c *Consumer) Poll(batch []Msg) error {
	for _, m := range batch {
		// @solution
		if !c.processed[m.ID] {
			if err := c.handle(m); err != nil {
				return err
			}
			c.processed[m.ID] = true
		}
		c.Committed = m.Offset + 1
		// @starter
		// c.Committed = m.Offset + 1 // TODO: 先提交后处理，处理失败消息就丢了；也没有去重
		// if err := c.handle(m); err != nil {
		// 	return err
		// }
		// @end
	}
	return nil
}

title: 估算积压消化时间
--- desc
消息积压时先算一笔账：加多少消费者、要多久才能追上？实现 `CatchUp(lag, produceRate, perConsumer, consumers, partitions)`（速率单位：条/秒）：

1. 有效消费者数 = `min(consumers, partitions)`——超过分区数的消费者是空闲的；
2. 净消化速率 = 有效消费者数 × `perConsumer` − `produceRate`；
3. 净速率 ≤ 0 时永远追不上，返回 `ErrNeverCatchUp`；
4. 否则返回 `lag / 净速率` 秒（向上取整到整秒）。
--- solution
package main

import (
	"errors"
	"time"
)

var ErrNeverCatchUp = errors.New("consumers can never catch up")

func CatchUp(lag, produceRate, perConsumer int64, consumers, partitions int) (time.Duration, error) {
	// @solution
	effective := int64(min(consumers, partitions))
	net := effective*perConsumer - produceRate
	if net <= 0 {
		return 0, ErrNeverCatchUp
	}
	secs := (lag + net - 1) / net
	return time.Duration(secs) * time.Second, nil
	// @starter
	// return 0, ErrNeverCatchUp // TODO
	// @end
}

title: 并发处理时安全地提交位点
race: true
--- desc
为了加快消费，常在一个分区内用多个 goroutine 并发处理消息，但它们会**乱序完成**。如果直接提交「最新完成的 offset」，而前面的消息还没处理完，进程一崩溃那些消息就丢了。

实现 `Tracker`：

- `MarkDone(offset)`：标记某条消息处理完成（可能乱序、并发调用）；
- `Committable()`：返回可以安全提交的位点——它之前的所有消息都已完成。初始值为构造时传入的起始 offset。
--- solution
package main

import "sync"

type Tracker struct {
	mu   sync.Mutex
	next int64
	done map[int64]bool
}

func NewTracker(start int64) *Tracker { return &Tracker{next: start, done: map[int64]bool{}} }

func (t *Tracker) MarkDone(offset int64) {
	// @solution
	t.mu.Lock()
	defer t.mu.Unlock()
	if offset < t.next {
		return
	}
	t.done[offset] = true
	for t.done[t.next] {
		delete(t.done, t.next)
		t.next++
	}
	// @starter
	// t.next = offset + 1 // TODO: 乱序完成时会跳过还没处理完的消息
	// @end
}

func (t *Tracker) Committable() int64 {
	t.mu.Lock()
	defer t.mu.Unlock()
	return t.next
}
